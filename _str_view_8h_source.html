203 virtual bool matches(const char* pattern) {
204 if (pattern == nullptr) return false;
205 // a completely empty pattern is a single (trivial) alternative that
206 // only matches an empty line - handle it before the splitting loop
207 // below, which assumes at least one non-empty alternative.
208 if (*pattern == '\0') return matchesGlob(this->chars, "");
209 const char* alt_start = pattern;
210 for (;;) {
211 // skip leading separators/whitespace
212 while (*alt_start == ',' || *alt_start == ';' || *alt_start == '|' ||
213 *alt_start == ' ') {
214 alt_start++;
215 }
216 if (*alt_start == '\0') return false;
217
218 // find the end of this alternative
219 const char* alt_end = alt_start;
220 while (*alt_end != '\0' && *alt_end != ',' && *alt_end != ';' &&
221 *alt_end != '|') {
222 alt_end++;
223 }
224 // trim trailing whitespace
225 const char* trimmed_end = alt_end;
226 while (trimmed_end > alt_start && *(trimmed_end - 1) == ' ') {
227 trimmed_end--;
228 }
229
230 size_t alt_len = trimmed_end - alt_start;
231 if (alt_len > 0) {
232 char buffer[64];
233 if (alt_len >= sizeof(buffer)) alt_len = sizeof(buffer) - 1;
234 memcpy(buffer, alt_start, alt_len);
235 buffer[alt_len] = '\0';
236 if (matchesGlob(this->chars, buffer)) return true;
237 }
238
239 if (*alt_end == '\0') return false;
240 alt_start = alt_end + 1;
241 }
242 }

247 static bool matchesGlob(const char* line, const char* pattern) {
250 int wildcard = 0;
251
252 const char* last_pattern_start = 0;
253 const char* last_line_start = 0;
254 while (*line) {
255 if (*pattern == *line) {
256 if (wildcard == 1) last_line_start = line + 1;
257
258 line++;
259 pattern++;
260 wildcard = 0;
261 } else if (*pattern == '?') {
262 if (*(line) == '\0')
263 return 0;
264 if (wildcard == 1) last_line_start = line + 1;
265 line++;
266 pattern++;
267 wildcard = 0;
268 } else if (*pattern == '*') {
269 if (*(pattern + 1) == '\0') {
270 return 1;
271 }
272
273 last_pattern_start = pattern;
274 // last_line_start = line + 1;
275 wildcard = 1;
276
277 pattern++;
278 } else if (wildcard) {
279 if (*line == *pattern) {
280 wildcard = 0;
281 line++;
282 pattern++;
283 last_line_start = line + 1;
284 } else {
285 line++;
286 }
287 } else if (last_pattern_start != 0) {
288 pattern = last_pattern_start;
289 line = last_line_start;
290 last_line_start = 0;
291 } else {
292 return false;
293 }
294 }
295

823
825 double roundingFactor = 0.5;
826 unsigned long mult = 1;
827 for (i = 0; i < precision; i++) {
828 roundingFactor /= 10.0;
829 mult *= 10;

834

844 strcat(outstr, ".\0");

865 int J = 0;
