253 void audioTaskLoop() {
254 if (p_audio != nullptr) {
255 uint8_t buf[kAudioReadChunkSize];
256 size_t n = p_audio->readBytes(buf, sizeof(buf));
257 if (n > 0) {
258 LockGuard guard(p_mutex);
259 p_muxer->setStreamType(StreamContentType::Audio);
260 if (p_audio_encoder != nullptr) {
261 p_audio_encoder->write(buf, n);
262 } else {
263 p_muxer->addAudioFrame(buf, n);
264 }
265 } else {
266 delay(1);
267 }
268 }
269 }
270
275 void videoTaskLoop() {
276 uint32_t start = millis();
277 if (p_video != nullptr && p_muxer != nullptr) {
278 size_t len = 0;
279 const uint8_t *data = p_video->nextFrame(len);
280 if (data != nullptr && len > 0) {
281 LockGuard guard(p_mutex);
282 p_muxer->setStreamType(StreamContentType::Video);
283 if (p_video_encoder != nullptr) {
284 p_video_encoder->write(data, len);
285 } else {
286 bool is_key = p_muxer->getVideoInfo().format == VideoFormat::H264
287 ? isH264KeyFrame(data, len)
288 : true;
289 p_muxer->addVideoFrame(data, len, is_key);
290 }
291 }
292 }
293 uint32_t elapsed = millis() - start;
294 if (elapsed < video_interval_ms) delay(video_interval_ms - elapsed);
295 }
