154 static void decodeAudio(const M4AAudioDemuxer::Frame& frame, void* ref) {
155 ContainerM4A* self = static_cast<ContainerM4A*>(ref);
156 if (self->p_decoder == nullptr) {
157 self->p_print->write(frame.data, frame.size);
158 return;
159 }
160 MultiDecoder& dec = *(self->p_decoder);
161 const char* old_mime = dec.selectedMime();
162
163 // select decoder based on mime type
164 if (!dec.selectDecoder(frame.mime)) {
165 const char* mime = frame.mime ? frame.mime : "(nullptr)";
166 LOGE("No decoder found for mime type: %s", mime);
167 return;
168 }
169
170 // for ALAC only: process magic cookie if not done yet
171 if (StrView(frame.mime) == "audio/alac" &&
172 !self->is_magic_cookie_processed) {
173 auto& magic_cookie = self->demux.getALACMagicCookie();
174 if (magic_cookie.size() > 0) {
175 if (!dec.setCodecConfig(magic_cookie.data(), magic_cookie.size())) {
176 LOGE("Failed to set ALAC magic cookie for decoder: %s",
177 dec.selectedMime());
178 }
179 }
180 self->is_magic_cookie_processed = true;
181 }
182 // write encoded data to decoder
183 dec.write(frame.data, frame.size);
184

186 dec.selectDecoder(old_mime);
