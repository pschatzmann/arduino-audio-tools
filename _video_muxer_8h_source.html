206 size_t copyAudio() {
207 uint8_t buf[kAudioReadChunkSize];
208 size_t n = p_audio->readBytes(buf, sizeof(buf));
209 if (n == 0) return 0;
210
211 p_muxer->setStreamType(StreamContentType::Audio);
212 if (p_audio_encoder != nullptr) {
213 p_audio_encoder->write(buf, n);
214 } else {
215 p_muxer->addAudioFrame(buf, n);
216 }
217
218 if (bytes_per_sample > 0) audio_samples_written += n / bytes_per_sample;
219
220 // catch up: mux/encode every video frame now due, given the audio
221 // time just reached - this, not a timer, is what paces video. Strict
222 // '<' (not '<=') so a duration landing exactly on a frame boundary
223 // doesn't pull one extra frame for the instant at that boundary.
224 if (have_video && sample_rate > 0) {
225 double audio_time_s = (double)audio_samples_written / sample_rate;
226 while ((double)video_frame_count / video_fps < audio_time_s) {
227 if (!copyVideo()) break; // source exhausted
228 }
229 }
230 return n;
231 }
232
233 bool copyVideo() {
234 size_t len = 0;
235 const uint8_t* data = p_video->nextFrame(len);
236 if (data == nullptr || len == 0) return false;
237
238 p_muxer->setStreamType(StreamContentType::Video);
239 if (p_video_encoder != nullptr) {
240 p_video_encoder->write(data, len);
241 } else {
242 bool is_key = p_muxer->getVideoInfo().format == VideoFormat::H264
243 ? isH264KeyFrame(data, len)
244 : true;
245 p_muxer->addVideoFrame(data, len, is_key);
246 }
247 video_frame_count++;
248 return true;
249 }
250
256 bool copyVideoPaced() {
257 uint32_t start = millis();
258 if (!copyVideo()) return false;
259 uint32_t elapsed = millis() - start;
260 if (elapsed < video_interval_ms) delay(video_interval_ms - elapsed);
261 return true;
262 }
