42class BLESerialPacketBuffer {
43 public:
44 void setMutex(MutexBase *mutex) { p_mutex = mutex; }
45
47 void resize(int bytes, int packets, bool framed) {
48 LockGuard guard(p_mutex);
49 is_framed = framed;
50 if (data.size() != (size_t)bytes) data.resize(bytes);
51 if (is_framed && sizes.size() != (size_t)packets) sizes.resize(packets);
52 }
53
54 bool isAllocated() { return data.size() > 0; }
55
57 void clear() {
58 LockGuard guard(p_mutex);
59 data.reset();
60 sizes.reset();
61 }
62
64 int write(const uint8_t *packet, int len) {
65 LockGuard guard(p_mutex);
66 if (!is_framed) return data.writeArray(packet, len);
67 if (sizes.availableForWrite() == 0 || data.availableForWrite() < len)
68 return 0;
69 sizes.write(len);
70 return data.writeArray(packet, len);
71 }
72
75 int read(uint8_t *result, int len) {
76 LockGuard guard(p_mutex);
77 if (!is_framed) return data.readArray(result, len);
78 uint16_t size = 0;
79 if (!sizes.peek(size)) return 0;
80 if (size > len) {
81 LOGE("BLESerial: read size %d too small: it must be >= %d", len, size);
82 return 0;
83 }
84 sizes.read(size);
85 return data.readArray(result, size);
86 }
87
89 int available() {
90 LockGuard guard(p_mutex);
91 if (!is_framed) return data.available();
92 uint16_t size = 0;
93 return sizes.peek(size) ? size : 0;
94 }
95
97 int availableForWrite() {
98 LockGuard guard(p_mutex);
99 if (is_framed && sizes.availableForWrite() == 0) return 0;
100 return data.availableForWrite();
101 }
102
103 protected:
104 RingBuffer<uint8_t> data{0};
105 RingBuffer<uint16_t> sizes{0};
106 MutexBase *p_mutex = nullptr;
107 bool is_framed = false;
108};

330 void setupBuffers() {
331 if (!rx_buffer.isAllocated()) {
332 LOGI("BLESerial: buffers %d / %d bytes", rx_buffer_size, tx_buffer_size);
333 rx_buffer.resize(rx_buffer_size, rx_packet_count, is_framed);
334 tx_buffer.resize(tx_buffer_size, tx_packet_count, is_framed);
335 }
336 }
337
339 void receiveData(const uint8_t *data, size_t len) {
340 if (len == 0) return;
341 setupBuffers();
342 uint32_t start = millis();
343 while (rx_buffer.availableForWrite() < (int)len) {
344 if (millis() - start >= (uint32_t)receive_timeout_ms) {
345 LOGW("BLESerial: receive buffer full: %d bytes lost", (int)len);
346 return;
347 }
348 // wait for the buffer to get freed up
349 delay(5);
350 }
351 rx_buffer.write(data, len);
352 }
353
356 int provideData(uint8_t *packet, int maxLen) {
357 setupBuffers();
358 int max = maxLen - BLE_SERIAL_HEADER_SIZE;
359 if (max > packetSize()) max = packetSize();
360 int len = 0;
361 if (max > 0 && tx_buffer.available() > 0) {
362 len = tx_buffer.read(packet + BLE_SERIAL_HEADER_SIZE, max);
363 }
364 setHeader(packet, len);
365 return len + BLE_SERIAL_HEADER_SIZE;
366 }
367
369 void onDisconnected() {
370 // do not send outdated data to the next client
371 tx_buffer.clear();
372 }

412 int available() override {

418
419 size_t write(const uint8_t *data, size_t len) override {
420 TRACED();
421 if (!connected()) return 0;
422 if (is_framed) {
423 if ((int)len > packetSize()) {
424 LOGE("BLESerial: frame of %d bytes > max %d", (int)len, packetSize());
425 return 0;
426 }
427 return sendPacket(data, len) ? len : 0;
428 }
429 // send only packets with the max size
430 for (size_t j = 0; j < len; j++) {
431 write_buffer.write(data[j]);
432 if (write_buffer.isFull()) {
433 if (!sendPacket(write_buffer.data(), write_buffer.available())) {
434 write_buffer.reset();
435 return j + 1;
436 }
437 write_buffer.reset();
438 }
439 }
440 return len;
441 }
442
444 void flush() override {
445 if (write_buffer.available() > 0 && connected()) {
446 sendPacket(write_buffer.data(), write_buffer.available());
447 }
448 write_buffer.reset();
449 }

451 int availableForWrite() override {
452 if (!connected()) return 0;
453 return is_framed ? packetSize() : DEFAULT_BUFFER_SIZE;
454 }

506 }
507 }
508 }

511 void receivePacket() {
512 int len = readPacket(read_packet.data(), read_packet.size());
513 int payload = getHeader(read_packet.data(), len);
514 if (payload > 0) {
515 rx_buffer.write(read_packet.data() + BLE_SERIAL_HEADER_SIZE, payload);
516 }
517 }
518

520 bool rc = writePacket(data, len);
521 if (write_throttle > 0) delay(write_throttle);
522 return rc;
523 }
