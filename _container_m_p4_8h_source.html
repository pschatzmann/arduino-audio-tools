622 void onAvcC(const MP4Parser::Box& box) {
623 if (current_track == nullptr || !box.is_complete || box.data_size < 6)
624 return;
625 const uint8_t* d = box.data;
626 current_track->nal_length_size = (d[4] & 0x03) + 1;
627 int numSps = d[5] & 0x1F;
628 size_t pos = 6;
629 auto appendUnit = [&](const uint8_t* unit, size_t len) {
630 static const uint8_t startCode[4] = {0, 0, 0, 1};
631 auto& v = current_track->sps_pps_annexb;
632 size_t off = v.size();
633 v.resize(off + 4 + len);
634 memcpy(v.data() + off, startCode, 4);
635 memcpy(v.data() + off + 4, unit, len);
636 };
637 for (int i = 0; i < numSps && pos + 2 <= box.data_size; i++) {
638 uint16_t len = readU16(d + pos);
639 pos += 2;
640 if (pos + len > box.data_size) break;
641 appendUnit(d + pos, len);
642 pos += len;
643 }
644 if (pos >= box.data_size) return;
645 int numPps = d[pos++];
646 for (int i = 0; i < numPps && pos + 2 <= box.data_size; i++) {
647 uint16_t len = readU16(d + pos);
648 pos += 2;
649 if (pos + len > box.data_size) break;
650 appendUnit(d + pos, len);
651 pos += len;
652 }

700 void onStsz(MP4Parser::Box& box) {
701 if (current_track == nullptr) return;
702 if (beginBoxAccum(box))
703 stsz_header_pending = true; // starting a fresh stsz box
704 box_accum.writeArray(box.data, box.available);
705
706 // The 12-byte header (version+flags, sampleSize, sampleCount) must be
707 // fully consumed *once* before any bytes are interpreted as per-sample
708 // sizes. Using "no sizes pushed yet" as a stand-in for "header not
709 // parsed yet" (as tried previously) is wrong: in the variable-size
710 // case, both are simultaneously true right after the header IS parsed,
711 // so under fine-grained incremental delivery the per-sample loop below
712 // could start consuming still-unparsed header bytes as if they were
713 // sample sizes. An explicit flag avoids that ambiguity.
714 if (stsz_header_pending) {
715 if (box_accum.available() < 12)
716 return; // wait for the rest of the header
717 Track* t = current_track;
718 uint32_t sampleSize = readU32(box_accum.data() + 4);
719 uint32_t sampleCount = readU32(box_accum.data() + 8);
720 box_accum.clearArray(12);
721 stsz_header_pending = false;
722 if (sampleSize != 0) {
723 t->fixed_sample_size = sampleSize;
724 t->fixed_sample_count = sampleCount;
725 }
726 // else: sample_sizes is filled below via push_back, one entry at a
727 // time as bytes arrive - do NOT pre-resize it (Vector::push_back
728 // always appends, unlike BaseBuffer::write() which fills pre-sized
729 // capacity in place)
730 }
731 // incrementally consume per-sample sizes (fixed-size case has none left
732 // to read; the table is only present in the box when sampleSize==0)
733 if (current_track->fixed_sample_size == 0) {
734 while (box_accum.available() >= 4) {
735 current_track->sample_sizes.push_back(readU32(box_accum.data()));
736 box_accum.clearArray(4);
737 }
738 }
739 if (box.is_complete) endBoxAccum();
740 }

777 uint64_t off = is64 ? readU64(p + i * 8) : readU32(p + i * 4);
778 current_track->chunk_offsets.push_back(off);
779 }
780 endBoxAccum();

795 void buildSchedule() {
796 // (track*, chunk_index, offset) working list
797 struct Item {
798 Track* track;
799 uint32_t chunk_index;
800 uint64_t offset;
801 };
802 Vector<Item> items;
803 for (size_t ti = 0; ti < tracks.size(); ti++) {
804 Track* t = tracks[ti];
805 if (t->kind == Track::Kind::Unknown) continue;
806 if (t->kind == Track::Kind::Video && t != p_video_track) continue;
807 if (t->kind == Track::Kind::Audio && t != p_audio_track) continue;
808 for (size_t ci = 0; ci < t->chunk_offsets.size(); ci++) {
809 items.push_back({t, (uint32_t)ci, t->chunk_offsets[ci]});
810 }
811 }
812 // simple insertion sort by offset (small N: chunk count is typically
813 // well under a few thousand for embedded-relevant clip lengths)
814 for (size_t i = 1; i < items.size(); i++) {
815 Item key = items[i];
816 long j = (long)i - 1;
817 while (j >= 0 && items[j].offset > key.offset) {
818 items[j + 1] = items[j];
819 j--;
820 }
821 items[j + 1] = key;
822 }
823 for (size_t i = 0; i < items.size(); i++) {
824 Track* t = items[i].track;
825 uint32_t samples = samplesInChunk(*t, items[i].chunk_index);
826 if (samples > 0) schedule.push_back({t, samples});
827 }
828 schedule_index = 0;
829 samples_left_in_entry = schedule.size() > 0 ? schedule[0].sample_count : 0;
830 }
831
833 uint32_t samplesInChunk(Track& t, uint32_t chunkIndex0) {
834 uint32_t chunk1 = chunkIndex0 + 1; // stsc uses 1-based chunk numbers
835 uint32_t result = 0;
836 for (size_t i = 0; i < t.stsc.size(); i++) {
837 if (t.stsc[i].first_chunk <= chunk1) {
838 result = t.stsc[i].samples_per_chunk;
839 } else {
840 break;
841 }
842 }
843 return result;
844 }
845
847 void advanceScheduleIfNeeded() {
848 while (samples_left_in_entry == 0 && schedule_index < schedule.size()) {
849 schedule_index++;
850 samples_left_in_entry = schedule_index < schedule.size()
851 ? schedule[schedule_index].sample_count
852 : 0;
853 }
854 }
855
859 void feed(const uint8_t* data, size_t len) {
860 size_t pos = 0;
861 while (pos < len) {
862 if (current_sample_track == nullptr) {
863 advanceScheduleIfNeeded();
864 if (schedule_index >= schedule.size()) return; // nothing more expected
865 Track* t = schedule[schedule_index].track;
866 uint32_t size = t->sampleSize(t->next_sample_index);
867 if (size == 0) {
868 // no more sizes for this track - stop
869 samples_left_in_entry = 0;
870 continue;
871 }
872 current_sample_track = t;
873 current_sample_size = size;
874 current_sample_is_first_for_track = (t->next_sample_index == 0);
875 // Vector::resize() sets the logical size immediately (unlike a
876 // capacity reserve), so track how much of it is actually filled
877 // separately rather than relying on sample_buffer.size().
878 sample_buffer.resize(size);
879 sample_buffer_filled = 0;
880 }
881
882 size_t need = current_sample_size - sample_buffer_filled;
883 size_t take = std::min(need, len - pos);
884 memcpy(sample_buffer.data() + sample_buffer_filled, data + pos, take);
885 sample_buffer_filled += take;
886 pos += take;
887
888 if (sample_buffer_filled >= current_sample_size) {
889 dispatchSample(*current_sample_track, sample_buffer.data(),
890 sample_buffer_filled, current_sample_is_first_for_track);
891 current_sample_track->next_sample_index++;
892 samples_left_in_entry--;
893 current_sample_track = nullptr;
894 }
895 }

898 void dispatchSample(Track& t, const uint8_t* data, size_t size,
899 bool isFirst) {
900 if (&t == p_video_track) {
901 dispatchVideo(t, data, size, isFirst);
902 } else if (&t == p_audio_track) {
903 dispatchAudio(t, data, size, isFirst);
904 }

906
907 void dispatchVideo(Track& t, const uint8_t* data, size_t size, bool isFirst) {
908 // nextPtsTicks() must run unconditionally, in sample order, to keep
909 // the track's timing cursor correct regardless of what we do below -
910 // call it before any early return.
911 uint64_t ptsTicks = t.nextPtsTicks();
912 if (p_video_out == nullptr || t.is_hevc_unsupported) return;
913
914 if (t.timescale > 0 && !t.stts.empty()) {
915 uint32_t scheduledMs = (uint32_t)((ptsTicks * 1000) / t.timescale);
916 uint32_t nowMs = millis();
917 if (!playback_start_set) {
918 playback_start_ms = nowMs;
919 playback_start_set = true;
920 }
921 uint32_t scheduledWallMs = playback_start_ms + scheduledMs;
922 if (nowMs > scheduledWallMs) {
923 LOGW("DemuxerMP4: video frame %u ms late - skipping",
924 (unsigned)(nowMs - scheduledWallMs));
925 return;
926 } else if (nowMs < scheduledWallMs) {
927 LOGI("DemuxerMP4: video frame %u ms early",
928 (unsigned)(scheduledWallMs - nowMs));
929 }
930 }
931
932 // convert AVCC length-prefixed NAL units to Annex-B start codes
933 nal_tmp.clear();
934 size_t pos = 0;
935 size_t lenSize = t.nal_length_size;
936 while (pos + lenSize <= size) {
937 uint32_t nalLen = 0;
938 for (size_t i = 0; i < lenSize; i++)
939 nalLen = (nalLen << 8) | data[pos + i];
940 pos += lenSize;
941 if (pos + nalLen > size) break;
942 size_t off = nal_tmp.size();
943 static const uint8_t startCode[4] = {0, 0, 0, 1};
944 nal_tmp.resize(off + 4 + nalLen);
945 memcpy(nal_tmp.data() + off, startCode, 4);
946 memcpy(nal_tmp.data() + off + 4, data + pos, nalLen);
947 pos += nalLen;
948 }
949
950 bool prependConfig =
951 isFirst && !t.avc_config_sent && t.sps_pps_annexb.size() > 0;
952 if (prependConfig) {
953 p_video_out->write(t.sps_pps_annexb.data(), t.sps_pps_annexb.size());
954 t.avc_config_sent = true;
955 }
956 if (nal_tmp.size() > 0) p_video_out->write(nal_tmp.data(), nal_tmp.size());
957 p_video_out->flush();
958 }
959
960 void dispatchAudio(Track& t, const uint8_t* data, size_t size, bool isFirst) {
961 if (p_output_audio == nullptr) return;
962 if (t.audio_codec == Codec::AAC) {
963 uint8_t adts[7];
964 writeAdtsHeader(adts, t.aacProfile, t.sampleRateIdx, t.channelCfg,
965 (int)size);
966 p_output_audio->write(adts, sizeof(adts));
967 p_output_audio->write(data, size);
968 } else {
969 // ALAC (magic cookie is exposed via audioALACMagicCookie() for the
970 // caller to configure their own decoder with) and any other codec:
971 // raw payload as-is.
972 p_output_audio->write(data, size);
973 }
974 }
975
976 static void writeAdtsHeader(uint8_t* adts, int aacProfile, int sampleRateIdx,
977 int channelCfg, int frameLen) {
978 adts[0] = 0xFF;
979 adts[1] = 0xF1;
980 adts[2] = ((aacProfile - 1) << 6) | (sampleRateIdx << 2) |
981 ((channelCfg >> 2) & 0x1);
982 adts[3] = ((channelCfg & 0x3) << 6) | ((frameLen + 7) >> 11);
983 adts[4] = ((frameLen + 7) >> 3) & 0xFF;
984 adts[5] = (((frameLen + 7) & 0x7) << 5) | 0x1F;
985 adts[6] = 0xFC;
986 }

1500 template <typename F>
1501 static void forEachAnnexBNal(const uint8_t* data, size_t len, F callback) {
1502 size_t i = 0;
1503 while (i + 3 <= len) {
1504 size_t scLen = 0;
1505 if (i + 4 <= len && data[i] == 0 && data[i + 1] == 0 &&
1506 data[i + 2] == 0 && data[i + 3] == 1) {
1507 scLen = 4;
1508 } else if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
1509 scLen = 3;
1510 }
1511 if (scLen == 0) {
1512 i++;
1513 continue;
1514 }
1515 size_t nalStart = i + scLen;
1516 size_t j = nalStart;
1517 size_t nextStart = len;
1518 while (j + 3 <= len) {
1519 if (data[j] == 0 && data[j + 1] == 0 &&
1520 (data[j + 2] == 1 ||
1521 (j + 3 < len && data[j + 2] == 0 && data[j + 3] == 1))) {
1522 nextStart = j;
1523 break;
1524 }
1525 j++;
1526 }
1527 if (nextStart > nalStart) callback(data + nalStart, nextStart - nalStart);
1528 i = nextStart;
1529 }
1530 }
