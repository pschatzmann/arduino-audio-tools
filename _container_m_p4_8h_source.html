330 uint64_t nextPtsTicks() {
331 uint64_t pts = next_pts_ticks;
332 if (stts_entry_idx < stts->size()) {
333 // fetch once - a single call regardless of store implementation
334 // (RAM: cheap either way; source-seek/spool: avoids a second
335 // seek+read for the same entry)

337 if (stts_entry_remaining == 0) stts_entry_remaining = entry.sample_count;

339 if (stts_entry_remaining > 0) {
340 stts_entry_remaining--;
341 if (stts_entry_remaining == 0) stts_entry_idx++;

439 bool beginBoxAccum(const MP4Parser::Box& box) {
440 if (box_accum_active) return false;
441 box_accum.resize(box.size);
442 box_accum.clear();
443 box_accum_active = true;
444 // box.file_offset is the absolute file offset of *this box's own*
445 // 8-byte header (size+type) - captured only here, on the box's FIRST
446 // delivery, because MP4Parser advances it per incremental delivery
447 // (see startIncrementalBox()/continueIncrementalBox() in
448 // MP4Parser.h): by the time a large box (e.g. a 147K-entry stco) is
449 // fully accumulated and its entries are actually iterated, box.file_
450 // offset reflects only the *last* delivery's start, not the box's -
451 // capturing it here instead of computing it later from
452 // total_bytes_received/box_accum.available() is what makes
453 // SourceSeekSampleTableStore's offsets land correctly.
454 box_start_file_offset = box.file_offset;

466 static uint32_t readU32(const uint8_t* p) {
467 return (p[0] << 24) | (p[1] << 16) | (p[2] << 8) | p[3];
468 }
469 static uint64_t readU64(const uint8_t* p) {
470 return ((uint64_t)readU32(p) << 32) | readU32(p + 4);
471 }
472 static uint16_t readU16(const uint8_t* p) { return (p[0] << 8) | p[1]; }

609 void onTrak() {
610 Track* t = new Track();
611 if (p_seek_source != nullptr) {
612 // RAM-backed defaults from Track's own member initializers aren't
613 // needed when a seek source is configured - swap them for
614 // SourceSeekSampleTableStore instead of paying for both.
615 delete t->sample_sizes;
616 t->sample_sizes = new SourceSeekSampleTableStore<uint32_t>(
617 *p_seek_source, sizeof(uint32_t), readU32);
618 delete t->chunk_offsets;
619 // on-disk width (4 bytes for 'stco', 8 for 'co64') isn't known yet
620 // here - defaults to 'stco' width and is corrected in onStco() via
621 // setOnDiskEntrySize() once the actual box type is seen, always
622 // before any get() call (which only happens much later, during
623 // 'mdat' playback).
624 t->chunk_offsets = new SourceSeekSampleTableStore<uint64_t>(
625 *p_seek_source, sizeof(uint32_t), readU32Widened);
626 delete t->stsc;
627 // stsc's on-disk entries are 12 bytes (first_chunk, samples_per_chunk,
628 // sample_description_index - all u32), matching onStsc()'s own i*12
629 // parsing stride below, even though readStscEntry() only decodes the
630 // first two fields into StscEntry. Passing 8 here previously
631 // (2 fields' worth) misaligned every get() past index 0, since
632 // get()'s offset math is entries*onDiskSize - reproducibly corrupted
633 // chunk-to-sample mapping for any track whose stsc has more than one
634 // entry (typically audio, not video, since a video track commonly
635 // only ever needs one stsc entry for the whole file).
636 t->stsc = new SourceSeekSampleTableStore<StscEntry>(
637 *p_seek_source, sizeof(uint32_t) * 3, readStscEntry);
638 delete t->stts;
639 t->stts = new SourceSeekSampleTableStore<SttsEntry>(
640 *p_seek_source, sizeof(uint32_t) * 2, readSttsEntry);
641 }
642 tracks.push_back(t);
643 current_track = t;
644 }

646 void onHdlr(MP4Parser::Box& box) {
647 if (current_track == nullptr) return;
648 beginBoxAccum(box);
649 box_accum.writeArray(box.data, box.available);
650 if (!box.is_complete || box_accum.available() < 12) return;
651 const uint8_t* handler = box_accum.data() + 8;
652 if (memcmp(handler, "vide", 4) == 0) {
653 current_track->kind = Track::Kind::Video;
654 if (p_video_track == nullptr) p_video_track = current_track;
655 } else if (memcmp(handler, "soun", 4) == 0) {
656 current_track->kind = Track::Kind::Audio;
657 if (p_audio_track == nullptr) p_audio_track = current_track;
658 }
659 // Spool-backed tables need to know the track's kind (to name/select
660 // spool files) - unlike source-seek, which is set up eagerly in
661 // onTrak() since it doesn't care about kind, this can only happen
662 // here, once 'hdlr' has actually told us what this track is.
663 if (p_spool_factory != nullptr && current_track->kind != Track::Kind::Unknown) {
664 configureSpoolStores(*current_track);
665 }
666 endBoxAccum();
667 }

921 void onStsz(MP4Parser::Box& box) {
922 if (current_track == nullptr) return;
923 if (beginBoxAccum(box))
924 stsz_header_pending = true; // starting a fresh stsz box
925 box_accum.writeArray(box.data, box.available);
926
927 // The 12-byte header (version+flags, sampleSize, sampleCount) must be
928 // fully consumed *once* before any bytes are interpreted as per-sample
929 // sizes. Using "no sizes pushed yet" as a stand-in for "header not
930 // parsed yet" (as tried previously) is wrong: in the variable-size
931 // case, both are simultaneously true right after the header IS parsed,
932 // so under fine-grained incremental delivery the per-sample loop below
933 // could start consuming still-unparsed header bytes as if they were
934 // sample sizes. An explicit flag avoids that ambiguity.
935 if (stsz_header_pending) {
936 if (box_accum.available() < 12)
937 return; // wait for the rest of the header
938 Track* t = current_track;
939 uint32_t sampleSize = readU32(box_accum.data() + 4);
940 uint32_t sampleCount = readU32(box_accum.data() + 8);
941 LOGD("DemuxerMP4: stsz sampleSize=%u sampleCount=%u", (unsigned)sampleSize,
942 (unsigned)sampleCount);
943 box_accum.clearArray(12);
944 stsz_header_pending = false;
945 if (sampleSize != 0) {
946 t->fixed_sample_size = sampleSize;
947 t->fixed_sample_count = sampleCount;
948 }
949 // else: sample_sizes is filled below via append(), one entry at a
950 // time as bytes arrive - do NOT pre-resize it (a RAM-backed store's
951 // Vector::push_back always appends, unlike BaseBuffer::write()
952 // which fills pre-sized capacity in place)
953 }
954 // incrementally consume per-sample sizes (fixed-size case has none left
955 // to read; the table is only present in the box when sampleSize==0)
956 if (current_track->fixed_sample_size == 0) {
957 while (box_accum.available() >= 4) {
958 // +12: this table's own version/flags+sampleSize+sampleCount
959 // sub-header (already drained from box_accum via clearArray(12)
960 // above, but boxPayloadFileOffset() itself doesn't know that)
961 current_track->sample_sizes->setNextEntryOffset(boxPayloadFileOffset() + 12);
962 current_track->sample_sizes->append(readU32(box_accum.data()));
963 box_accum.clearArray(4);
964 size_t n = current_track->sample_sizes->size();
965 if (n % 20000 == 0)
966 LOGD("DemuxerMP4: stsz sample_sizes progress: %u", (unsigned)n);
967 }
968 }
969 if (box.is_complete) {
970 LOGD("DemuxerMP4: stsz complete, %u sample_sizes",
971 (unsigned)current_track->sample_sizes->size());
972 endBoxAccum();
973 }
974 }

1067 void advanceChunkIfNeeded() {
1068 while (samples_left_in_chunk == 0) {
1069 Track* next = nullptr;
1070 uint64_t best_offset = 0;
1071 if (p_video_track != nullptr &&
1072 p_video_track->next_chunk_index < p_video_track->chunk_offsets->size()) {
1073 best_offset = p_video_track->chunk_offsets->get(p_video_track->next_chunk_index);
1074 next = p_video_track;
1075 }
1076 if (p_audio_track != nullptr &&
1077 p_audio_track->next_chunk_index < p_audio_track->chunk_offsets->size()) {
1078 uint64_t off = p_audio_track->chunk_offsets->get(p_audio_track->next_chunk_index);
1079 if (next == nullptr || off < best_offset) {
1080 best_offset = off;
1081 next = p_audio_track;
1082 }
1083 }
1084 if (next == nullptr) {
1085 current_chunk_track = nullptr; // nothing more expected
1086 return;
1087 }
1088 uint32_t samples = samplesInChunk(*next, next->next_chunk_index);
1089 next->next_chunk_index++;
1090 if (samples == 0) continue; // empty chunk entry - keep looking
1091 current_chunk_track = next;
1092 samples_left_in_chunk = samples;
1093 }

1100 void feed(const uint8_t* data, size_t len) {
1101 size_t pos = 0;
1102 while (pos < len) {
1103 if (current_sample_track == nullptr) {
1104 advanceChunkIfNeeded();
1105 if (current_chunk_track == nullptr) return; // nothing more expected
1106 Track* t = current_chunk_track;
1107 uint32_t size = t->sampleSize(t->next_sample_index);
1108 if (size == 0) {
1109 // no more sizes for this track - stop
1110 samples_left_in_chunk = 0;
1111 continue;
1112 }
1113 current_sample_track = t;
1114 current_sample_size = size;
1115 current_sample_is_first_for_track = (t->next_sample_index == 0);
1116 // Vector::resize() sets the logical size immediately (unlike a
1117 // capacity reserve), so track how much of it is actually filled
1118 // separately rather than relying on sample_buffer.size().
1119 sample_buffer.resize(size);
1120 sample_buffer_filled = 0;
1121 }
1122
1123 size_t need = current_sample_size - sample_buffer_filled;
1124 size_t take = std::min(need, len - pos);
1125 memcpy(sample_buffer.data() + sample_buffer_filled, data + pos, take);
1126 sample_buffer_filled += take;
1127 pos += take;
1128
1129 if (sample_buffer_filled >= current_sample_size) {
1130 dispatchSample(*current_sample_track, sample_buffer.data(),
1131 sample_buffer_filled, current_sample_is_first_for_track);
1132 current_sample_track->next_sample_index++;
1133 samples_left_in_chunk--;
1134 current_sample_track = nullptr;
1135 dispatched_sample_count++;
1136 if (dispatched_sample_count % 5000 == 0) {
1137 LOGI("DemuxerMP4: dispatched %u samples so far",
1138 (unsigned)dispatched_sample_count);
1139 }
1140 }
1141 }

1153 void dispatchVideo(Track& t, const uint8_t* data, size_t size, bool isFirst) {
1154 // nextPtsTicks() must run unconditionally, in sample order, to keep
1155 // the track's timing cursor correct regardless of what we do below -
1156 // call it before any early return.
1157 uint64_t ptsTicks = t.nextPtsTicks();
1158 if ((p_video_out == nullptr && p_video_out_video == nullptr) ||
1159 t.is_hevc_unsupported)
1160 return;
1161
1162 if (t.timescale > 0 && t.stts->size() > 0) {
1163 uint32_t scheduledMs = (uint32_t)((ptsTicks * 1000) / t.timescale);
1164 uint32_t nowMs = millis();
1165 if (!playback_start_set) {
1166 playback_start_ms = nowMs;
1167 playback_start_set = true;
1168 }
1169 uint32_t scheduledWallMs = playback_start_ms + scheduledMs;
1170 if (nowMs > scheduledWallMs) {
1171 LOGW("DemuxerMP4: video frame %u ms late - skipping",
1172 (unsigned)(nowMs - scheduledWallMs));
1173 return;
1174 } else if (nowMs < scheduledWallMs) {
1175 // Spend the slack draining buffered audio steadily instead of
1176 // idling (or, previously, doing nothing at all and letting the
1177 // pipeline race ahead) - this is what actually fixes bursty
1178 // audio delivery; see setVideoAudioSync(). Capped at 200ms so a
1179 // very early frame (e.g. right after a moov-parsing-driven
1180 // pause) can't stall video for unreasonably long.
1181 uint32_t waitMs = scheduledWallMs - nowMs;
1182 LOGI("DemuxerMP4: video frame %u ms early", (unsigned)waitMs);
1183 p_synch->delayVideoFrame((int32_t)(std::min(waitMs, 200u) * 1000), 0);
1184 }
1185 }
1186
1187 // convert AVCC length-prefixed NAL units to Annex-B start codes
1188 nal_tmp.clear();
1189 size_t pos = 0;
1190 size_t lenSize = t.nal_length_size;
1191 while (pos + lenSize <= size) {
1192 uint32_t nalLen = 0;
1193 for (size_t i = 0; i < lenSize; i++)
1194 nalLen = (nalLen << 8) | data[pos + i];
1195 pos += lenSize;
1196 if (pos + nalLen > size) break;
1197 size_t off = nal_tmp.size();
1198 static const uint8_t startCode[4] = {0, 0, 0, 1};
1199 nal_tmp.resize(off + 4 + nalLen);
1200 memcpy(nal_tmp.data() + off, startCode, 4);
1201 memcpy(nal_tmp.data() + off + 4, data + pos, nalLen);
1202 pos += nalLen;
1203 }
1204
1205 bool prependConfig =
1206 isFirst && !t.avc_config_sent && t.sps_pps_annexb.size() > 0;
1207 if (prependConfig) {
1208 writeVideo(t.sps_pps_annexb.data(), t.sps_pps_annexb.size());
1209 t.avc_config_sent = true;
1210 }
1211 if (nal_tmp.size() > 0) writeVideo(nal_tmp.data(), nal_tmp.size());
1212 flushVideo();
1213 }

1228 void dispatchAudio(Track& t, const uint8_t* data, size_t size, bool isFirst) {
1229 if (p_output_audio == nullptr) return;
1230 // routed through p_synch instead of writing p_output_audio directly -
1231 // see setVideoAudioSync(): the default buffers this instead of
1232 // handing it straight to a (likely real-time) sink in bursts.
1233 if (t.audio_codec == Codec::AAC) {
1234 uint8_t adts[7];
1235 writeAdtsHeader(adts, t.aacProfile, t.sampleRateIdx, t.channelCfg,
1236 (int)size);
1237 p_synch->writeAudio(p_output_audio, adts, sizeof(adts));
1238 p_synch->writeAudio(p_output_audio, const_cast<uint8_t*>(data), size);
1239 } else {
1240 // ALAC (magic cookie is exposed via audioALACMagicCookie() for the
1241 // caller to configure their own decoder with) and any other codec:
1242 // raw payload as-is.
1243 p_synch->writeAudio(p_output_audio, const_cast<uint8_t*>(data), size);
1244 }

1247 static void writeAdtsHeader(uint8_t* adts, int aacProfile, int sampleRateIdx,
1248 int channelCfg, int frameLen) {
1249 adts[0] = 0xFF;
1250 adts[1] = 0xF1;
1251 adts[2] = ((aacProfile - 1) << 6) | (sampleRateIdx << 2) |
1252 ((channelCfg >> 2) & 0x1);
1253 adts[3] = ((channelCfg & 0x3) << 6) | ((frameLen + 7) >> 11);
1254 adts[4] = ((frameLen + 7) >> 3) & 0xFF;
1255 adts[5] = (((frameLen + 7) & 0x7) << 5) | 0x1F;
1256 adts[6] = 0xFC;
1257 }

1771 template <typename F>
1772 static void forEachAnnexBNal(const uint8_t* data, size_t len, F callback) {
1773 size_t i = 0;
1774 while (i + 3 <= len) {
1775 size_t scLen = 0;
1776 if (i + 4 <= len && data[i] == 0 && data[i + 1] == 0 &&
1777 data[i + 2] == 0 && data[i + 3] == 1) {
1778 scLen = 4;
1779 } else if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
1780 scLen = 3;
1781 }
1782 if (scLen == 0) {
1783 i++;
1784 continue;
1785 }
1786 size_t nalStart = i + scLen;
1787 size_t j = nalStart;
1788 size_t nextStart = len;
1789 while (j + 3 <= len) {
1790 if (data[j] == 0 && data[j + 1] == 0 &&
1791 (data[j + 2] == 1 ||
1792 (j + 3 < len && data[j + 2] == 0 && data[j + 3] == 1))) {
1793 nextStart = j;
1794 break;
1795 }
1796 j++;
1797 }
1798 if (nextStart > nalStart) callback(data + nalStart, nextStart - nalStart);
1799 i = nextStart;
1800 }
1801 }

1855 void writeEsds(MP4BoxWriter& b) {
1856 int sampleRateIdx = aacSampleRateIndex(audio_info.sample_rate);
1857 uint8_t asc[2];
1858 asc[0] = (uint8_t)((audio_profile << 3) | (sampleRateIdx >> 1));
1859 asc[1] = (uint8_t)(((sampleRateIdx & 1) << 7) | (audio_info.channels << 3));
1860
1861 size_t pos = b.beginBox("esds");
1862 b.u32(0); // version + flags
1863 // ES_Descriptor
1864 b.u8(0x03);
1865 // size: ES_ID(2)+flags(1)+DecoderConfigDescr(19 incl. its own tag+size)
1866 // +SLConfigDescr(3 incl. its own tag+size) = 25
1867 b.u8(25);
1868 b.u16(kAudioTrackId); // ES_ID
1869 b.u8(0); // flags
1870 // DecoderConfigDescr
1871 b.u8(0x04);
1872 b.u8(
1873 17); // size:
1874 // objType(1)+streamType(1)+bufSize(3)+maxBr(4)+avgBr(4)+DecSpecificInfo(4)
1875 b.u8(0x40); // objectTypeIndication: AAC
1876 b.u8((5 << 2) | 1); // streamType: audio(5), upstream=0, reserved=1
1877 b.u24(0); // bufferSizeDB
1878 b.u32(128000); // maxBitrate
1879 b.u32(128000); // avgBitrate
1880 // DecSpecificInfo (the 2-byte AudioSpecificConfig)
1881 b.u8(0x05);
1882 b.u8(2);
1883 b.bytes(asc, 2);
1884 // SLConfigDescr
1885 b.u8(0x06);
1886 b.u8(1);
1887 b.u8(0x02);
1888 b.endBox(pos);
1889 }
