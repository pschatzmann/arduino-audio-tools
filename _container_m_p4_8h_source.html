235 size_t write(const uint8_t* data, size_t len) override {
236 if (!is_active) return 0;
237 // After quickStart(), 'moov' still physically trails 'mdat' on disk
238 // (only relocated logically) - cap feeding at 'mdat's declared size
239 // and discard the trailing, already-parsed 'moov' bytes instead of
240 // re-parsing them.
241 size_t feed_len = len;
242 bool truncated = false;
243 if (quickstart_bytes_remaining >= 0 &&
244 (int64_t)feed_len > quickstart_bytes_remaining) {
245 feed_len = (size_t)quickstart_bytes_remaining;
246 truncated = true;
247 }
248 size_t written = feed_len > 0 ? parser.write(data, feed_len) : 0;
249 total_bytes_received += written;
250 if (quickstart_bytes_remaining >= 0) quickstart_bytes_remaining -= written;
251 // Only swallow the truncated remainder if 'feed_len' was fully
252 // accepted - otherwise this is an ordinary partial write.
253 return (truncated && written == feed_len) ? len : written;
254 }

271 bool quickStart() {
272 if (p_seek_source == nullptr) return false;
273
274 uint64_t pos = 0, moov_offset = 0, moov_size = 0, mdat_offset = 0,
275 mdat_size = 0;
276 bool moov_found = false, mdat_found = false;
277 const int max_top_level_boxes = 64; // safety bound against a corrupt file
278
279 for (int i = 0; i < max_top_level_boxes; i++) {
280 uint8_t hdr[8];
281 if (!p_seek_source->seek((size_t)pos)) break;
282 if (p_seek_source->readBytes(hdr, 8) < 8) break;
283 uint64_t box_size = readU32(hdr);
284 // 64-bit ('size'==1) and "extends to EOF" ('size'==0) boxes aren't
285 // understood here or by MP4Parser's own walker - bail out and let
286 // the caller fall back to normal sequential parsing.
287 if (box_size < 8) break;
288
289 if (memcmp(hdr + 4, "moov", 4) == 0) {
290 moov_offset = pos;
291 moov_size = box_size;
292 moov_found = true;
293 } else if (memcmp(hdr + 4, "mdat", 4) == 0) {
294 mdat_offset = pos;
295 mdat_size = box_size;
296 mdat_found = true;
297 }
298 if (moov_found && mdat_found) break;
299 pos += box_size;
300 }
301
302 if (!moov_found || !mdat_found || moov_offset < mdat_offset) {
303 return false; // nothing found, or already faststart
304 }
305
306 LOGI("DemuxerMP4::quickStart: moov at %u (size %u), mdat at %u - "
307 "parsing moov directly",
308 (unsigned)moov_offset, (unsigned)moov_size, (unsigned)mdat_offset);
309
310 // Feed exactly 'moov's bytes through the same callback-driven parser
311 // used for normal streaming - just sourced via seek()+readBytes().
312 parser.begin(moov_offset);
313 uint8_t buf[512];
314 uint64_t remaining = moov_size;
315 uint64_t off = moov_offset;
316 while (remaining > 0) {
317 size_t chunk = (size_t)std::min<uint64_t>(remaining, sizeof(buf));
318 if (!p_seek_source->seek((size_t)off)) break;
319 size_t got = p_seek_source->readBytes(buf, chunk);
320 if (got == 0) break;
321 parser.write(buf, got);
322 off += got;
323 remaining -= got;
324 }
325
326 // Hand off: reset the parser's box state machine to expect a fresh
327 // box at 'mdat's offset, and reposition the seek source there so the
328 // caller's normal write()-loop resumes from 'mdat', not offset 0.
329 parser.begin(mdat_offset);
330 p_seek_source->seek((size_t)mdat_offset);
331 // caps write() at 'mdat's size so the trailing 'moov' isn't re-parsed
332 quickstart_bytes_remaining = (int64_t)mdat_size;
333 return true;
334 }

590 static bool isSafeToSkipAvcc(const uint8_t* data, size_t size, size_t lenSize) {
591 size_t pos = 0;
592 while (pos + lenSize <= size) {
593 uint32_t nalLen = 0;
594 for (size_t i = 0; i < lenSize; i++) nalLen = (nalLen << 8) | data[pos + i];
595 pos += lenSize;

598 if (nal_ref_idc != 0) return false;

739 void onTrak() {
740 Track* t = new Track();
741 if (p_seek_source != nullptr) {
742 // RAM-backed defaults from Track's own member initializers aren't
743 // needed when a seek source is configured - swap them for
744 // SourceSeekSampleTableStore instead of paying for both.
745 delete t->sample_sizes;
746 t->sample_sizes = new SourceSeekSampleTableStore<uint32_t>(
747 *p_seek_source, sizeof(uint32_t), readU32);
748 delete t->chunk_offsets;
749 // on-disk width (4 bytes for 'stco', 8 for 'co64') isn't known yet
750 // here - defaults to 'stco' width and is corrected in onStco() via
751 // setOnDiskEntrySize() once the actual box type is seen, always
752 // before any get() call (which only happens much later, during
753 // 'mdat' playback).
754 t->chunk_offsets = new SourceSeekSampleTableStore<uint64_t>(
755 *p_seek_source, sizeof(uint32_t), readU32Widened);
756 delete t->stsc;
757 // stsc's on-disk entries are 12 bytes (first_chunk, samples_per_chunk,
758 // sample_description_index - all u32), matching onStsc()'s own i*12
759 // parsing stride below, even though readStscEntry() only decodes the
760 // first two fields into StscEntry. Passing 8 here previously
761 // (2 fields' worth) misaligned every get() past index 0, since
762 // get()'s offset math is entries*onDiskSize - reproducibly corrupted
763 // chunk-to-sample mapping for any track whose stsc has more than one
764 // entry (typically audio, not video, since a video track commonly
765 // only ever needs one stsc entry for the whole file).
766 t->stsc = new SourceSeekSampleTableStore<StscEntry>(
767 *p_seek_source, sizeof(uint32_t) * 3, readStscEntry);
768 delete t->stts;
769 t->stts = new SourceSeekSampleTableStore<SttsEntry>(
770 *p_seek_source, sizeof(uint32_t) * 2, readSttsEntry);
771 }
772 tracks.push_back(t);
773 current_track = t;
774 }

1051 void onStsz(MP4Parser::Box& box) {
1052 if (current_track == nullptr) return;
1053 if (beginBoxAccum(box))
1054 stsz_header_pending = true; // starting a fresh stsz box
1055 box_accum.writeArray(box.data, box.available);
1056
1057 // The 12-byte header (version+flags, sampleSize, sampleCount) must be
1058 // fully consumed *once* before any bytes are interpreted as per-sample
1059 // sizes. Using "no sizes pushed yet" as a stand-in for "header not
1060 // parsed yet" (as tried previously) is wrong: in the variable-size
1061 // case, both are simultaneously true right after the header IS parsed,
1062 // so under fine-grained incremental delivery the per-sample loop below
1063 // could start consuming still-unparsed header bytes as if they were
1064 // sample sizes. An explicit flag avoids that ambiguity.
1065 if (stsz_header_pending) {
1066 if (box_accum.available() < 12)
1067 return; // wait for the rest of the header
1068 Track* t = current_track;
1069 uint32_t sampleSize = readU32(box_accum.data() + 4);
1070 uint32_t sampleCount = readU32(box_accum.data() + 8);
1071 LOGD("DemuxerMP4: stsz sampleSize=%u sampleCount=%u", (unsigned)sampleSize,
1072 (unsigned)sampleCount);
1073 box_accum.clearArray(12);
1074 stsz_header_pending = false;
1075 if (sampleSize != 0) {
1076 t->fixed_sample_size = sampleSize;
1077 t->fixed_sample_count = sampleCount;
1078 }
1079 // else: sample_sizes is filled below via append(), one entry at a
1080 // time as bytes arrive - do NOT pre-resize it (a RAM-backed store's
1081 // Vector::push_back always appends, unlike BaseBuffer::write()
1082 // which fills pre-sized capacity in place)
1083 }
1084 // incrementally consume per-sample sizes (fixed-size case has none left
1085 // to read; the table is only present in the box when sampleSize==0)
1086 if (current_track->fixed_sample_size == 0) {
1087 while (box_accum.available() >= 4) {
1088 // +12: this table's own version/flags+sampleSize+sampleCount
1089 // sub-header (already drained from box_accum via clearArray(12)
1090 // above, but boxPayloadFileOffset() itself doesn't know that)
1091 current_track->sample_sizes->setNextEntryOffset(boxPayloadFileOffset() + 12);
1092 current_track->sample_sizes->append(readU32(box_accum.data()));
1093 box_accum.clearArray(4);
1094 size_t n = current_track->sample_sizes->size();
1095 if (n % 20000 == 0)
1096 LOGD("DemuxerMP4: stsz sample_sizes progress: %u", (unsigned)n);
1097 }
1098 }
1099 if (box.is_complete) {
1100 LOGD("DemuxerMP4: stsz complete, %u sample_sizes",
1101 (unsigned)current_track->sample_sizes->size());
1102 endBoxAccum();
1103 }
1104 }

1179 uint32_t samplesInChunk(Track& t, uint32_t chunkIndex0) {
1180 if (t.stsc->size() == 0) return 0;
1181 uint32_t chunk1 = chunkIndex0 + 1; // stsc uses 1-based chunk numbers
1182 while (t.stsc_cursor + 1 < t.stsc->size() &&
1183 t.stsc->get(t.stsc_cursor + 1).first_chunk <= chunk1) {
1184 t.stsc_cursor++;
1185 }
1186 return t.stsc->get(t.stsc_cursor).samples_per_chunk;
1187 }

1197 void advanceChunkIfNeeded() {
1198 while (samples_left_in_chunk == 0) {
1199 Track* next = nullptr;
1200 uint64_t best_offset = 0;
1201 if (p_video_track != nullptr &&
1202 p_video_track->next_chunk_index < p_video_track->chunk_offsets->size()) {
1203 best_offset = p_video_track->chunk_offsets->get(p_video_track->next_chunk_index);
1204 next = p_video_track;
1205 }
1206 if (p_audio_track != nullptr &&
1207 p_audio_track->next_chunk_index < p_audio_track->chunk_offsets->size()) {
1208 uint64_t off = p_audio_track->chunk_offsets->get(p_audio_track->next_chunk_index);
1209 if (next == nullptr || off < best_offset) {
1210 best_offset = off;
1211 next = p_audio_track;
1212 }
1213 }
1214 if (next == nullptr) {
1215 current_chunk_track = nullptr; // nothing more expected
1216 return;
1217 }
1218 uint32_t samples = samplesInChunk(*next, next->next_chunk_index);
1219 next->next_chunk_index++;
1220 if (samples == 0) continue; // empty chunk entry - keep looking
1221 current_chunk_track = next;
1222 samples_left_in_chunk = samples;
1223 }
1224 }

1230 void feed(const uint8_t* data, size_t len) {
1231 size_t pos = 0;
1232 while (pos < len) {
1233 if (current_sample_track == nullptr) {
1234 advanceChunkIfNeeded();
1235 if (current_chunk_track == nullptr) return; // nothing more expected
1236 Track* t = current_chunk_track;
1237 uint32_t size = t->sampleSize(t->next_sample_index);
1238 if (size == 0) {
1239 // no more sizes for this track - stop
1240 samples_left_in_chunk = 0;
1241 continue;
1242 }
1243 current_sample_track = t;
1244 current_sample_size = size;
1245 current_sample_is_first_for_track = (t->next_sample_index == 0);
1246 // Vector::resize() sets the logical size immediately (unlike a
1247 // capacity reserve), so track how much of it is actually filled
1248 // separately rather than relying on sample_buffer.size().
1249 sample_buffer.resize(size);
1250 sample_buffer_filled = 0;
1251 }
1252
1253 size_t need = current_sample_size - sample_buffer_filled;
1254 size_t take = std::min(need, len - pos);
1255 memcpy(sample_buffer.data() + sample_buffer_filled, data + pos, take);
1256 sample_buffer_filled += take;
1257 pos += take;
1258
1259 if (sample_buffer_filled >= current_sample_size) {
1260 dispatchSample(*current_sample_track, sample_buffer.data(),
1261 sample_buffer_filled, current_sample_is_first_for_track);
1262 current_sample_track->next_sample_index++;
1263 samples_left_in_chunk--;
1264 current_sample_track = nullptr;
1265 dispatched_sample_count++;
1266 if (dispatched_sample_count % 5000 == 0) {
1267 LOGI("DemuxerMP4: dispatched %u samples so far",
1268 (unsigned)dispatched_sample_count);
1269 }
1270 }
1271 }

1283 void dispatchVideo(Track& t, const uint8_t* data, size_t size, bool isFirst) {
1284 // nextPtsTicks() must run unconditionally, in sample order, to keep
1285 // the track's timing cursor correct regardless of what we do below -
1286 // call it before any early return.
1287 uint64_t ptsTicks = t.nextPtsTicks();
1288 if ((p_video_out == nullptr && p_video_out_video == nullptr) ||
1289 t.is_hevc_unsupported)
1290 return;
1291
1292 if (t.timescale > 0 && t.stts->size() > 0) {
1293 uint32_t scheduledMs = (uint32_t)((ptsTicks * 1000) / t.timescale);
1294 uint32_t nowMs = millis();
1295 if (!playback_start_set) {
1296 playback_start_ms = nowMs;
1297 playback_start_set = true;
1298 }
1299 uint32_t scheduledWallMs = playback_start_ms + scheduledMs;
1300 if (nowMs > scheduledWallMs) {
1301 // Only skip if nothing else references this sample (see
1302 // isSafeToSkipAvcc()) - dropping a referenced I/P-frame here
1303 // would corrupt every later frame's decode until the next
1304 // keyframe, not just this one. An unsafe-to-skip late frame is
1305 // decoded immediately instead (no wait either, since we're
1306 // already behind).
1307 if (isSafeToSkipAvcc(data, size, t.nal_length_size)) {
1308 LOGW("DemuxerMP4: video frame %u ms late - skipping (non-reference)",
1309 (unsigned)(nowMs - scheduledWallMs));
1310 return;
1311 }
1312 LOGW("DemuxerMP4: video frame %u ms late - decoding anyway (referenced by later frames)",
1313 (unsigned)(nowMs - scheduledWallMs));
1314 } else if (nowMs < scheduledWallMs) {
1315 // Spend the slack draining buffered audio steadily instead of
1316 // idling (or, previously, doing nothing at all and letting the
1317 // pipeline race ahead) - this is what actually fixes bursty
1318 // audio delivery; see setVideoAudioSync(). Capped at 200ms so a
1319 // very early frame (e.g. right after a moov-parsing-driven
1320 // pause) can't stall video for unreasonably long.
1321 uint32_t waitMs = scheduledWallMs - nowMs;
1322 LOGI("DemuxerMP4: video frame %u ms early", (unsigned)waitMs);
1323 p_synch->delayVideoFrame((int32_t)(std::min(waitMs, 200u) * 1000), 0);
1324 }
1325 }
1326
1327 // convert AVCC length-prefixed NAL units to Annex-B start codes
1328 nal_tmp.clear();
1329 size_t pos = 0;
1330 size_t lenSize = t.nal_length_size;
1331 while (pos + lenSize <= size) {
1332 uint32_t nalLen = 0;
1333 for (size_t i = 0; i < lenSize; i++)
1334 nalLen = (nalLen << 8) | data[pos + i];
1335 pos += lenSize;
1336 if (pos + nalLen > size) break;
1337 size_t off = nal_tmp.size();
1338 static const uint8_t startCode[4] = {0, 0, 0, 1};
1339 nal_tmp.resize(off + 4 + nalLen);
1340 memcpy(nal_tmp.data() + off, startCode, 4);
1341 memcpy(nal_tmp.data() + off + 4, data + pos, nalLen);
1342 pos += nalLen;
1343 }
1344
1345 bool prependConfig =
1346 isFirst && !t.avc_config_sent && t.sps_pps_annexb.size() > 0;
1347 if (prependConfig) {
1348 writeVideo(t.sps_pps_annexb.data(), t.sps_pps_annexb.size());
1349 t.avc_config_sent = true;
1350 }
1351 if (nal_tmp.size() > 0) writeVideo(nal_tmp.data(), nal_tmp.size());
1352 flushVideo();
1353 }

1387 static void writeAdtsHeader(uint8_t* adts, int aacProfile, int sampleRateIdx,
1388 int channelCfg, int frameLen) {
1389 adts[0] = 0xFF;
1390 adts[1] = 0xF1;
1391 adts[2] = ((aacProfile - 1) << 6) | (sampleRateIdx << 2) |
1392 ((channelCfg >> 2) & 0x1);

1911 template <typename F>
1912 static void forEachAnnexBNal(const uint8_t* data, size_t len, F callback) {
1913 size_t i = 0;
1914 while (i + 3 <= len) {
1915 size_t scLen = 0;
1916 if (i + 4 <= len && data[i] == 0 && data[i + 1] == 0 &&
1917 data[i + 2] == 0 && data[i + 3] == 1) {
1918 scLen = 4;
1919 } else if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
1920 scLen = 3;
1921 }
1922 if (scLen == 0) {
1923 i++;
1924 continue;
1925 }
1926 size_t nalStart = i + scLen;
1927 size_t j = nalStart;
1928 size_t nextStart = len;
1929 while (j + 3 <= len) {
1930 if (data[j] == 0 && data[j + 1] == 0 &&
1931 (data[j + 2] == 1 ||
1932 (j + 3 < len && data[j + 2] == 0 && data[j + 3] == 1))) {
1933 nextStart = j;
1934 break;
1935 }
1936 j++;
1937 }
1938 if (nextStart > nalStart) callback(data + nalStart, nextStart - nalStart);
1939 i = nextStart;
1940 }
1941 }
