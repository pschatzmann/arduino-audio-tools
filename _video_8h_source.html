61inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,
62 uint16_t height) {
63 size_t px = (size_t)width * (size_t)height;
64 switch (format) {

66 case VideoFormat::RGB666:
67 case VideoFormat::RGB888:
68 return px * 3;

70 case VideoFormat::RGB565:

72 case VideoFormat::I420:

86inline bool isH264KeyFrame(const uint8_t *data, size_t len) {
87 for (size_t i = 0; i + 3 < len; i++) {
88 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
89 if ((data[i + 3] & 0x1F) == 5) return true; // IDR slice
90 i += 2;
91 }
92 }
93 return false;
94}

259 void delayVideoFrame(int32_t microsecondsPerFrame, uint32_t time_used_ms) {
260 // p_out is only ever set inside writeAudio() - if this is called
261 // before any audio has been written yet (e.g. a file's first few
262 // dispatched chunks happen to be video-only), there's nothing to
263 // drain and no sink to drain it to; skip the wait entirely rather
264 // than dereferencing a null p_out in the loop below.
265 if (p_out == nullptr) return;
266 uint32_t delay_ms = microsecondsPerFrame / 1000;
267 uint64_t timeout = millis() + delay_ms + correction_ms;
268
269 // 'p_out' here is the same Print DemuxerMP4::setOutputAudio() was
270 // given - for an encoded codec (AAC, MP3, ...) that's a decoder's
271 // input, not a raw PCM sink, so what's buffered in ring_buffer is
272 // still-encoded bitstream bytes. Draining it 8 bytes at a time
273 // (the previous approach) fed those bytes into the decoder in
274 // fragments far smaller than one ADTS/frame boundary, repeated
275 // thousands of times a second - reproducibly desynced libhelix's AAC
276 // decoder on a real HE-AAC/SBR file, sending it down a path that
277 // computed garbage (negative) band-count parameters and hung
278 // indefinitely inside its SBR patch-building code. Draining the
279 // current backlog in one shot (in generously-sized gulps, not
280 // byte-by-byte) keeps writes closer to how the bytes were originally
281 // framed by dispatchAudio() and avoided the hang in testing; only the
282 // wait for the remaining window is spun out afterwards, with no
283 // further writes into the decoder.
284 uint8_t audio[512];
285 int n;
286 while ((n = ring_buffer.readArray(audio, sizeof(audio))) > 0) {
287 p_out->write(audio, n);
288 }
289 while (millis() < timeout) {
290 delay(1);
291 }
292 }
293
294 protected:
295 RingBuffer<uint8_t> ring_buffer{0};
296 Print *p_out = nullptr;
297 int correction_ms = 0;
298};

321class VideoAudioClockSync : public VideoAudioSync {
322 public:
326 void delayVideoFrame(int32_t microsecondsPerFrame,
327 uint32_t time_used_ms) override {
328 uint32_t nowMs = millis();
329 if (!playback_start_set) {
330 playback_start_ms = nowMs;
331 playback_start_set = true;
332 frame_index = 0;
333 }
334 uint32_t scheduledWallMs =
335 playback_start_ms +
336 (uint32_t)((uint64_t)frame_index * microsecondsPerFrame / 1000);
337 frame_index++;
338 if (nowMs < scheduledWallMs) {
339 delay(scheduledWallMs - nowMs);
340 }
341 // else: already behind schedule - don't add to the lag
342 }
343
344 protected:
345 bool playback_start_set = false;
346 uint32_t playback_start_ms = 0;
347 uint32_t frame_index = 0;
