61inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,
62 uint16_t height) {
63 size_t px = (size_t)width * (size_t)height;
64 switch (format) {
65 case VideoFormat::RAW:
66 case VideoFormat::RGB666:
67 case VideoFormat::RGB888:
68 return px * 3;
69 case VideoFormat::YUV422:
70 case VideoFormat::RGB565:
71 return px * 2;
72 case VideoFormat::I420:
73 return px + px / 2;
74 default:
75 return 0;
76 }

86inline bool isH264KeyFrame(const uint8_t* data, size_t len) {
87 for (size_t i = 0; i + 3 < len; i++) {
88 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
89 if ((data[i + 3] & 0x1F) == 5) return true; // IDR slice
90 i += 2;
91 }
92 }
93 return false;
94}

253 size_t write(const uint8_t* data, size_t len) override {
254 uint32_t now = millis();
255 if (has_last_call) {
256 total_turnaround_ms += (now - last_call_ms);
257 }
258 last_call_ms = now;
259 has_last_call = true;
260
261 // VideoOutput and Print are unrelated types (no common base), so the
262 // constructor that received a VideoOutput& only has somewhere to
263 // store it as p_target_video, not p_target - check that one first.
264 uint32_t forwardStart = millis();
265 size_t result = 0;
266 if (p_target_video != nullptr) {
267 result = p_target_video->write(data, len);
268 } else if (p_target != nullptr) {
269 result = p_target->write(data, len);
270 }
271 total_forward_ms += millis() - forwardStart;
272
273 count_++;
274 return result;
275 }
276
277 void flush() override {
278 if (p_target_video != nullptr) p_target_video->flush();
279 }
280 void setSkipRender(bool skip) override {
281 if (p_target_video != nullptr) p_target_video->setSkipRender(skip);
282 }
