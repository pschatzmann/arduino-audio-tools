60inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,
61 uint16_t height) {
62 size_t px = (size_t)width * (size_t)height;
63 switch (format) {
64 case VideoFormat::RAW:
65 case VideoFormat::RGB666:
66 case VideoFormat::RGB888:
67 return px * 3;
68 case VideoFormat::YUV422:
69 case VideoFormat::RGB565:
70 return px * 2;
71 case VideoFormat::I420:
72 return px + px / 2;
73 default:
74 return 0;
75 }
76}

85inline bool isH264KeyFrame(const uint8_t *data, size_t len) {
86 for (size_t i = 0; i + 3 < len; i++) {
87 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
88 if ((data[i + 3] & 0x1F) == 5) return true; // IDR slice
89 i += 2;
90 }
91 }
92 return false;
93}

235 void writeAudio(Print *out, uint8_t *data, size_t size) {
236 p_out = out;
237 if (ring_buffer.availableForWrite() < size) {
238 int bytes_to_play = size - ring_buffer.availableForWrite();
239 uint8_t audio[bytes_to_play];
240 ring_buffer.readArray(audio, bytes_to_play);
241 p_out->write(audio, bytes_to_play);
242 }
243 size_t written = ring_buffer.writeArray(data, size);
244 assert(written = size);
245 }

252 uint8_t audio[8];
253 // output audio
