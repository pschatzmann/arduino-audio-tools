61inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,

86inline bool isH264KeyFrame(const uint8_t *data, size_t len) {
87 for (size_t i = 0; i + 3 < len; i++) {
88 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
89 if ((data[i + 3] & 0x1F) == 5) return true; // IDR slice
90 i += 2;
91 }
92 }
93 return false;
94}

312 void writeAudio(Print *out, uint8_t *data, size_t size) {
313 p_out = out;
314 if (ring_buffer.availableForWrite() < size) {
315 int bytes_to_play = size - ring_buffer.availableForWrite();
316 uint8_t audio[bytes_to_play];
317 ring_buffer.readArray(audio, bytes_to_play);
318 p_out->write(audio, bytes_to_play);
319 }
320 size_t written = ring_buffer.writeArray(data, size);
321 assert(written == size);
322 }

326 void delayVideoFrame(int32_t microsecondsPerFrame, uint32_t time_used_ms) {
327 // p_out is only ever set inside writeAudio() - if this is called
328 // before any audio has been written yet (e.g. a file's first few
329 // dispatched chunks happen to be video-only), there's nothing to
330 // drain and no sink to drain it to; skip the wait entirely rather
331 // than dereferencing a null p_out in the loop below.
332 if (p_out == nullptr) return;
333 uint32_t delay_ms = microsecondsPerFrame / 1000;
334 uint64_t timeout = millis() + delay_ms + correction_ms;
335
336 // 'p_out' here is the same Print DemuxerMP4::setOutputAudio() was
337 // given - for an encoded codec (AAC, MP3, ...) that's a decoder's
338 // input, not a raw PCM sink, so what's buffered in ring_buffer is
339 // still-encoded bitstream bytes. Draining it 8 bytes at a time
340 // (the previous approach) fed those bytes into the decoder in
341 // fragments far smaller than one ADTS/frame boundary, repeated
342 // thousands of times a second - reproducibly desynced libhelix's AAC
343 // decoder on a real HE-AAC/SBR file, sending it down a path that
344 // computed garbage (negative) band-count parameters and hung
345 // indefinitely inside its SBR patch-building code. Draining the
346 // current backlog in one shot (in generously-sized gulps, not
347 // byte-by-byte) keeps writes closer to how the bytes were originally
348 // framed by dispatchAudio() and avoided the hang in testing; only the
349 // wait for the remaining window is spun out afterwards, with no
350 // further writes into the decoder.
351 uint8_t audio[512];
352 int n;
353 while ((n = ring_buffer.readArray(audio, sizeof(audio))) > 0) {
354 p_out->write(audio, n);
355 }
356 while (millis() < timeout) {
357 delay(1);
358 }
359 }
360
361 protected:
362 RingBuffer<uint8_t> ring_buffer{0};
363 Print *p_out = nullptr;
364 int correction_ms = 0;
365};

400class VideoAudioClockSync : public VideoAudioSync {
401 public:
405 void delayVideoFrame(int32_t microsecondsPerFrame,
406 uint32_t time_used_ms) override {
407 uint32_t nowMs = millis();
408 if (!playback_start_set) {
409 playback_start_ms = nowMs;
410 playback_start_set = true;
411 frame_index = 0;
412 }
413 uint32_t scheduledWallMs =
414 playback_start_ms +
415 (uint32_t)((uint64_t)frame_index * microsecondsPerFrame / 1000);
416 frame_index++;
417 // Positive: this frame is ahead of schedule by this many ms (about to
418 // wait that long); negative: already late by this many ms (playback
419 // is lagging - decode/render is taking longer than the frame period).
420 int32_t aheadMs = (int32_t)scheduledWallMs - (int32_t)nowMs;
421 LOGI("video frame %u: %s%d ms (decode+render took %u ms)", frame_index,
422 aheadMs >= 0 ? "early by " : "late by ",
423 aheadMs >= 0 ? aheadMs : -aheadMs, (unsigned)time_used_ms);
424 last_ahead_ms = aheadMs;
425 total_frames++;
426 total_decode_write_ms += time_used_ms;
427 if (aheadMs < p_frame_skip_lateness_ms) skip_p_frames_in_gop = true;
428 if (nowMs < scheduledWallMs) {
429 delay(scheduledWallMs - nowMs);
430 }
431 // else: already behind schedule - don't add to the lag
432 }
