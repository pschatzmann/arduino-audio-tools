39inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,
40 uint16_t height) {
41 size_t px = (size_t)width * (size_t)height;
42 switch (format) {
43 case VideoFormat::RAW:
44 return px * 3;
45 case VideoFormat::YUV422:
46 case VideoFormat::RGB565:
47 return px * 2;
48 case VideoFormat::I420:
49 return px + px / 2;
50 default:
51 return 0;
52 }

220 void writeAudio(Print *out, uint8_t *data, size_t size) {
221 p_out = out;
222 if (ring_buffer.availableForWrite() < size) {
223 int bytes_to_play = size - ring_buffer.availableForWrite();
224 uint8_t audio[bytes_to_play];
225 ring_buffer.readArray(audio, bytes_to_play);
226 p_out->write(audio, bytes_to_play);
227 }
228 size_t written = ring_buffer.writeArray(data, size);
229 assert(written = size);
230 }

234 void delayVideoFrame(int32_t microsecondsPerFrame, uint32_t time_used_ms) {
235 uint32_t delay_ms = microsecondsPerFrame / 1000;
236 uint64_t timeout = millis() + delay_ms + correction_ms;
237 uint8_t audio[8];
238 // output audio
239 while (millis() < timeout) {
