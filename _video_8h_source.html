57inline size_t videoFrameSizeBytes(VideoFormat format, uint16_t width,
58 uint16_t height) {
59 size_t px = (size_t)width * (size_t)height;
60 switch (format) {
61 case VideoFormat::RAW:
62 case VideoFormat::RGB666:
63 case VideoFormat::RGB888:
64 return px * 3;
65 case VideoFormat::YUV422:
66 case VideoFormat::RGB565:
67 return px * 2;
68 case VideoFormat::I420:
69 return px + px / 2;
70 default:
71 return 0;
72 }
73}

82inline bool isH264KeyFrame(const uint8_t *data, size_t len) {
83 for (size_t i = 0; i + 3 < len; i++) {
84 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
85 if ((data[i + 3] & 0x1F) == 5) return true; // IDR slice
86 i += 2;
87 }
88 }
89 return false;
90}

372 void writeAudio(Print *out, uint8_t *data, size_t size) {
373 p_out = out;
374 if (ring_buffer.availableForWrite() < size) {
375 int bytes_to_play = size - ring_buffer.availableForWrite();
376 uint8_t audio[bytes_to_play];
377 ring_buffer.readArray(audio, bytes_to_play);
378 p_out->write(audio, bytes_to_play);
379 }
380 size_t written = ring_buffer.writeArray(data, size);
381 assert(written = size);
382 }

386 void delayVideoFrame(int32_t microsecondsPerFrame, uint32_t time_used_ms) {
387 uint32_t delay_ms = microsecondsPerFrame / 1000;
388 uint64_t timeout = millis() + delay_ms + correction_ms;
389 uint8_t audio[8];
390 // output audio
391 while (millis() < timeout) {
392 ring_buffer.readArray(audio, 8);
393 p_out->write(audio, 8);
394 }
395 }
