189 static void baseMimeType(const char* mime, char* out, size_t outSize) {
190 size_t len = 0;
191 while (mime[len] != '\0' && mime[len] != ';' && len < outSize - 1) len++;
192 while (len > 0 && (mime[len - 1] == ' ' || mime[len - 1] == '\t')) len--;
193 memcpy(out, mime, len);
194 out[len] = '\0';
195 }

197 bool selectDecoder(const char* mime) {
198 bool result = false;
199 if (mime == nullptr) return false;
200 // do nothing if no change
201 if (StrView(mime).equalsIgnoreCase(actual_decoder.mime)) {
202 is_first = false;
203 return true;
204 }
205 // close actual decoder
206 if (actual_decoder.decoder != this) end();
207
208 // A source may report a MIME type with an RFC 6381-style "codecs"
209 // (or other) parameter, e.g. "audio/mpeg; codecs=\"mpeg1-layer2\"" -
210 // a decoder registered for that exact parameterized type (a more
211 // specific match) must win over one only registered for the plain
212 // base type ("audio/mpeg"). Two passes, not one combined check: a
213 // single pass stops at the first entry that matches *either* way, so
214 // a generic base-type decoder registered before the specific one
215 // would otherwise shadow it purely by list order.
216 char base[64];
217 baseMimeType(mime, base, sizeof(base));
218 int exact_j = -1, base_j = -1;
219 for (int j = 0; j < decoders.size(); j++) {
220 if (StrView(decoders[j].mime).equalsIgnoreCase(mime)) { exact_j = j; break; }
221 if (base_j < 0 && StrView(decoders[j].mime).equalsIgnoreCase(base)) base_j = j;
222 }
223 int match_j = exact_j >= 0 ? exact_j : base_j;
224
225 // find the corresponding decoder
226 selected_mime = nullptr;
227 if (match_j >= 0) {
228 DecoderInfo info = decoders[match_j];
229 LOGI("Using decoder for %s (%s)", info.mime, mime);
230 actual_decoder = info;
231 // define output if it has not been defined
232 if (p_print != nullptr && actual_decoder.decoder != this
233 && actual_decoder.decoder->getOutput() == nullptr) {
234 actual_decoder.decoder->setOutput(*p_print);
235 }
236 if (!*actual_decoder.decoder) {
237 actual_decoder.decoder->begin();
238 LOGI("Decoder %s started", actual_decoder.mime);
239 }
240 result = true;
241 selected_mime = mime;
242 }
243 is_first = false;
244 return result;
245 }

269 size_t write(const uint8_t* data, size_t len) override {
270 if (is_first) {
271 const char* mime = nullptr;
272 if (p_mime_source != nullptr) {
273 // get content type from http header
274 mime = p_mime_source->mime();
275 if (mime) LOGI("mime from http request: %s", mime);
276 }
277 if (mime == nullptr) {
278 // use the mime detector
279 mime_detector.write((uint8_t*)data, len);
280 mime = mime_detector.mime();
281 if (mime) LOGI("mime from mime_detector: %s", mime);
282 }
283 if (mime != nullptr) {
284 // select the decoder based on the detemined mime type
285 if (!selectDecoder(mime)) {
286 LOGE("The decoder could not be found for %s", mime);
287 actual_decoder.decoder = &nop;
288 actual_decoder.is_open = true;
289 }
290 }
291 is_first = false;
292 }
293 // check if we have a decoder
294 if (actual_decoder.decoder == nullptr) return 0;
295 // decode the data
296 return actual_decoder.decoder->write(data, len);
297 }
