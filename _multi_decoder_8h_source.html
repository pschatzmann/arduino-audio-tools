180 static void baseMimeType(const char* mime, char* out, size_t outSize) {
181 size_t len = 0;
182 while (mime[len] != '\0' && mime[len] != ';' && len < outSize - 1) len++;
183 while (len > 0 && (mime[len - 1] == ' ' || mime[len - 1] == '\t')) len--;
184 memcpy(out, mime, len);
185 out[len] = '\0';
186 }

188 bool selectDecoder(const char* mime) {
189 bool result = false;
190 if (mime == nullptr) return false;
191 // do nothing if no change
192 if (StrView(mime).equalsIgnoreCase(actual_decoder.mime)) {
193 is_first = false;
194 return true;
195 }
196 // close actual decoder
197 if (actual_decoder.decoder != this) end();
198
199 // A source may report a MIME type with an RFC 6381-style "codecs"
200 // (or other) parameter, e.g. "audio/mpeg; codecs=\"mpeg1-layer2\"" -
201 // a decoder registered for that exact parameterized type (a more
202 // specific match) must win over one only registered for the plain
203 // base type ("audio/mpeg"). Two passes, not one combined check: a
204 // single pass stops at the first entry that matches *either* way, so
205 // a generic base-type decoder registered before the specific one
206 // would otherwise shadow it purely by list order.
207 char base[64];
208 baseMimeType(mime, base, sizeof(base));
209 int exact_j = -1, base_j = -1;
210 for (int j = 0; j < decoders.size(); j++) {
211 if (StrView(decoders[j].mime).equalsIgnoreCase(mime)) { exact_j = j; break; }
212 if (base_j < 0 && StrView(decoders[j].mime).equalsIgnoreCase(base)) base_j = j;
213 }
214 int match_j = exact_j >= 0 ? exact_j : base_j;
215
216 // find the corresponding decoder
217 selected_mime = nullptr;
218 if (match_j >= 0) {
219 DecoderInfo info = decoders[match_j];
220 LOGI("Using decoder for %s (%s)", info.mime, mime);
221 actual_decoder = info;
222 // define output if it has not been defined
223 if (p_print != nullptr && actual_decoder.decoder != this
224 && actual_decoder.decoder->getOutput() == nullptr) {
225 actual_decoder.decoder->setOutput(*p_print);
226 }
227 if (!*actual_decoder.decoder) {
228 actual_decoder.decoder->begin();
229 LOGI("Decoder %s started", actual_decoder.mime);
230 }
231 result = true;
232 selected_mime = mime;
233 }
234 is_first = false;
235 return result;
236 }

260 size_t write(const uint8_t* data, size_t len) override {
261 if (is_first) {
262 const char* mime = nullptr;
263 if (p_mime_source != nullptr) {
264 // get content type from http header
265 mime = p_mime_source->mime();
266 if (mime) LOGI("mime from http request: %s", mime);
267 }
268 if (mime == nullptr) {
269 // use the mime detector
270 mime_detector.write((uint8_t*)data, len);
271 mime = mime_detector.mime();
272 if (mime) LOGI("mime from mime_detector: %s", mime);
273 }
274 if (mime != nullptr) {
275 // select the decoder based on the detemined mime type
276 if (!selectDecoder(mime)) {
277 LOGE("The decoder could not be found for %s", mime);
278 actual_decoder.decoder = &nop;
279 actual_decoder.is_open = true;
280 }
281 }
282 is_first = false;
283 }
284 // check if we have a decoder
285 if (actual_decoder.decoder == nullptr) return 0;
286 // decode the data
287 return actual_decoder.decoder->write(data, len);
288 }
