11// ── Platform-specific default USB endpoint numbers ──────────────────────────
12// Some MCUs' USB peripherals restrict isochronous transfers to one hardwired
13// endpoint number instead of allowing the stack to allocate any free one.
14//
15// nRF52833/nRF52840 caveats:
16// - The USBD peripheral has exactly one isochronous endpoint pair, hardwired
17// to endpoint number 8 (tinyusb dcd_nrf5x.c: EP_ISO_NUM == 8). Opening an
18// isochronous endpoint at any other number hits
19// TU_ASSERT(epnum == EP_ISO_NUM) and the device stops responding -- this
20// is why ep_in/ep_out/ep_fb are pinned here instead of using
21// TinyUSBDevice.allocEndpoint() like on RP2040/SAMD/STM32.
22// - ep_in and ep_fb both resolve to 0x88 (the same physical IN endpoint),
23// which is safe because they're mutually exclusive: USB_AUDIO_MAX_ISO_EP
24// is 2, so enableFeedbackEp() (see USBAudio2DescriptorBuilder) is false
25// in RXTX mode and enable_feedback_ep_rxtx has no effect here, matching
26// the hardware's single ISO IN + single ISO OUT pair exactly.
27// - ep_int (the AC status/change interrupt endpoint) is NOT restricted --
28// endpoints 1-7 (control/bulk/interrupt) are freely assignable on this
29// chip -- so it keeps the generic -1 default below and is still allocated
30// dynamically even on nRF52.
31// - Because the fixed 0x88/0x08 numbers bypass allocEndpoint() entirely,
32// TinyUSBDevice's internal endpoint counters don't know endpoint 8 is
33// taken. Audio + a single CDC Serial (the common composite case) is safe
34// since CDC only needs a handful of low-numbered endpoints, but stacking
35// many other USB classes alongside Audio could in theory let one of them
36// also land on endpoint 8 and collide -- an edge case, not a concern for
37// a simple Audio+CDC composite device.
38#if defined(NRF52840_XXAA) || defined(NRF52833_XXAA)
39#ifndef USB_AUDIO_EP_IN
40#define USB_AUDIO_EP_IN 0x88
41#endif
42#ifndef USB_AUDIO_EP_OUT
43#define USB_AUDIO_EP_OUT 0x08
44#endif
45#ifndef USB_AUDIO_EP_FB
46#define USB_AUDIO_EP_FB 0x88
47#endif
48#ifndef USB_AUDIO_MAX_ISO_EP
49#define USB_AUDIO_MAX_ISO_EP 2
50#endif
51#endif

53// USBAudioDeviceESP32 (arduino-esp32) previously pinned 0x83/0x03/0x84/0x85
54// here under the assumption that CDC uses 0x81/0x82/0x02. That assumption is
55// wrong: arduino-esp32's own USBCDC::load_cdc_descriptor() (USBCDC.cpp)
56// hardcodes notification=0x85, OUT=0x03, IN=0x84 -- i.e. exactly the "safe"
57// defaults this block used to define -- so a composite CDC + Audio device
58// silently collided on three endpoints (confirmed independently: Adafruit's
59// own TinyUSBDevice::allocEndpoint() carries a matching ESP32 workaround
60// that explicitly steers around 0x03/0x84/0x85 for the same reason). Rather
61// than guess another set of "safe" fixed numbers, ESP32 now falls through to
62// the generic -1 ("allocate dynamically") default below: USBAudioDeviceESP32
63// resolves it via arduino-esp32's own tinyusb_get_free_in_endpoint() /
64// tinyusb_get_free_out_endpoint(), which already know about CDC's reserved
65// endpoints. See https://github.com/pschatzmann/arduino-audio-tools/issues/2396.
66
67// Arduino UNO R4 / Nano R4 (Renesas RA4M1, arduino:renesas_uno core) caveats:
68// - This core's USB.cpp builds one fixed composite descriptor (CDC + HID +
69// MSD) at runtime and has no dynamic endpoint allocator API (no
70// TinyUSBDevice.allocEndpoint(), no tinyusb_get_free_in_endpoint()), so
71// endpoint numbers must be fixed constants here, same as nRF52 above.
72// - Values must avoid the ones the core's own USB.cpp hardcodes for its
73// built-in interfaces: CDC uses 0x81 (notification), 0x02/0x82 (data);
74// HID uses 0x83; MSD uses 0x04/0x84 (see __USBGetCustomInterfaceDescriptor
75// in USB.h for the authoritative list). EP5-EP7 are free on every
76// variant since none of the built-ins use them.
77// - The RA4M1's RUSB2 USB peripheral (dcd_rusb2.c) only has 2 isochronous-
78// capable pipes, so at most 2 ISO endpoints can be open at once
79// (USB_AUDIO_MAX_ISO_EP). TX-only, RX-only-with-feedback, and
80// RXTX-without-feedback each need 2 or fewer ISO endpoints; RXTX with
81// feedback needs 3, so enable_feedback_ep_rxtx has no effect here.
82#if defined(ARDUINO_ARCH_RENESAS)
83#ifndef USB_AUDIO_EP_IN
84#define USB_AUDIO_EP_IN 0x85
85#endif
86#ifndef USB_AUDIO_EP_OUT
87#define USB_AUDIO_EP_OUT 0x05
88#endif
89#ifndef USB_AUDIO_EP_FB
90#define USB_AUDIO_EP_FB 0x86
91#endif
92#ifndef USB_AUDIO_EP_INT
93#define USB_AUDIO_EP_INT 0x87
94#endif
95#ifndef USB_AUDIO_MAX_ISO_EP
96#define USB_AUDIO_MAX_ISO_EP 2
97#endif
98#endif
