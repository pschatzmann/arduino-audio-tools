156 bool openStream(uint8_t daddr, const UsbEndpointDescriptorView& ep,
157 UsbHostStreamRole role) override {
158 if (dev_ == nullptr || daddr != daddr_) return false;
159 Stream* s = findStream(ep.bEndpointAddress);
160 if (s == nullptr) {
161 streams_.push_back(Stream());
162 s = &streams_.back();
163 }
164 if (s->in_flight > 0) {
165 LOGE("Endpoint 0x%x is still busy", ep.bEndpointAddress);
166 return false;
167 }
168 freeTransfers(*s);
169 s->ep_addr = ep.bEndpointAddress;
170 s->role = role;
171 s->max_packet = ep.packetSize();
172 s->packets = role == UsbHostStreamRole::Feedback ? 1 : kPacketsPerTransfer;
173 for (int j = 0; j < kTransfersInFlight; j++) {
174 usb_transfer_t* xfer = nullptr;
175 esp_err_t rc = usb_host_transfer_alloc(
176 (size_t)s->max_packet * s->packets, s->packets, &xfer);
177 if (rc != ESP_OK) {
178 LOGE("usb_host_transfer_alloc: %s", esp_err_to_name(rc));
179 freeTransfers(*s);
180 return false;
181 }
182 xfer->device_handle = dev_;
183 xfer->bEndpointAddress = ep.bEndpointAddress;
184 xfer->callback = isoComplete;
185 xfer->context = this;
186 s->transfers.push_back(xfer);
187 }
188 s->open = true;
189 for (auto xfer : s->transfers) submit(*s, xfer);
190 return true;
191 }

281 static void clientEvent(const usb_host_client_event_msg_t* msg, void* arg) {
282 USBAudioHostBackendESP32* self = (USBAudioHostBackendESP32*)arg;
283 switch (msg->event) {
284 case USB_HOST_CLIENT_EVENT_NEW_DEV:
285 self->openDevice(msg->new_dev.address);

301 void openDevice(uint8_t address) {
302 if (dev_ != nullptr || events_ == nullptr) return;
303 usb_device_handle_t dev = nullptr;
304 if (usb_host_device_open(client_, address, &dev) != ESP_OK) return;
305 const usb_device_desc_t* dev_desc = nullptr;
306 const usb_config_desc_t* cfg_desc = nullptr;
307 if (usb_host_get_device_descriptor(dev, &dev_desc) != ESP_OK ||
308 usb_host_get_active_config_descriptor(dev, &cfg_desc) != ESP_OK) {
309 usb_host_device_close(client_, dev);
310 return;
311 }
312 const uint8_t* start = (const uint8_t*)cfg_desc;
313 const uint8_t* end = start + cfg_desc->wTotalLength;
314 // the class logic sends control requests to dev_
315 dev_ = dev;
316 daddr_ = address;
317 bool accepted = false;
318 for (const uint8_t* p = start; p + 2 <= end && p[0] >= 2; p += p[0]) {
319 if (p[1] != kUsbDescTypeInterface || p[0] < 9) continue;
320 if (p[5] != kUsbClassAudio || p[6] != kAudioSubclassControl) continue;
321 uint16_t len = events_->onAttach(address, dev_desc->idVendor,
322 dev_desc->idProduct, p,
323 (uint16_t)(end - p));
324 if (len > 0) {
325 accepted = true;
326 break;
327 }
328 }
329 if (!accepted) {
330 dev_ = nullptr;
331 daddr_ = 0;
332 usb_host_device_close(client_, dev);
333 return;
334 }
335 // ESP-IDF has already selected the configuration
336 events_->onConfigured(address);
337 }

397 void submit(Stream& s, usb_transfer_t* xfer) {
398 int total = 0;
399 for (int j = 0; j < s.packets; j++) {
400 int len = s.max_packet;
401 if (s.role == UsbHostStreamRole::TxData) {
402 len = events_ ? events_->onIsoOut(xfer->data_buffer + total,
403 s.max_packet)
404 : 0;
405 }
406 xfer->isoc_packet_desc[j].num_bytes = len;
407 total += len;
408 }
409 xfer->num_bytes = total;
410 if (usb_host_transfer_submit(xfer) == ESP_OK) {
411 s.in_flight++;

417 static void isoComplete(usb_transfer_t* xfer) {
418 USBAudioHostBackendESP32* self = (USBAudioHostBackendESP32*)xfer->context;
419 Stream* s = self->findStream(xfer);
420 if (s == nullptr) return;
421 s->in_flight--;
422 if (!s->open || self->dev_ == nullptr) {
423 if (s->in_flight == 0 && self->dev_ == nullptr) self->freeTransfers(*s);
424 return;
425 }
426 if (s->role != UsbHostStreamRole::TxData && self->events_ &&
427 xfer->status == USB_TRANSFER_STATUS_COMPLETED) {
428 // IN: packet j starts at j * max_packet
429 for (int j = 0; j < s->packets; j++) {
430 usb_isoc_packet_desc_t& pd = xfer->isoc_packet_desc[j];
431 if (pd.status != USB_TRANSFER_STATUS_COMPLETED) continue;
432 self->events_->onIsoIn(s->role, xfer->data_buffer + j * s->max_packet,
433 (uint16_t)pd.actual_num_bytes);
434 }
435 }
436 if (xfer->status == USB_TRANSFER_STATUS_NO_DEVICE) return;
437 self->submit(*s, xfer);
438 }

440 static void controlComplete(usb_transfer_t* xfer) {
441 USBAudioHostBackendESP32* self = (USBAudioHostBackendESP32*)xfer->context;
442 bool ok = xfer->status == USB_TRANSFER_STATUS_COMPLETED;
443 uint16_t len = 0;
444 if (ok && xfer->actual_num_bytes > 8) {
445 len = (uint16_t)(xfer->actual_num_bytes - 8);
446 if (len > self->ctrl_len_) len = self->ctrl_len_;
447 if (self->ctrl_data_) memcpy(self->ctrl_data_, xfer->data_buffer + 8, len);
448 }
449 if (self->events_) self->events_->onControlComplete(ok, len);
450 }
