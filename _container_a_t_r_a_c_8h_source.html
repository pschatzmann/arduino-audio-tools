65 size_t write(const uint8_t *data, size_t len) override {
66 if (!is_active || is_failed) return 0;
67 const uint8_t *p = data;
68 size_t n = len;
69 while (n > 0 && !is_failed && !is_done) {
70 if (skip_count > 0) {
71 size_t s = skip_count < n ? skip_count : n;
72 p += s;
73 n -= s;
74 skip_count -= s;
75 } else if (is_streaming) {
76 writePayload(p, n);
77 } else {
78 int state_before = state;
79 size_t n_before = n;
80 parse(p, n);
81 if (!is_failed && state == state_before && n == n_before &&
82 !is_streaming) {
83 LOGE("ATRAC header parser is not progressing");
84 is_failed = true;
85 }
86 }
87 }
88 return is_failed ? 0 : len;
89 }

108 bool collect(const uint8_t *&p, size_t &n, size_t need) {
109 while ((size_t)hdr.size() < need && n > 0) {
110 uint8_t value = *p;

112 p++;
113 n--;
114 }

124 bool startDecoder(ATRACType type, int sample_rate, int channels,
125 int block_align, const uint8_t *extra = nullptr,
126 size_t extra_size = 0, bool known = false,
127 uint32_t length = 0) {
128 if (!dec.begin(type, sample_rate, channels, block_align, extra,
129 extra_size)) {
130 fail("ATRAC decoder could not be started");
131 return false;
132 }
133 setAudioInfo(dec.audioInfo());
134 payload_known = known;
135 payload_left = length;
136 is_streaming = true;
137 if (known && length == 0) is_done = true;
138 return true;
139 }
140
141 void writePayload(const uint8_t *&p, size_t &n) {
142 size_t chunk = n;
143 if (payload_known && chunk > payload_left) chunk = payload_left;
144 if (chunk > 0) dec.write(p, chunk);
145 p += chunk;
146 n -= chunk;
147 if (payload_known) {
148 payload_left -= chunk;
149 // ignore everything which follows the audio data (e.g. other chunks)
150 if (payload_left == 0) is_done = true;
151 }
152 }

154 static uint16_t be16(const uint8_t *p) { return (p[0] << 8) | p[1]; }
155 static uint32_t be24(const uint8_t *p) {
156 return (p[0] << 16) | (p[1] << 8) | p[2];
157 }

169class ContainerATRACOMA : public ContainerATRAC {
170 public:
171 const char *mime() override { return "audio/oma"; }
172
173 protected:
174 enum State { Start = 0, Ea3 };
175 static const size_t ID3_HEADER_SIZE = 10;
176 static const size_t EA3_HEADER_SIZE = 96;
177 enum CodecId { CodecATRAC3 = 0, CodecATRAC3Plus = 1 };
178
179 void parse(const uint8_t *&p, size_t &n) override {
180 switch (state) {
181 case Start:
182 if (!collect(p, n, ID3_HEADER_SIZE)) return;
183 if (memcmp(hdr.data(), "ea3", 3) == 0 ||
184 memcmp(hdr.data(), "ID3", 3) == 0) {
185 size_t tag_size = syncSafe(hdr.data() + 6);
186 // footer
187 if (hdr.data()[5] & 0x10) tag_size += 10;
188 skip_count = tag_size;
189 hdr.clear();
190 } else if (memcmp(hdr.data(), "EA3", 3) != 0) {
191 fail("Not an OMA stream");
192 return;
193 }
194 // an EA3 header without the tag is completed in the next state
195 state = Ea3;
196 break;
197
198 case Ea3:
199 if (!collect(p, n, EA3_HEADER_SIZE)) return;
200 parseEa3();
201 break;
202 }
203 }
204
205 static size_t syncSafe(const uint8_t *p) {
206 return ((p[0] & 0x7F) << 21) | ((p[1] & 0x7F) << 14) |
207 ((p[2] & 0x7F) << 7) | (p[3] & 0x7F);
208 }
209
210 void parseEa3() {
211 const uint8_t *h = hdr.data();
212 if (memcmp(h, "EA3", 3) != 0 || h[4] != 0 || h[5] != EA3_HEADER_SIZE) {
213 fail("OMA: EA3 header not found");
214 return;
215 }
216 // 0xFFFF and 0xFF80: not encrypted
217 uint16_t encryption_id = be16(h + 6);
218 if (encryption_id != 0xFFFF && encryption_id != 0xFF80) {
219 fail("OMA: encrypted files are not supported");
220 return;
221 }
222 uint8_t codec_id = h[32];
223 uint32_t params = be24(h + 33);
224 static const int rates[] = {32000, 44100, 48000, 88200, 96000};
225 unsigned rate_idx = (params >> 13) & 7;
226 if (rate_idx >= sizeof(rates) / sizeof(rates[0])) {
227 fail("OMA: invalid sample rate");
228 return;
229 }
230 int sample_rate = rates[rate_idx];
231 int frame_size = (params & 0x3FF) * 8;
232
233 if (codec_id == CodecATRAC3) {
234 // joint stereo flag: fake the extradata of the WAV format
235 uint16_t js = (params >> 17) & 1;
236 uint8_t extra[14] = {1, 0, 0, 0x10, 0, 0, (uint8_t)js, 0,
237 (uint8_t)js, 0, 1, 0, 0, 0};
238 hdr.clear();
239 startDecoder(ATRAC3, sample_rate, 2, frame_size, extra, sizeof(extra));
240 } else if (codec_id == CodecATRAC3Plus) {
241 static const int channel_ids[] = {0, 1, 2, 3, 4, 6, 7, 8};
242 int channels = channel_ids[(params >> 10) & 7];
243 if (channels == 0) {
244 fail("OMA: invalid channel configuration");
245 return;
246 }
247 hdr.clear();
248 startDecoder(ATRAC3PLUS, sample_rate, channels, frame_size + 8);
249 } else {
250 fail("OMA: unsupported codec (only ATRAC3 and ATRAC3+)");
251 }
252 }
253};

272 void parse(const uint8_t *&p, size_t &n) override {
273 if (!collect(p, n, CHANNELS_POS + 1)) return;
274 static const uint8_t magic[4] = {0x00, 0x08, 0x00, 0x00};
275 if (memcmp(hdr.data(), magic, 4) != 0) {
276 fail("Not an AEA stream");
277 return;

280 hdr.clear();

283 return;
284 }
285 startDecoder(ATRAC1, 44100, channels, SOUND_UNIT_SIZE * channels);
286 // the rest of the header is not needed
