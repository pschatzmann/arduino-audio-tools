189 void decodeFrame() {

192 if (samples <= 0) {
193 // the first frame(s) of some streams do not produce any output
194 LOGD("ATRAC decode: no output (error: %d)", p_decoder->lastError());
195 return;
196 }
197 if (p_print == nullptr) return;
198 size_t open = samples * p_decoder->channels() * sizeof(int16_t);
199 uint8_t *ptr = (uint8_t *)pcm.data();
200 while (open > 0) {
201 size_t written = p_print->write(ptr, open);
202 if (written == 0) {
203 LOGE("write error: %d bytes lost", (int)open);
204 break;
205 }
206 ptr += written;

213 static int atrac9FrameSize(const uint8_t *extra, size_t size) {
214 if (extra == nullptr || size != 12) return 0;
215 const uint8_t *e = extra + 4;
216 int frame_size = ((e[2] << 3) | (e[3] >> 5)) + 1;
217 int superframe = (e[3] >> 3) & 0x3;
