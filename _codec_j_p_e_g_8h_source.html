199 bool writeBlock(int16_t x, int16_t y, uint16_t w, uint16_t h,
200 uint16_t *data) {
201 if (x < 0 || y < 0) return true;
202 uint32_t max_x = width_;
203 uint32_t max_y = height_;
204 if ((uint32_t)x >= max_x || (uint32_t)y >= max_y) return true;
205 uint32_t win_w = std::min((uint32_t)w, max_x - (uint32_t)x);
206 uint32_t win_h = std::min((uint32_t)h, max_y - (uint32_t)y);
207 uint16_t *dst = (uint16_t *)frame_buffer.data();
208 for (uint32_t row = 0; row < win_h; row++) {
209 memcpy(dst + ((uint32_t)y + row) * max_x + (uint32_t)x,
210 data + row * w, win_w * sizeof(uint16_t));
211 }
212 return true;
213 }
