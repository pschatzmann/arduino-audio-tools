141 void flush() override {
142 if (pos == 0) return;
143 if (p_out == nullptr && p_out_video == nullptr) {
144 pos = 0;
145 return;
146 }
147 uint16_t w = 0, h = 0;
148 if (decoder_.getJpgSize(&w, &h, img_vector.data(), pos) != JDR_OK ||
149 w == 0 || h == 0) {
150 LOGE("MJPEGDecoder: could not determine JPEG size");
151 pos = 0;
152 return;
153 }
154 width_ = w;
155 height_ = h;
156 size_t needed = (size_t)width_ * height_ * 2;
157 if (frame_buffer.size() < needed) {
158 frame_buffer.resize(needed);
159 if (frame_buffer.data() == nullptr) {
160 LOGE("MJPEGDecoder: frame buffer allocation failed");
161 pos = 0;
162 return;
163 }
164 }
165 if (decoder_.drawJpg(0, 0, img_vector.data(), pos) == JDR_OK) {
166 writeToOutput(frame_buffer.data(), needed);
167 }
168 pos = 0;
169 }

210 bool writeBlock(int16_t x, int16_t y, uint16_t w, uint16_t h,
211 uint16_t *data) {
212 if (x < 0 || y < 0) return true;
213 uint32_t max_x = width_;
214 uint32_t max_y = height_;
215 if ((uint32_t)x >= max_x || (uint32_t)y >= max_y) return true;
216 uint32_t win_w = std::min((uint32_t)w, max_x - (uint32_t)x);
217 uint32_t win_h = std::min((uint32_t)h, max_y - (uint32_t)y);
218 uint16_t *dst = (uint16_t *)frame_buffer.data();
219 for (uint32_t row = 0; row < win_h; row++) {
220 memcpy(dst + ((uint32_t)y + row) * max_x + (uint32_t)x,
221 data + row * w, win_w * sizeof(uint16_t));
222 }
223 return true;
224 }

228 if (p_out != nullptr) result = p_out->write(data, len);
229 if (p_out_video != nullptr) result += p_out_video->write(data, len);
230 return result;

324 size_t write(const uint8_t *data, size_t len) override {
325 if (p_out == nullptr && p_out_video == nullptr) return 0;
326 size_t expected = videoFrameSizeBytes(input_format, width_, height_);
327 if (expected == 0 || len < expected) {
328 LOGE("MJPEGEncoder: %u bytes, expected %u", (unsigned)len,
329 (unsigned)expected);
330 return 0;

332 if (jpeg.size() == 0) jpeg.resize(kDefaultBufferSize);
333 size_t size = 0;
334 JERESULT rc = encode(data, size);
335 // the image didn't fit: grow the buffer and try again
336 for (int attempt = 0; rc == JER_INTR && attempt < kMaxGrowRetries;
337 ++attempt) {
338 jpeg.resize(jpeg.size() * 2);
339 if (jpeg.data() == nullptr) {
340 LOGE("MJPEGEncoder: buffer allocation failed");
341 return 0;
342 }
343 rc = encode(data, size);
344 }
345 if (rc != JER_OK) {
346 LOGE("MJPEGEncoder: encoding failed (%d)", (int)rc);
347 return 0;
348 }
349 if (p_out != nullptr) p_out->write(jpeg.data(), size);
350 if (p_out_video != nullptr) p_out_video->write(jpeg.data(), size);
