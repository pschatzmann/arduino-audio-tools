164 void decodeToFrameBuffer() {
165 uint16_t mcu_w = JpegDec.MCUWidth;
166 uint16_t mcu_h = JpegDec.MCUHeight;
167 uint32_t max_x = JpegDec.width;
168 uint32_t max_y = JpegDec.height;
169 size_t needed = (size_t)max_x * max_y * 2;
170 if (frame_buffer.size() < needed) {
171 frame_buffer.resize(needed);
172 if (frame_buffer.data() == nullptr) {
173 LOGE("MJPEGDecoder: frame buffer allocation failed");
174 return;
175 }
176 }
177 uint16_t *dst = (uint16_t *)frame_buffer.data();
178 while (JpegDec.read()) {
179 uint16_t *src = JpegDec.pImage;
180 uint32_t mcu_x = JpegDec.MCUx * mcu_w;
181 uint32_t mcu_y = JpegDec.MCUy * mcu_h;
182 uint32_t win_w = std::min((uint32_t)mcu_w, max_x - mcu_x);
183 uint32_t win_h = std::min((uint32_t)mcu_h, max_y - mcu_y);
184 for (uint32_t row = 0; row < win_h; row++) {
185 uint16_t *d = dst + (mcu_y + row) * max_x + mcu_x;
186 uint16_t *s = src + row * mcu_w;
187 if (byte_swap) {
188 for (uint32_t col = 0; col < win_w; col++) {
189 uint16_t v = s[col];
190 d[col] = (uint16_t)((v << 8) | (v >> 8));
191 }
192 } else {
193 memcpy(d, s, win_w * sizeof(uint16_t));
194 }
195 }
196 }
197 }
