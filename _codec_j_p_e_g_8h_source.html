207 bool writeBlock(int16_t x, int16_t y, uint16_t w, uint16_t h,
208 uint16_t *data) {
209 if (x < 0 || y < 0) return true;
210 uint32_t max_x = width_;
211 uint32_t max_y = height_;
212 if ((uint32_t)x >= max_x || (uint32_t)y >= max_y) return true;
213 uint32_t win_w = std::min((uint32_t)w, max_x - (uint32_t)x);
214 uint32_t win_h = std::min((uint32_t)h, max_y - (uint32_t)y);
215 uint16_t *dst = (uint16_t *)frame_buffer.data();
216 for (uint32_t row = 0; row < win_h; row++) {
217 memcpy(dst + ((uint32_t)y + row) * max_x + (uint32_t)x,
218 data + row * w, win_w * sizeof(uint16_t));
219 }
220 return true;
221 }
