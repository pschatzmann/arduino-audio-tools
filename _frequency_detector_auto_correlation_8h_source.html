270 template <class T>
271 float detectFrequencyForChannel(int ch, T* samples, size_t len,
272 float& confidenceOut) {
273 LOGD("detectFrequencyForChannel: %d / len: %u", ch, (unsigned int)len);
274 confidenceOut = 0.0f;
275 // Prepare variables for autocorrelation
276 int sample_rate = info.sample_rate;
277 int channels = info.channels;
278 int buffer_size = len / info.channels;
279
280 // Autocorrelation lag range: 1000 Hz max, 50 Hz min
281 size_t min_lag = sample_rate / 1000;
282 size_t max_lag = sample_rate / 50;
283 if (max_lag >= buffer_size) max_lag = buffer_size - 1;
284
285 LOGD("lag min/max: %u / %u", (unsigned)min_lag, (unsigned)max_lag);
286
287 // Energy at lag 0 - used to normalize the correlation peak into a
288 // loudness-independent confidence score
289 double energy = 0.0;
290 for (int i = 0; i < buffer_size; ++i) {
291 double s = (double)samples[i * channels];
292 energy += s * s;
293 }
294
295 double max_corr = 0.0;
296 size_t best_lag = 0;
297 for (size_t lag = min_lag; lag < max_lag; ++lag) {
298 double sum = 0.0;
299 for (size_t i = 0; i < buffer_size - lag; ++i) {
300 sum += (double)samples[i * channels] * (double)samples[(i + lag) * channels];
301 }
302 if (sum > max_corr) {
303 max_corr = sum;
304 best_lag = lag;
305 }
306 }
307
308 LOGD("best_lag: %u / max_corr: %f", (unsigned)best_lag, max_corr);
309
310 if (energy > 0.0) {
311 confidenceOut = (float)(max_corr / energy);
312 if (confidenceOut > 1.0f) confidenceOut = 1.0f;
313 if (confidenceOut < 0.0f) confidenceOut = 0.0f;
314 }
315
316 if (best_lag == 0) return 0.0f;
317 return (float)sample_rate / best_lag;
318 }
