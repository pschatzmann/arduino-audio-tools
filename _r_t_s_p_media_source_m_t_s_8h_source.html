127 void addPacket() {
128 if (!is_active) return; // no client is streaming
129 int pid = ((packet[1] & 0x1F) << 8) | packet[2];
130 bool pusi = packet[1] & 0x40;
131 if (pusi && isVideoPes()) video_pid = pid;
132 bool is_video = pid == video_pid;
133 int free = buffer.availableForWrite();
134
135 // too slow: drop video from the start of the next frame on, until the
136 // buffer has recovered and the next keyframe (preceded by the PAT)
137 if (dropping_video) {
138 if (pid == 0 && free >= capacity / 2) {
139 dropping_video = false;
140 LOGI("RTSPMediaSourceMTS: resuming video at the next keyframe");
141 }
142 } else if (is_video && pusi && free < capacity / 4) {
143 dropping_video = true;
144 LOGW("RTSPMediaSourceMTS: network too slow - dropping video");
145 }
146 if (dropping_video && is_video) {
147 dropped_video++;
148 return;
149 }
150
151 // only whole packets: the free space can only grow while we write
152 if (free < kPacketSize) {
153 if (dropped++ % 100 == 0) {
154 LOGW("RTSPMediaSourceMTS: buffer full, %u packets dropped",
155 (unsigned)dropped);
156 }
157 return;
158 }
159 buffer.writeArray(packet, kPacketSize);
160 }
161
164 bool isVideoPes() {
165 int afc = (packet[3] >> 4) & 3;
166 if (!(afc & 1)) return false; // no payload
167 int pos = (afc & 2) ? 5 + packet[4] : 4;
168 if (pos + 4 > kPacketSize) return false;
169 return packet[pos] == 0 && packet[pos + 1] == 0 && packet[pos + 2] == 1 &&
170 (packet[pos + 3] & 0xF0) == 0xE0;
171 }
