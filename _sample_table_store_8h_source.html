130template <typename T, size_t ChunkSize = 4096>
131class ChunkedSampleTableStore : public SampleTableStore<T> {
132 public:
133 ~ChunkedSampleTableStore() { clear(); }
134
135 void clear() override {
136 for (size_t i = 0; i < chunks.size(); i++) delete[] chunks[i];
137 chunks.clear();
138 count = 0;
139 }
140
141 void append(T value) override {
142 size_t chunk_idx = count / ChunkSize;
143 size_t local_pos = count % ChunkSize;
144 if (chunk_idx >= (size_t)chunks.size()) chunks.push_back(new T[ChunkSize]);
145 chunks[(int)chunk_idx][local_pos] = value;
146 count++;
147 }
148
149 size_t size() override { return count; }
150
151 T get(size_t index) override {
152 return chunks[(int)(index / ChunkSize)][index % ChunkSize];
153 }
154
155 protected:
156 Vector<T*> chunks; // small - one pointer per ChunkSize entries, not one per entry
157 size_t count = 0;
158};

209template <typename FileT, typename WriterT>
210class FileSeekableSource : public SeekableSource {
211 public:
217 FileSeekableSource(FileT& file, WriterT& writer, int bufferSize = 1024,
218 bool quickStart = true)
219 : p_file(&file),
220 p_writer(&writer),
221 buffer_len(bufferSize),
222 quick_start_enabled(quickStart) {
223 writer.setSeekSource(*this);
224 }
225
226 bool seek(size_t pos) override {
227 saved_pos = p_file->position();
228 bool ok = p_file->seek(pos);
229 if (debug_trace)
230 LOGI("FileSeekableSource: seek(%u) saved_pos=%u post-seek position()=%u ok=%d",
231 (unsigned)pos, (unsigned)saved_pos, (unsigned)p_file->position(), (int)ok);
232 return ok;
233 }
234
235 size_t readBytes(uint8_t* data, size_t len) override {
236 // File::readBytes() on some platforms (incl. this project's desktop
237 // emulator) takes char*, not uint8_t* - readBytes(uint8_t*,size_t) is
238 // the more common Arduino Stream signature, but not universal.
239 size_t pre_pos = p_file->position();
240 size_t n = p_file->readBytes((char*)data, len);
241 if (debug_trace)
242 LOGI("FileSeekableSource: readBytes len=%u pre_pos=%u got=%u restoring to saved_pos=%u",
243 (unsigned)len, (unsigned)pre_pos, (unsigned)n, (unsigned)saved_pos);
244 p_file->seek(saved_pos); // restore the sequential-read position
245 return n;
246 }
247
252 bool debug_trace = false;
253
264 size_t copy() {
265 if (quick_start_enabled && !quick_start_done) {
266 quick_start_done = true;
267 quickstart_detail::callQuickStartIfAvailable(*p_writer);
268 }
269 uint8_t buffer[buffer_len];
270 size_t to_read = p_file->readBytes((char*)buffer, buffer_len);
271 if (to_read == 0) return 0;
272 size_t written = 0;
273 int attempts = 0;
274 while (written < to_read) {
275 size_t w = p_writer->write(buffer + written, to_read - written);
276 attempts++;
277 if (w == 0) {
278 LOGW("FileSeekableSource::copy(): writer stalled after accepting %u of %u "
279 "bytes (%d attempt(s)) - dropping the remaining %u bytes",
280 (unsigned)written, (unsigned)to_read, attempts, (unsigned)(to_read - written));
281 break;
282 }
283 written += w;
284 }
285 if (debug_trace && attempts > 1)
286 LOGI("FileSeekableSource::copy() needed %d attempts for to_read=%u", attempts, (unsigned)to_read);
287 return written;
288 }
289
290 protected:
291 FileT* p_file;
292 WriterT* p_writer;
293 int buffer_len;
294 size_t saved_pos = 0;
295 bool quick_start_enabled;
296 bool quick_start_done = false;
297};

358 cache_capacity(cacheEntries),

365 }
366
367 void clear() override {
368 count = 0;
369 start_offset = -1;
370 cache_len = 0;
371 }
372
373 void setNextEntryOffset(uint64_t fileOffset) override {
374 if (start_offset < 0) start_offset = (int64_t)fileOffset;
375 }
376
381 void setOnDiskEntrySize(size_t bytes) override {
382 on_disk_size = bytes;
383 cache_len = 0; // invalidate: any cached block used the old width
384 }
385
386 void append(T) override {
387 // value itself is intentionally discarded - only its position matters
388 count++;
389 }
390
391 size_t size() override { return count; }
392
393 T get(size_t index) override {
394 if (start_offset < 0 || p_source == nullptr) return T{};
395 if (index < cache_start || index >= cache_start + cache_len)
396 refillCache(index);
397 if (index >= cache_start && index < cache_start + cache_len)
398 return cache[index - cache_start];
399 return T{};
400 }
401
402 protected:
403 void refillCache(size_t index) {
404 p_source->seek((size_t)(start_offset + (int64_t)(index * on_disk_size)));
405 size_t got = p_source->readBytes(raw_buf, cache_capacity * on_disk_size);
406 size_t got_entries = got / on_disk_size;
407 for (size_t i = 0; i < got_entries; i++)
408 cache[i] = decode(raw_buf + i * on_disk_size);
409 cache_start = index;
410 cache_len = got_entries;
411 }
412
413 SeekableSource* p_source;
414 size_t on_disk_size;
415 T (*decode)(const uint8_t*);
416 int64_t start_offset = -1;
417 size_t count = 0;
418
419 size_t cache_capacity;
420 T* cache;
421 uint8_t* raw_buf;
422 size_t cache_start = 0;

484template <typename T>
485class SpoolFileSampleTableStore : public SampleTableStore<T> {
486 public:
489 SpoolFileSampleTableStore(SpoolStorage& spool) : p_spool(&spool) {}
490
491 void clear() override {
492 entry_count = 0;
493 if (p_spool != nullptr) p_spool->seek(0);
494 }
495
496 void append(T value) override {
497 if (p_spool == nullptr) return;
498 p_spool->seek(entry_count * sizeof(T));
499 p_spool->write((const uint8_t*)&value, sizeof(T));
500 entry_count++;
501 }
502
503 size_t size() override { return entry_count; }
504
505 T get(size_t index) override {
506 T value{};
507 if (p_spool == nullptr) return value;
508 p_spool->seek(index * sizeof(T));
509 p_spool->readBytes((uint8_t*)&value, sizeof(T));
510 return value;
511 }
512
513 protected:
514 SpoolStorage* p_spool;
515 size_t entry_count = 0;
516};
