148 void recalculatePlayTime() {
149 if (upPercent + downPercent > 100) {
150 downPercent = 100 - upPercent;
151 }
152 playSamples = info.sample_rate / 1000 * playMs;
153 upSamples = (playSamples * upPercent) / 100;
154 rampDownSamples = (playSamples * downPercent) / 100;

165 size_t readBytesFrames(uint8_t* buffer, size_t lengthBytes, int frames,
166 int channels) {
167 T* result_buffer = (T*)buffer;
168 int frames_written = 0;
169 if (playMs > 0 && currentSample > playSamples) {
170 return 0;
171 }
172
173 for (int j = 0; j < frames; j++) {
174 T sample = readSample();
175
176 // if we requested a play time
177 if (playMs > 0) {
178 currentSample++;
179 sample = applyRamp(sample);
180 }
181
182 for (int ch = 0; ch < channels; ch++) {
183 *result_buffer++ = sample;
184 }
185
186 frames_written++;
187 // exit loop if we have reached the requested play time
188 if (playMs > 0 && currentSample > playSamples) {
189 break;
190 }
191 }
192 return frames_written * sizeof(T) * channels;
193 }
194
195 // Applies ramp up and ramp down logic to the sample
196 T applyRamp(T sample) {
197 // Ramp up
198 if (rampUpInc > 0 && currentSample <= upSamples) {
199 factor += rampUpInc;
200 if (factor > 1.0f) {
201 factor = 1.0f;
202 }
203 }
204 // Ramp down
205 else if (rampDownDec > 0 && currentSample >= playSamples - rampDownSamples) {
206 factor -= rampDownDec;
207 if (factor < 0.0f) {
208 factor = 0.0f;
209 }
210 }
211 // Sustain
212 else {
213 factor = 1.0f;
214 }
215 return (T)(factor * sample);
216 }
217
218 size_t readBytesFromBuffer(uint8_t* buffer, size_t lengthBytes,
219 int frame_size, int channels) {
220 // fill ringbuffer with one frame
221 if (ring_buffer.isEmpty()) {
222 uint8_t tmp[frame_size];
223 readBytesFrames(tmp, frame_size, 1, channels);
224 ring_buffer.writeArray(tmp, frame_size);
225 }
226 // provide result
227 return ring_buffer.readArray(buffer, lengthBytes);
228 }

777 unsigned int sum;
778

789 sum += white_values[i];
790 }
791 return sum;
792 }
