140 size_t write(const uint8_t* data, size_t len) {
141 if (!is_started_) return 0;
142 serviceUSB();
143
144 // disregard data if the TX stream has not been opened (alt=0)
145 if (!isStreamingActiveTx()) return len;
146
147 // update the volume
148 if (config_.volume_active) processVolume((uint8_t*)data, len);
149
150 // Write all data, retrying if the buffer is full. On single-core
151 // platforms (RP2040), serviceUSB() drains the buffer by running
152 // the USB stack task → xfer_cb. On dual-core (ESP32), the USB task
153 // drains independently and the RTOS buffer blocks internally.
154 size_t written = 0;
155 while (written < len) {
156 int n = bufferTx().writeArray(data + written, len - written);
157 written += n;
158 if (written < len) {
159 serviceUSB(); // drain buffer to make space
160 if (!isStreamingActiveTx()) break; // stream stopped
161 }
162 }
163 return written;
164 }

171 size_t readBytes(uint8_t* buffer, size_t bufsize) {
172 if (!is_started_) return 0;
173 serviceUSB();
174
175 if (!rx_primed_) {
176 uint8_t start_fill = config_.rx_start_fill_percent > 100
177 ? 100
178 : config_.rx_start_fill_percent;
179 if (start_fill > 0 && bufferRx().levelPercent() < start_fill) return 0;
180 rx_primed_ = true;
181 }
182
183 // get the data from the buffer
184 size_t ret = bufferRx().readArray(buffer, bufsize);
185 // upate the volume
186 if (config_.volume_active) processVolume(buffer, ret);
187
188 return ret;
189 }

264 void processVolume(uint8_t* data, size_t len) {
265 switch (config_.bits_per_sample) {
266 case 8:
267 processVolume<int8_t>((int8_t*)data, len);
268 break;
269 case 16:
270 processVolume<int16_t>((int16_t*)data, len / 2);
271 break;
272 case 24:
273 processVolume<int24_3bytes_t>((int24_3bytes_t*)data, len / 3);
274 break;
275 case 32:
276 processVolume<int32_t>((int32_t*)data, len / 4);
277 break;
278 default:
279 // Unsupported bit depth; do nothing.
280 break;
281 }
282 }

286 float getVolumeExt(uint8_t channel) const {
287 if (volume_.empty()) return 1.0f;
288 if (mute_[0]) return 0.0f; // master mute
289 float master = volume_[0];
290 if (channel >= volume_.size()) return master; // no per-channel entry
291 if (mute_[channel]) return 0.0f; // per-channel mute
292 return master * volume_[channel];
293 }

295 template <typename T>
296 void processVolume(T* data, size_t sample_count) {
297 uint8_t ch_count = config_.channels;
298 for (size_t i = 0; i < sample_count; i++) {
299 uint8_t ch = (uint8_t)(i % ch_count) + 1; // 1-based per-channel index
300 float vol = getVolumeExt(ch);
301 data[i] = (T)(data[i] * vol);
302 }
303 }

311 static int16_t floatToUac2(float vol) {
312 if (vol <= 0.0f) return (int16_t)0x8000;
313 if (vol >= 1.0f) return 0;
314 return (int16_t)((1.0f - vol) * kVolumeMinDb256);
315 }

319 static float uac2ToFloat(int16_t v) {
320 if (v == (int16_t)0x8000) return 0.0f;
321 if (v >= 0) return 1.0f;
322 if (v <= kVolumeMinDb256) return 0.0f;
323 return 1.0f - (float)v / (float)kVolumeMinDb256;
324 }
