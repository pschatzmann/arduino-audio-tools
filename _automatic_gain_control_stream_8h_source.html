284 void analyze(const T* data, size_t samples, float& rms, float& peak) {
285#if PREFER_FIXEDPOINT
286 constexpr int shift = AGCPCMTraits<T>::fixed_shift;
287 constexpr float max_sample = AGCPCMTraits<T>::max_sample;
288
289 int64_t sum_sq = 0;
290 int32_t peak_raw = 0;
291
292 for (size_t i = 0; i < samples; ++i) {
293 // int24_t's conversion operators are not const-qualified, so a
294 // local (non-const) copy is needed before the cast.
295 T sample = data[i];
296 int32_t value = static_cast<int32_t>(sample) >> shift;
297 int32_t abs_value = value < 0 ? -value : value;
298
299 if (abs_value > peak_raw) {
300 peak_raw = abs_value;
301 }
302
303 sum_sq += static_cast<int64_t>(value) * value;
304 }
305
306 float scaled_max = max_sample / static_cast<float>(1 << shift);
307
308 rms = samples == 0 ? 0.0f
309 : sqrtf(static_cast<float>(sum_sq) /
310 static_cast<float>(samples)) /
311 scaled_max;
312 peak = static_cast<float>(peak_raw) / scaled_max;
313#else
314 float sum = 0.0f;
315 peak = 0.0f;
316
317 constexpr float max_sample = AGCPCMTraits<T>::max_sample;
318
319 for (size_t i = 0; i < samples; ++i) {
320 // int24_t's conversion operators are not const-qualified, so a
321 // local (non-const) copy is needed before the cast.
322 T sample = data[i];
323 float value = static_cast<float>(sample) / max_sample;
324
325 float abs_value = fabsf(value);
326
327 sum += value * value;
328
329 if (abs_value > peak) {
330 peak = abs_value;
331 }
332 }
333
334 rms = samples == 0 ? 0.0f : sqrtf(sum / static_cast<float>(samples));
335#endif
336 }

370 float desiredGainDb(float rms_db, float peak_db) const {
371 /*
372 * Do not amplify silence.
373 *
374 * This is particularly important for fades and prevents
375 * background noise from being amplified after a song becomes
376 * quiet.
377 */
378 if (rms_db <= config.silence_db) {
379 return current_gain_db;
380 }
381
382 /*
383 * RMS normalization.
384 *
385 * Example:
386 *
387 * target = -18 dB
388 * measured = -24 dB
389 *
390 * desired = +6 dB
391 */
392 float desired = config.target_db - rms_db;
393
394 /*
395 * Automatic gain limits.
396 */
397 if (desired > config.max_gain_db) {
398 desired = config.max_gain_db;
399 }
400
401 if (desired < config.min_gain_db) {
402 desired = config.min_gain_db;
403 }
404
405 /*
406 * Peak protection.
407 *
408 * Never intentionally request enough gain to put the detected
409 * peak above the configured headroom.
410 */
411 float peak_after_gain = peak_db + desired;
412
413 if (peak_after_gain > config.peak_headroom_db) {
414 float peak_limited_gain = config.peak_headroom_db - peak_db;
415
416 if (peak_limited_gain < desired) {
417 desired = peak_limited_gain;
418 }
419 }
420
421 /*
422 * Peak limiting can have pushed the gain below our configured
423 * lower bound, so apply the minimum one final time.
424 */
425 if (desired < config.min_gain_db) {
426 desired = config.min_gain_db;
427 }
428
429 return desired;
430 }

432 void updateGain(float desired_gain_db, size_t samples) {
433 float coefficient;
434
435 if (desired_gain_db < current_gain_db) {
436 /*
437 * Input got louder.
438 *
439 * Reduce gain relatively quickly.
440 */
441 coefficient = smoothingCoefficient(config.attack_seconds, samples);
442
443 } else {
444 /*
445 * Input got quieter.
446 *
447 * Increase gain slowly.
448 *
449 * This is what preserves fades.
450 */
451 coefficient = smoothingCoefficient(config.release_seconds, samples);
452 }
453
454 current_gain_db += coefficient * (desired_gain_db - current_gain_db);
455
456 /*
457 * Avoid unnecessary floating point movement when we are
458 * already effectively at the target.
459 */
460 if (fabsf(current_gain_db - desired_gain_db) < 0.001f) {
461 current_gain_db = desired_gain_db;
462 }
463 }

479 template <typename T>
480 void apply(T* data, size_t samples, float gain) {
481#if PREFER_FIXEDPOINT
482 q1_14_t fixed_gain(gain);
483
484 for (size_t i = 0; i < samples; ++i) {
485 data[i] = fixed_gain.scale(data[i]);
486 }
487#else
488 constexpr float min_value = AGCPCMTraits<T>::min_value;
489
490 constexpr float max_value = AGCPCMTraits<T>::max_value;
491
492 const float dgain = static_cast<float>(gain);
493
494 for (size_t i = 0; i < samples; ++i) {
495 float value = static_cast<float>(data[i]) * dgain;
496
497 if (value > max_value) {
498 value = max_value;
499 }
500
501 if (value < min_value) {
502 value = min_value;
503 }
504
505 data[i] = static_cast<T>(value);
506 }
507#endif
508 }

535 void process(const uint8_t* input, size_t bytes) {
536 if (input == nullptr || bytes == 0 || channels == 0) {
537 return;
538 }
539
540 /*
541 * VolumeStream also modifies the supplied PCM buffer in-place,
542 * so AutomaticGainControlStream follows the same ModifyingStream
543 * convention.
544 */
545 uint8_t* buffer = const_cast<uint8_t*>(input);
546
547 float rms = 0.0;
548 float peak = 0.0;
549
550 /*
551 * First analyze the ORIGINAL signal.
552 *
553 * It is important that this happens before applying the new
554 * gain.
555 */
556 if (!analyze(buffer, bytes, rms, peak)) {
557 return;
558 }
559
560 /*
561 * Convert RMS and peak to dBFS.
562 */
563 float rms_db = linearToDb(static_cast<float>(rms));
564
565 float peak_db = linearToDb(static_cast<float>(peak));
566
567 measured_db = rms_db;
568 has_level = true;
569
570 size_t sample_size = bits_per_sample / 8;
571
572 if (sample_size == 0) {
573 return;
574 }
575
576 size_t samples = bytes / sample_size;
577
578 /*
579 * Determine desired gain.
580 */
581 float desired_gain_db = desiredGainDb(rms_db, peak_db);
582
583 /*
584 * Move current gain towards desired gain using asymmetric
585 * attack/release smoothing.
586 */
587 updateGain(desired_gain_db, samples);
588
589 /*
590 * Convert dB gain into a linear multiplier.
591 */
592 float gain = dbToLinear(current_gain_db);
593
594 /*
595 * Apply gain to the PCM data.
596 */
597 apply(buffer, bytes, gain);
598 }
