126 void writeFrame(tinympg::TinyMPGDecoder<Allocator> &decoder) {
127 if (p_out == nullptr) return;
128
129 size_t w = decoder.width();
130 size_t h = decoder.height();
131 size_t n = 0;
132
133 switch (pixel_format) {
134 case VideoFormat::RGB565: {
135 size_t needed = w * h;
136 if (frame_buffer.size() < needed * 2) frame_buffer.resize(needed * 2);
137 n = decoder.toRGB565((uint16_t *)frame_buffer.data(), needed);
138 if (n > 0) p_out->write(frame_buffer.data(), n * 2);
139 break;
140 }
141 case VideoFormat::RGB888: {
142 size_t needed = w * h * 3;
143 if (frame_buffer.size() < needed) frame_buffer.resize(needed);
144 n = decoder.toRGB888(frame_buffer.data(), needed);
145 if (n > 0) p_out->write(frame_buffer.data(), n);
146 break;
147 }
148 case VideoFormat::I420: {
149 size_t needed = w * h + 2 * (w / 2) * (h / 2);
150 if (frame_buffer.size() < needed) frame_buffer.resize(needed);
151 n = decoder.toYUV420(frame_buffer.data(), needed);
152 if (n > 0) p_out->write(frame_buffer.data(), n);
153 break;
154 }
155 default:
156 break;
157 }
158 }
