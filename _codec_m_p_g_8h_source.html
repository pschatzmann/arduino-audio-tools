105 void writeFrame(tinympg::TinyMPGDecoder<MPG_DEFAULT_ALLOCATOR> &decoder) {
106 if (p_out == nullptr) return;
107
108 size_t w = decoder.width();
109 size_t h = decoder.height();
110 size_t n = 0;
111
112 switch (pixel_format) {
113 case VideoFormat::RGB565: {
114 size_t needed = w * h;
115 if (frame_buffer.size() < needed * 2) {
116 frame_buffer.resize(needed * 2);
117 if (frame_buffer.data() == nullptr) {
118 LOGE("MPGDecoder: frame buffer allocation failed (RGB565)");
119 return;
120 }
121 }
122 n = decoder.toRGB565((uint16_t *)frame_buffer.data(), needed);
123 if (n > 0) p_out->write(frame_buffer.data(), n * 2);
124 break;
125 }
126 case VideoFormat::RGB888: {
127 size_t needed = w * h * 3;
128 if (frame_buffer.size() < needed) {
129 frame_buffer.resize(needed);
130 if (frame_buffer.data() == nullptr) {
131 LOGE("MPGDecoder: frame buffer allocation failed (RGB888)");
132 return;
133 }
134 }
135 n = decoder.toRGB888(frame_buffer.data(), needed);
136 if (n > 0) p_out->write(frame_buffer.data(), n);
137 break;
138 }
139 case VideoFormat::RGB666: {
140 size_t needed = w * h * 3;
141 if (frame_buffer.size() < needed) {
142 frame_buffer.resize(needed);

144 LOGE("MPGDecoder: frame buffer allocation failed (RGB666)");

146 }

148 n = decoder.toRGB666(frame_buffer.data(), needed);
149 if (n > 0) p_out->write(frame_buffer.data(), n);

153 size_t needed = w * h + 2 * (w / 2) * (h / 2);

313 template <typename Fn>
314 size_t writeOut(Fn encodeInto) {
315 if (p_out == nullptr) return 0;
316 size_t encoded = encodeInto(bitstream_buffer.data(), bitstream_buffer.size());
317 for (int attempt = 0; encoded == 0 && attempt < kMaxGrowRetries; ++attempt) {
318 bitstream_buffer.resize(bitstream_buffer.size() * 2);
319 if (bitstream_buffer.data() == nullptr) {
320 LOGE("MPGEncoder: bitstream buffer grow failed");
321 return 0;
322 }
323 encoded = encodeInto(bitstream_buffer.data(), bitstream_buffer.size());
324 }
325 if (encoded == 0) {
326 LOGE("MPGEncoder: encode failed (status=%d)", (int)encoder_.lastStatus());
327 return 0;
328 }
329 return p_out->write(bitstream_buffer.data(), encoded);
330 }

332 bool ensureBitstreamBuffer() {
333 if (width_ <= 0 || height_ <= 0) return false;
334 size_t needed = (size_t)width_ * (size_t)height_ * 2 + 4096;
335 if (bitstream_buffer.size() < needed) {
336 bitstream_buffer.resize(needed);
337 if (bitstream_buffer.data() == nullptr) {
338 return false;
339 }
340 }
341 return bitstream_buffer.data() != nullptr;
