104 void writeFrame(tinympg::TinyMPGDecoder<MPG_DEFAULT_ALLOCATOR> &decoder) {
105 if (p_out == nullptr) return;
106
107 size_t w = decoder.width();
108 size_t h = decoder.height();
109 size_t n = 0;
110
111 switch (pixel_format) {
112 case VideoFormat::RGB565: {
113 size_t needed = w * h;
114 if (frame_buffer.size() < needed * 2) {
115 frame_buffer.resize(needed * 2);
116 if (frame_buffer.data() == nullptr) {
117 LOGE("MPGDecoder: frame buffer allocation failed (RGB565)");
118 return;
119 }
120 }
121 n = decoder.toRGB565((uint16_t *)frame_buffer.data(), needed);
122 if (n > 0) p_out->write(frame_buffer.data(), n * 2);
123 break;
124 }
125 case VideoFormat::RGB888: {
126 size_t needed = w * h * 3;
127 if (frame_buffer.size() < needed) {
128 frame_buffer.resize(needed);
129 if (frame_buffer.data() == nullptr) {
130 LOGE("MPGDecoder: frame buffer allocation failed (RGB888)");
131 return;
132 }
133 }
134 n = decoder.toRGB888(frame_buffer.data(), needed);
135 if (n > 0) p_out->write(frame_buffer.data(), n);
136 break;
137 }
138 case VideoFormat::I420: {
139 size_t needed = w * h + 2 * (w / 2) * (h / 2);
140 if (frame_buffer.size() < needed) {
141 frame_buffer.resize(needed);
142 if (frame_buffer.data() == nullptr) {
143 LOGE("MPGDecoder: frame buffer allocation failed (I420)");
144 return;
145 }
146 }
147 n = decoder.toYUV420(frame_buffer.data(), needed);
148 if (n > 0) p_out->write(frame_buffer.data(), n);
149 break;
150 }
151 default:
152 break;
153 }
