155 void writeFrame(tinympg::TinyMPGDecoder<MPG_DEFAULT_ALLOCATOR> &decoder) {
156 if (p_out == nullptr && p_out_video == nullptr) return;
157
158 size_t w = decoder.width();
159 size_t h = decoder.height();
160 size_t n = 0;
161
162 switch (pixel_format) {
163 case VideoFormat::RGB565: {
164 size_t needed = w * h;
165 if (frame_buffer.size() < needed * 2) {
166 frame_buffer.resize(needed * 2);
167 if (frame_buffer.data() == nullptr) {
168 LOGE("MPGDecoder: frame buffer allocation failed (RGB565)");
169 return;
170 }
171 }
172 n = decoder.toRGB565((uint16_t *)frame_buffer.data(), needed);
173 if (n > 0) writeToOutput(frame_buffer.data(), n * 2);
174 break;
175 }
176 case VideoFormat::RGB888: {
177 size_t needed = w * h * 3;
178 if (frame_buffer.size() < needed) {
179 frame_buffer.resize(needed);
180 if (frame_buffer.data() == nullptr) {
181 LOGE("MPGDecoder: frame buffer allocation failed (RGB888)");
182 return;
183 }
184 }
185 n = decoder.toRGB888(frame_buffer.data(), needed);
186 if (n > 0) writeToOutput(frame_buffer.data(), n);
187 break;
188 }
189 case VideoFormat::RGB666: {
190 size_t needed = w * h * 3;
191 if (frame_buffer.size() < needed) {
192 frame_buffer.resize(needed);
193 if (frame_buffer.data() == nullptr) {
194 LOGE("MPGDecoder: frame buffer allocation failed (RGB666)");
195 return;
196 }
197 }
198 n = decoder.toRGB666(frame_buffer.data(), needed);
199 if (n > 0) writeToOutput(frame_buffer.data(), n);
200 break;
201 }
202 case VideoFormat::I420: {
203 size_t needed = w * h + 2 * (w / 2) * (h / 2);
204 if (frame_buffer.size() < needed) {
205 frame_buffer.resize(needed);
206 if (frame_buffer.data() == nullptr) {
207 LOGE("MPGDecoder: frame buffer allocation failed (I420)");
208 return;
209 }
210 }
211 n = decoder.toYUV420(frame_buffer.data(), needed);
212 if (n > 0) writeToOutput(frame_buffer.data(), n);
213 break;
214 }
215 default:
216 break;
217 }
218 }

376 template <typename Fn>
377 size_t writeOut(Fn encodeInto) {
378 if (p_out == nullptr && p_out_video == nullptr) return 0;
379 size_t encoded = encodeInto(bitstream_buffer.data(), bitstream_buffer.size());
380 for (int attempt = 0; encoded == 0 && attempt < kMaxGrowRetries; ++attempt) {
381 bitstream_buffer.resize(bitstream_buffer.size() * 2);
382 if (bitstream_buffer.data() == nullptr) {
383 LOGE("MPGEncoder: bitstream buffer grow failed");
384 return 0;
385 }
386 encoded = encodeInto(bitstream_buffer.data(), bitstream_buffer.size());
387 }
388 if (encoded == 0) {
389 LOGE("MPGEncoder: encode failed (status=%d)", (int)encoder_.lastStatus());
390 return 0;
391 }
392 if (p_out != nullptr) p_out->write(bitstream_buffer.data(), encoded);
393 if (p_out_video != nullptr) p_out_video->write(bitstream_buffer.data(), encoded);
394 return encoded;
395 }
396
397 bool ensureBitstreamBuffer() {
398 if (width_ <= 0 || height_ <= 0) return false;
399 size_t needed = (size_t)width_ * (size_t)height_ * 2 + 4096;
400 if (bitstream_buffer.size() < needed) {
401 bitstream_buffer.resize(needed);
402 if (bitstream_buffer.data() == nullptr) {
403 return false;
404 }
405 }
406 return bitstream_buffer.data() != nullptr;
