116 void writeFrame() {
117 if (p_out == nullptr) return;
118 size_t w = decoder_.width();
119 size_t h = decoder_.height();
120 size_t n = 0;
121 switch (pixel_format) {
122 case H264PixelFormat::RGB565: {
123 size_t needed = w * h;
124 if (frame_buffer.size() < needed * 2) frame_buffer.resize(needed * 2);
125 n = decoder_.toRGB565((uint16_t *)frame_buffer.data(), needed);
126 if (n > 0) p_out->write(frame_buffer.data(), n * 2);
127 break;
128 }
129 case H264PixelFormat::RGB666: {
130 size_t needed = w * h * 3;
131 if (frame_buffer.size() < needed) frame_buffer.resize(needed);
132 n = decoder_.toRGB666(frame_buffer.data(), needed);
133 if (n > 0) p_out->write(frame_buffer.data(), n);
134 break;
135 }
136 case H264PixelFormat::RGB888: {
137 size_t needed = w * h * 3;

142 }
