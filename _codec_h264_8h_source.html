145 if (p_out == nullptr) return;
146 size_t w = decoder_.width();
147 size_t h = decoder_.height();
148 size_t n = 0;
149 switch (pixel_format) {

152 if (frame_buffer.size() < needed * 2) frame_buffer.resize(needed * 2);
153 n = decoder_.toRGB565((uint16_t *)frame_buffer.data(), needed);
154 if (n > 0) p_out->write(frame_buffer.data(), n * 2);
155 break;
156 }
