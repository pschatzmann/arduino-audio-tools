76 bool isValid(const uint8_t *data, size_t len) override {
77 for (size_t i = 0; i + 3 < len; i++) {
78 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {

186 void writeToOutput() {
187 if (p_out == nullptr && p_out_video == nullptr) return;
188 // Every decoded picture is always shown - see PacedVideoOutput
189 // (Video/PacedVideoOutput.h) for how pacing/scheduling actually
190 // happens.
191 size_t w = decoder_.width();
192 size_t h = decoder_.height();
193 size_t n = 0;
194 switch (pixel_format) {
195 case VideoFormat::RGB565: {
196 size_t needed = w * h;
197 if (frame_buffer.size() < needed * 2) {
198 frame_buffer.resize(needed * 2);
199 if (frame_buffer.data() == nullptr) {
200 LOGE("H264Decoder: frame buffer allocation failed (RGB565)");
201 return;
202 }
203 }
204 n = decoder_.toRGB565((uint16_t *)frame_buffer.data(), needed);
205 if (n > 0) writeToOutput(frame_buffer.data(), n * 2);
206 break;
207 }
208 case VideoFormat::RGB666: {
209 size_t needed = w * h * 3;
210 if (frame_buffer.size() < needed) {
211 frame_buffer.resize(needed);
212 if (frame_buffer.data() == nullptr) {
213 LOGE("H264Decoder: frame buffer allocation failed (RGB666)");
214 return;
215 }
216 }
217 n = decoder_.toRGB666(frame_buffer.data(), needed);
218 if (n > 0) writeToOutput(frame_buffer.data(), n);
219 break;
220 }
221 case VideoFormat::RGB888: {
222 size_t needed = w * h * 3;
223 if (frame_buffer.size() < needed) {
224 frame_buffer.resize(needed);
225 if (frame_buffer.data() == nullptr) {
226 LOGE("H264Decoder: frame buffer allocation failed (RGB888)");
227 return;
228 }
229 }
230 n = decoder_.toRGB888(frame_buffer.data(), needed);
231 if (n > 0) writeToOutput(frame_buffer.data(), n);
232 break;
233 }
234 case VideoFormat::I420: {
235 size_t needed = w * h + 2 * (w / 2) * (h / 2);
236 if (frame_buffer.size() < needed) {
237 frame_buffer.resize(needed);
238 if (frame_buffer.data() == nullptr) {
239 LOGE("H264Decoder: frame buffer allocation failed (I420)");
240 return;
241 }
242 }
243 n = decoder_.toYUV420(frame_buffer.data(), needed);
244 if (n > 0) writeToOutput(frame_buffer.data(), n);
245 break;
246 }
247 default:
248 break;
249 }
250 }

405 size_t write(const uint8_t *data, size_t len) override {
406 switch (input_format) {
407 case VideoFormat::I420: {
408 size_t ySize = (size_t)width_ * height_;
409 size_t cSize = (size_t)(width_ / 2) * (height_ / 2);
410 if (len < ySize + 2 * cSize) return 0;
411 const uint8_t *srcY = data;
412 const uint8_t *srcU = data + ySize;
413 const uint8_t *srcV = data + ySize + cSize;
414 return writeOut([&](uint8_t *dst, size_t cap) {
415 return encoder_.encodeFrame(srcY, srcU, srcV, dst, cap);
416 });
417 }
418 case VideoFormat::YUV422:
419 return writeOut([&](uint8_t *dst, size_t cap) {
420 return encoder_.encodeFrameYuv422(data, dst, cap);
421 });
422 case VideoFormat::RGB565:
423 return writeOut([&](uint8_t *dst, size_t cap) {
424 return encoder_.encodeFrameRgb565((const uint16_t *)data, dst, cap);
425 });
426 case VideoFormat::RGB666:

456 template <typename Fn>
457 size_t writeOut(Fn encodeInto) {
458 if (p_out == nullptr && p_out_video == nullptr) return 0;
459 if (bitstream.size() == 0) {
460 bitstream.resize(kDefaultBitstreamBufferSize);
461 if (bitstream.data() == nullptr) {
462 LOGE("H264Encoder: bitstream buffer allocation failed");
463 return 0;
464 }
465 }
466 size_t n = encodeInto(bitstream.data(), bitstream.size());
467 for (int attempt = 0; n == 0 && attempt < kMaxGrowRetries; ++attempt) {
468 bitstream.resize(bitstream.size() * 2);
469 if (bitstream.data() == nullptr) {
470 LOGE("H264Encoder: bitstream buffer grow failed");
471 return 0;
472 }
473 n = encodeInto(bitstream.data(), bitstream.size());
474 }
475 if (p_out !=nullptr && n > 0) p_out->write(bitstream.data(), n);
476 if (p_out_video !=nullptr && n > 0) p_out_video->write(bitstream.data(), n);
477 return n;
478 }
