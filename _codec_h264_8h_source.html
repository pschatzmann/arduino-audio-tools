143 // Isolates pure CAVLC/reconstruction cost from everything the caller's
144 // own timing (e.g. DemuxerAVI's video_frame_start_ms) also bundles in
145 // - SD reads for this chunk, demux state-machine overhead - since that
146 // timer starts before this frame's bytes were even read, not here.
147 uint32_t decodeStart = millis();
148 decoder_.write(data, len);
149 total_decode_ms_ += millis() - decodeStart;

205 static bool isIdrFrame(const uint8_t *data, size_t len) {
206 size_t i = 0;
207 while (i + 3 <= len) {
208 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1) {
209 size_t nalStart = i + 3;
210 if (nalStart >= len) break;
211 uint8_t nalType = data[nalStart] & 0x1F;
212 if (nalType == 1) return false;
213 if (nalType == 5) return true;
214 i = nalStart;
215 } else {
216 i++;
217 }
218 }
219 return false;
220 }

241 void writeToOutput() {
242 if (p_out == nullptr && p_out_video == nullptr) return;
243 // No render-skip hint here: shouldAdmitFrame() (see write()) already
244 // decides whether a frame is worth decoding at all under sustained
245 // lateness - once we've reached its bounded steady-state lag, every
246 // *admitted* frame is exactly the amount of work the pipeline can
247 // afford, and isBehindSchedule() would stay true forever at that
248 // equilibrium (any nonzero lag still counts as "behind"), permanently
249 // skipping the render of the few frames that DO get decoded. So any
250 // frame that makes it this far is always shown.
251 size_t w = decoder_.width();
252 size_t h = decoder_.height();
253 size_t n = 0;
254 switch (pixel_format) {
255 case VideoFormat::RGB565: {
256 size_t needed = w * h;
257 if (frame_buffer.size() < needed * 2) {
258 frame_buffer.resize(needed * 2);
259 if (frame_buffer.data() == nullptr) {
260 LOGE("H264Decoder: frame buffer allocation failed (RGB565)");
261 return;
262 }
263 }
264 n = decoder_.toRGB565((uint16_t *)frame_buffer.data(), needed);
265 if (n > 0) writeToOutput(frame_buffer.data(), n * 2);
266 break;
267 }
268 case VideoFormat::RGB666: {
269 size_t needed = w * h * 3;
270 if (frame_buffer.size() < needed) {
271 frame_buffer.resize(needed);
272 if (frame_buffer.data() == nullptr) {
273 LOGE("H264Decoder: frame buffer allocation failed (RGB666)");
274 return;
275 }
276 }
277 n = decoder_.toRGB666(frame_buffer.data(), needed);
278 if (n > 0) writeToOutput(frame_buffer.data(), n);
279 break;
280 }
281 case VideoFormat::RGB888: {
282 size_t needed = w * h * 3;
283 if (frame_buffer.size() < needed) {
284 frame_buffer.resize(needed);
285 if (frame_buffer.data() == nullptr) {
286 LOGE("H264Decoder: frame buffer allocation failed (RGB888)");
287 return;
288 }
289 }
290 n = decoder_.toRGB888(frame_buffer.data(), needed);
291 if (n > 0) writeToOutput(frame_buffer.data(), n);
292 break;
293 }
294 case VideoFormat::I420: {
295 size_t needed = w * h + 2 * (w / 2) * (h / 2);
296 if (frame_buffer.size() < needed) {
297 frame_buffer.resize(needed);
298 if (frame_buffer.data() == nullptr) {
299 LOGE("H264Decoder: frame buffer allocation failed (I420)");
300 return;
301 }
302 }
303 n = decoder_.toYUV420(frame_buffer.data(), needed);
304 if (n > 0) writeToOutput(frame_buffer.data(), n);
305 break;
306 }
307 default:
308 break;
309 }
310 }

467 case VideoFormat::I420: {
468 size_t ySize = (size_t)width_ * height_;
469 size_t cSize = (size_t)(width_ / 2) * (height_ / 2);
470 if (len < ySize + 2 * cSize) return 0;
471 const uint8_t *srcY = data;

516 template <typename Fn>
517 size_t writeOut(Fn encodeInto) {
518 if (p_out == nullptr && p_out_video == nullptr) return 0;
519 if (bitstream.size() == 0) {
520 bitstream.resize(kDefaultBitstreamBufferSize);
521 if (bitstream.data() == nullptr) {
522 LOGE("H264Encoder: bitstream buffer allocation failed");
523 return 0;
524 }
525 }
526 size_t n = encodeInto(bitstream.data(), bitstream.size());
527 for (int attempt = 0; n == 0 && attempt < kMaxGrowRetries; ++attempt) {
528 bitstream.resize(bitstream.size() * 2);
529 if (bitstream.data() == nullptr) {
530 LOGE("H264Encoder: bitstream buffer grow failed");
531 return 0;
532 }
533 n = encodeInto(bitstream.data(), bitstream.size());
534 }
535 if (p_out !=nullptr && n > 0) p_out->write(bitstream.data(), n);
536 if (p_out_video !=nullptr && n > 0) p_out_video->write(bitstream.data(), n);
