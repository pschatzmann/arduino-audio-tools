157 void writeToOutput() {
158 if (p_out == nullptr && p_out_video == nullptr) return;
159 // Every decoded picture is always shown - see VideoAudioSyncTask
160 // (Video/VideoAudioSyncTask.h) for how pacing/scheduling actually
161 // happens.
162 size_t w = decoder_.width();
163 size_t h = decoder_.height();
164 size_t n = 0;
165 switch (pixel_format) {
166 case VideoFormat::RGB565: {
167 size_t needed = w * h;
168 if (frame_buffer.size() < needed * 2) {
169 frame_buffer.resize(needed * 2);
170 if (frame_buffer.data() == nullptr) {
171 LOGE("H264Decoder: frame buffer allocation failed (RGB565)");
172 return;
173 }
174 }
175 n = decoder_.toRGB565((uint16_t *)frame_buffer.data(), needed);
176 if (n > 0) writeToOutput(frame_buffer.data(), n * 2);
177 break;
178 }
179 case VideoFormat::RGB666: {
180 size_t needed = w * h * 3;
181 if (frame_buffer.size() < needed) {
182 frame_buffer.resize(needed);
183 if (frame_buffer.data() == nullptr) {
184 LOGE("H264Decoder: frame buffer allocation failed (RGB666)");
185 return;
186 }
187 }
188 n = decoder_.toRGB666(frame_buffer.data(), needed);
189 if (n > 0) writeToOutput(frame_buffer.data(), n);
190 break;
191 }
192 case VideoFormat::RGB888: {
193 size_t needed = w * h * 3;
194 if (frame_buffer.size() < needed) {
195 frame_buffer.resize(needed);
196 if (frame_buffer.data() == nullptr) {
197 LOGE("H264Decoder: frame buffer allocation failed (RGB888)");
198 return;
199 }
200 }
201 n = decoder_.toRGB888(frame_buffer.data(), needed);
202 if (n > 0) writeToOutput(frame_buffer.data(), n);
203 break;
204 }
205 case VideoFormat::I420: {
206 size_t needed = w * h + 2 * (w / 2) * (h / 2);
207 if (frame_buffer.size() < needed) {
208 frame_buffer.resize(needed);
209 if (frame_buffer.data() == nullptr) {
210 LOGE("H264Decoder: frame buffer allocation failed (I420)");
211 return;
212 }
213 }
214 n = decoder_.toYUV420(frame_buffer.data(), needed);
215 if (n > 0) writeToOutput(frame_buffer.data(), n);
216 break;
217 }
218 default:
219 break;
220 }
221 }

377 switch (input_format) {
378 case VideoFormat::I420: {
379 size_t ySize = (size_t)width_ * height_;
380 size_t cSize = (size_t)(width_ / 2) * (height_ / 2);
381 if (len < ySize + 2 * cSize) return 0;
382 const uint8_t *srcY = data;
383 const uint8_t *srcU = data + ySize;
384 const uint8_t *srcV = data + ySize + cSize;
385 return writeOut([&](uint8_t *dst, size_t cap) {
386 return encoder_.encodeFrame(srcY, srcU, srcV, dst, cap);
387 });
388 }
389 case VideoFormat::YUV422:

391 return encoder_.encodeFrameYuv422(data, dst, cap);
392 });
393 case VideoFormat::RGB565:

395 return encoder_.encodeFrameRgb565((const uint16_t *)data, dst, cap);

427 template <typename Fn>
428 size_t writeOut(Fn encodeInto) {
429 if (p_out == nullptr && p_out_video == nullptr) return 0;
430 if (bitstream.size() == 0) {
431 bitstream.resize(kDefaultBitstreamBufferSize);
432 if (bitstream.data() == nullptr) {
433 LOGE("H264Encoder: bitstream buffer allocation failed");
434 return 0;
435 }
436 }
437 size_t n = encodeInto(bitstream.data(), bitstream.size());
438 for (int attempt = 0; n == 0 && attempt < kMaxGrowRetries; ++attempt) {
439 bitstream.resize(bitstream.size() * 2);
440 if (bitstream.data() == nullptr) {
441 LOGE("H264Encoder: bitstream buffer grow failed");
442 return 0;
443 }
444 n = encodeInto(bitstream.data(), bitstream.size());
445 }
446 if (p_out !=nullptr && n > 0) p_out->write(bitstream.data(), n);
447 if (p_out_video !=nullptr && n > 0) p_out_video->write(bitstream.data(), n);
448 return n;
449 }
