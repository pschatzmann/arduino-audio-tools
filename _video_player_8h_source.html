430 size_t copy() {
431 if (!active || p_source == nullptr) return 0;
432 float fps = default_demuxer.getVideoInfo().fps;
433 if (fps > 0) video_sync.setFps(fps);
434 uint8_t buffer[buffer_size];
435 size_t len = p_source->readBytes(buffer, buffer_size);
436 if (len == 0) return 0;
437 // Retry on a partial accept instead of dropping the remainder, which
438 // would desync the demuxer from the real byte stream (duplicated here
439 // rather than reusing CodecCopy::writeAll() since CodecCopy binds its
440 // Stream& at construction time, before begin(Stream&)'s source is
441 // known).
442 size_t written = 0;
443 while (written < len) {
444 size_t n = default_demuxer.write(buffer + written, len - written);
445 if (n == 0) break;
446 written += n;
447 }
448 return written;
449 }
