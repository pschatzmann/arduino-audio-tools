373 size_t copy() {
374 if (!active || p_source == nullptr) return 0;
375 float fps = default_demuxer.getVideoInfo().fps;
376 if (fps > 0) video_sync.setFps(fps);
377 uint8_t buffer[buffer_size];
378 size_t len = p_source->readBytes(buffer, buffer_size);
379 if (len == 0) return 0;
380 // Retry on a partial accept instead of dropping the remainder, which
381 // would desync the demuxer from the real byte stream (see
382 // CodecCopy::writeAll()'s own comment - duplicated here rather than
383 // reused since CodecCopy binds its Stream& at construction time,
384 // before begin(Stream&)'s source is known).
385 size_t written = 0;
386 while (written < len) {
387 size_t n = default_demuxer.write(buffer + written, len - written);
388 if (n == 0) break;
389 written += n;
390 }
391 return written;
392 }
