323 size_t copy() {
324 if (!active || p_source == nullptr || p_demuxer == nullptr) return 0;
325 float fps = p_demuxer->getVideoInfo().fps;
326 if (fps > 0) video_sync.setFps(fps);
327 uint8_t buffer[buffer_size];
328 size_t len = p_source->readBytes(buffer, buffer_size);
329 if (len == 0) return 0;
330 // Retry on a partial accept instead of dropping the remainder, which
331 // would desync the demuxer from the real byte stream (see
332 // CodecCopy::writeAll()'s own comment - duplicated here rather than
333 // reused since CodecCopy binds its Stream& at construction time,
334 // before begin(Stream&)'s source is known).
335 size_t written = 0;
336 while (written < len) {
337 size_t n = p_demuxer->write(buffer + written, len - written);
338 if (n == 0) break;
339 written += n;
340 }
341 return written;
342 }
