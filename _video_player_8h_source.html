313 size_t copy() {
314 if (!active || p_source == nullptr || p_demuxer == nullptr) return 0;
315 float fps = p_demuxer->getVideoInfo().fps;
316 if (fps > 0) video_sync.setFps(fps);
317 uint8_t buffer[buffer_size];
318 size_t len = p_source->readBytes(buffer, buffer_size);
319 if (len == 0) return 0;
320 // Retry on a partial accept instead of dropping the remainder, which
321 // would desync the demuxer from the real byte stream (see
322 // CodecCopy::writeAll()'s own comment - duplicated here rather than
323 // reused since CodecCopy binds its Stream& at construction time,
324 // before begin(Stream&)'s source is known).
325 size_t written = 0;
326 while (written < len) {
327 size_t n = p_demuxer->write(buffer + written, len - written);
328 if (n == 0) break;
329 written += n;
330 }
331 return written;
332 }
