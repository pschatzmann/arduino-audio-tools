318 size_t copy() {
319 if (!active || p_source == nullptr || p_demuxer == nullptr) return 0;
320 float fps = p_demuxer->getVideoInfo().fps;
321 if (fps > 0) video_sync.setFps(fps);
322 uint8_t buffer[buffer_size];
323 size_t len = p_source->readBytes(buffer, buffer_size);
324 if (len == 0) return 0;
325 // Retry on a partial accept instead of dropping the remainder, which
326 // would desync the demuxer from the real byte stream (see
327 // CodecCopy::writeAll()'s own comment - duplicated here rather than
328 // reused since CodecCopy binds its Stream& at construction time,
329 // before begin(Stream&)'s source is known).
330 size_t written = 0;
331 while (written < len) {
332 size_t n = p_demuxer->write(buffer + written, len - written);
333 if (n == 0) break;
334 written += n;
335 }
336 return written;
337 }
