368 size_t copy() {
369 if (!active || p_source == nullptr) return 0;
370 float fps = default_demuxer.getVideoInfo().fps;
371 if (fps > 0) video_sync.setFps(fps);
372 uint8_t buffer[buffer_size];
373 size_t len = p_source->readBytes(buffer, buffer_size);
374 if (len == 0) return 0;
375 // Retry on a partial accept instead of dropping the remainder, which
376 // would desync the demuxer from the real byte stream (see
377 // CodecCopy::writeAll()'s own comment - duplicated here rather than
378 // reused since CodecCopy binds its Stream& at construction time,
379 // before begin(Stream&)'s source is known).
380 size_t written = 0;
381 while (written < len) {
382 size_t n = default_demuxer.write(buffer + written, len - written);
383 if (n == 0) break;
384 written += n;
385 }
386 return written;
387 }

390 size_t copyAll() {
391 size_t result = 0;
392 size_t step = copy();
393 while (step > 0) {
394 result += step;
395 step = copy();
396 }
397 return result;
398 }
