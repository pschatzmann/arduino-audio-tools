357 size_t copy() {
358 if (!active || p_source == nullptr) return 0;
359 float fps = default_demuxer.getVideoInfo().fps;
360 if (fps > 0) video_sync.setFps(fps);
361 uint8_t buffer[buffer_size];
362 size_t len = p_source->readBytes(buffer, buffer_size);
363 if (len == 0) return 0;
364 // Retry on a partial accept instead of dropping the remainder, which
365 // would desync the demuxer from the real byte stream (see
366 // CodecCopy::writeAll()'s own comment - duplicated here rather than
367 // reused since CodecCopy binds its Stream& at construction time,
368 // before begin(Stream&)'s source is known).
369 size_t written = 0;
370 while (written < len) {
371 size_t n = default_demuxer.write(buffer + written, len - written);
372 if (n == 0) break;
373 written += n;
374 }
375 return written;
376 }
