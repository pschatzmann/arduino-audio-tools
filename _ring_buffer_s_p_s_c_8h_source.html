62 bool peek(T& result) override {
63 size_t h = head_.load(std::memory_order_acquire);
64 size_t t = tail_.load(std::memory_order_relaxed);
65 if (h == t) return false;

70 // Bulk write — called only by the producer.
71 int writeArray(const T data[], int len) override {
72 if (capacity_ == 0 || len <= 0) return 0;
73 size_t t = tail_.load(std::memory_order_acquire);
74 size_t h = head_.load(std::memory_order_relaxed);
75 size_t space = capacity_ - (h - t);
76 size_t n = (size_t)len < space ? (size_t)len : space;
77 if (n == 0) return 0;
78
79 T* buf = buf_.data();
80 size_t h_idx = h & mask_;
81 size_t first = capacity_ - h_idx; // bytes until end of physical buffer
82 if (first > n) first = n;
83 memcpy(buf + h_idx, data, first * sizeof(T));
84 memcpy(buf, data + first, (n - first) * sizeof(T));
85
86 head_.store(h + n, std::memory_order_release);
87 return (int)n;
88 }
89
90 // Bulk read — called only by the consumer.
91 int readArray(T data[], int len) override {
92 if (capacity_ == 0 || len <= 0) return 0;
93 size_t h = head_.load(std::memory_order_acquire);
94 size_t t = tail_.load(std::memory_order_relaxed);
95 size_t avail = h - t;
96 size_t n = (size_t)len < avail ? (size_t)len : avail;
97 if (n == 0) return 0;
98
99 T* buf = buf_.data();
100 size_t t_idx = t & mask_;
101 size_t first = capacity_ - t_idx; // bytes until end of physical buffer
102 if (first > n) first = n;
103 memcpy(data, buf + t_idx, first * sizeof(T));
104 memcpy(data + first, buf, (n - first) * sizeof(T));
105
106 tail_.store(t + n, std::memory_order_release);
107 return (int)n;
108 }

137 bool resize(size_t capacity) override {
138 // Round up to the next power of two so masking replaces modulo.
139 size_t pow2 = 1;
140 while (pow2 < capacity) pow2 <<= 1;
141 buf_.setAllocator(use_psram_ ? DefaultAllocator : DefaultAllocatorRAM);
142 buf_.resize(capacity > 0 ? pow2 : 0);
143 capacity_ = (capacity > 0 && buf_.data() != nullptr) ? pow2 : 0;
