88 bool isI420 = info_.format == VideoFormat::I420;
89 // I420: one byte/pixel Y plane, plus two quarter-size (w/2 x h/2)
90 // U/V planes - the standard w*h*3/2 packed-planar size.
91 size_t needed = isI420 ? (size_t)info_.width * info_.height * 3 / 2
92 : (size_t)info_.width * info_.height * sizeof(RGB565);
93 if (len < needed) {
94 LOGE("OutputTinyGPU: frame too small: %d < %d bytes", (int)len,
95 (int)needed);
96 return 0;
97 }
98
99 if (skip_render) {
100 return len;
101 }
102
103 if (isI420) {
104 // Unlike the RGB565 path below, this always goes row-by-row (even
105 // when scale_to_fit is off) - I420's planar layout isn't a raw byte
106 // pattern the driver can view directly the way packed RGB565 is, so
107 // conversion is unavoidable regardless of whether scaling also
108 // happens in the same pass.
109 writeI420(data, info_.width, info_.height);
110 return len;
111 }
112
113 if (scale_to_fit && (info_.width != width || info_.height != height)) {
114 writeScaled(data, info_.width, info_.height);
115 return len;
116 }
117
118 // View directly over the caller's buffer - no allocation, no copy.
119 SurfaceWithExternalBuffer<RGB565> frame;
120 frame.setExternalBuffer(const_cast<uint8_t*>(data), len);
121 frame.resize(info_.width, info_.height);
122 tftDriver.writeData(frame, 0, 0);
123
124 return len;

134 if ((int)scale_line_.size() != width) scale_line_.resize(width);

160 void writeI420(const uint8_t* data, int srcW, int srcH) {
161 const uint8_t* yPlane = data;
162 const uint8_t* uPlane = yPlane + (size_t)srcW * srcH;
163 const uint8_t* vPlane = uPlane + (size_t)(srcW / 2) * (srcH / 2);
164 size_t srcPixels = (size_t)srcW * srcH;
165 if (i420_rgb_buf_.size() != srcPixels) i420_rgb_buf_.resize(srcPixels);
166 for (int y = 0; y < srcH; y++) {
167 const uint8_t* yRow = yPlane + (size_t)y * srcW;
168 const uint8_t* uRow = uPlane + (size_t)(y / 2) * (srcW / 2);
169 const uint8_t* vRow = vPlane + (size_t)(y / 2) * (srcW / 2);
170 RGB565* dstRow = i420_rgb_buf_.data() + (size_t)y * srcW;
171 for (int x = 0; x < srcW; x++) {
172 dstRow[x] = yuvToRgb565(yRow[x], uRow[x / 2], vRow[x / 2]);
173 }
174 }
175
176 if (scale_to_fit && (srcW != width || srcH != height)) {
177 writeScaled((const uint8_t*)i420_rgb_buf_.data(), srcW, srcH);
178 return;
179 }
180 SurfaceWithExternalBuffer<RGB565> frame;
181 frame.setExternalBuffer((uint8_t*)i420_rgb_buf_.data(),
182 srcPixels * sizeof(RGB565));

190 static RGB565 yuvToRgb565(uint8_t Y, uint8_t U, uint8_t V) {
191 int c = (int)Y - 16;
192 int d = (int)U - 128;
193 int e = (int)V - 128;
194 int r = (298 * c + 409 * e + 128) >> 8;
195 int g = (298 * c - 100 * d - 208 * e + 128) >> 8;
196 int b = (298 * c + 516 * d + 128) >> 8;
197 r = r < 0 ? 0 : (r > 255 ? 255 : r);
198 g = g < 0 ? 0 : (g > 255 ? 255 : g);
199 b = b < 0 ? 0 : (b > 255 ? 255 : b);
200 return RGB565((uint8_t)r, (uint8_t)g, (uint8_t)b);
201 }
