107 size_t write(const uint8_t* data, size_t len) override {
108 auto start = millis();
109 if (p_info != nullptr) {
110 info_ = p_info->videoInfo();
111 }
112
113 if (info_.width == 0 || info_.height == 0) {
114 LOGE("OutputTinyGPU: invalid video size: %d x %d", (int)info_.width,
115 (int)info_.height);
116 return 0;
117 }
118
119 bool isI420 = info_.format == VideoFormat::I420;
120 // I420: one byte/pixel Y plane, plus two quarter-size (w/2 x h/2)
121 // U/V planes - the standard w*h*3/2 packed-planar size.
122 size_t needed = isI420 ? (size_t)info_.width * info_.height * 3 / 2
123 : (size_t)info_.width * info_.height * sizeof(RGB565);
124 if (len < needed) {
125 LOGE("OutputTinyGPU: frame too small: %d < %d bytes", (int)len,
126 (int)needed);
127 return 0;
128 }
129
130 if (skip_render) {
131 return len;
132 }
133
134 if (isI420) {
135 // Unlike the RGB565 path below, this always goes row-by-row (even
136 // when scale_to_fit is off) - I420's planar layout isn't a raw byte
137 // pattern the driver can view directly the way packed RGB565 is, so
138 // conversion is unavoidable regardless of whether scaling also
139 // happens in the same pass.
140 writeI420(data, info_.width, info_.height);
141 write_time_ms = millis() - start;
142 return len;
143 }
144
145 if (scale_to_fit && (info_.width != (int)tftDriver.width() ||
146 info_.height != (int)tftDriver.height())) {
147 writeScaled(data, info_.width, info_.height);
148 write_time_ms = millis() - start;
149 return len;
150 }
151
152 // View directly over the caller's buffer - no allocation, no copy.
153 SurfaceWithExternalBuffer<RGB565> frame;
154 frame.setExternalBuffer(const_cast<uint8_t*>(data), len);
155 frame.resize(info_.width, info_.height);
156 tftDriver.writeData(frame, 0, 0);
157 write_time_ms = millis() - start;
158
159 return len;
160 }

168 void writeScaled(const uint8_t* data, int srcW, int srcH) {
169 int width = (int)tftDriver.width();
170 int height = (int)tftDriver.height();
171 if ((int)scale_line_.size() != width) scale_line_.resize(width);
172 const RGB565* src = (const RGB565*)data;
173 for (int y = 0; y < height; y++) {
174 int srcY = (int)((int64_t)y * srcH / height);
175 const RGB565* srcRow = src + (size_t)srcY * srcW;
176 for (int x = 0; x < width; x++) {
177 int srcX = (int)((int64_t)x * srcW / width);
178 scale_line_[x] = srcRow[srcX];
179 }
180 SurfaceWithExternalBuffer<RGB565> row;
181 row.setExternalBuffer((uint8_t*)scale_line_.data(),
182 (size_t)width * sizeof(RGB565));
183 row.resize(width, 1);
184 tftDriver.writeData(row, 0, y);
185 }

197 void writeI420(const uint8_t* data, int srcW, int srcH) {
198 const uint8_t* yPlane = data;
199 const uint8_t* uPlane = yPlane + (size_t)srcW * srcH;
200 const uint8_t* vPlane = uPlane + (size_t)(srcW / 2) * (srcH / 2);
201 size_t srcPixels = (size_t)srcW * srcH;
202 if (i420_rgb_buf_.size() != srcPixels) i420_rgb_buf_.resize(srcPixels);
203 for (int y = 0; y < srcH; y++) {
204 const uint8_t* yRow = yPlane + (size_t)y * srcW;
205 const uint8_t* uRow = uPlane + (size_t)(y / 2) * (srcW / 2);
206 const uint8_t* vRow = vPlane + (size_t)(y / 2) * (srcW / 2);
207 RGB565* dstRow = i420_rgb_buf_.data() + (size_t)y * srcW;
208 for (int x = 0; x < srcW; x++) {
209 dstRow[x] = yuvToRgb565(yRow[x], uRow[x / 2], vRow[x / 2]);
210 }
211 }
212
213 if (scale_to_fit && (srcW != (int)tftDriver.width() ||
214 srcH != (int)tftDriver.height())) {
215 writeScaled((const uint8_t*)i420_rgb_buf_.data(), srcW, srcH);
216 return;
217 }
218 SurfaceWithExternalBuffer<RGB565> frame;
219 frame.setExternalBuffer((uint8_t*)i420_rgb_buf_.data(),
220 srcPixels * sizeof(RGB565));
221 frame.resize(srcW, srcH);
222 tftDriver.writeData(frame, 0, 0);
223 }

228 static RGB565 yuvToRgb565(uint8_t Y, uint8_t U, uint8_t V) {
229 int c = (int)Y - 16;
230 int d = (int)U - 128;
231 int e = (int)V - 128;
232 int r = (298 * c + 409 * e + 128) >> 8;
233 int g = (298 * c - 100 * d - 208 * e + 128) >> 8;
234 int b = (298 * c + 516 * d + 128) >> 8;
235 r = r < 0 ? 0 : (r > 255 ? 255 : r);
236 g = g < 0 ? 0 : (g > 255 ? 255 : g);
237 b = b < 0 ? 0 : (b > 255 ? 255 : b);
238 return RGB565((uint8_t)r, (uint8_t)g, (uint8_t)b);
239 }
