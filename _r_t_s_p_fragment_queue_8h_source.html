36class RTSPFragmentQueue {
37 protected:
38 // RTSPMediaStreamer infers the RTP marker bit from "nothing left queued
39 // after this fragment", which is only exact when at most one frame's
40 // fragments are queued at a time - so keep the backlog capped at 1 frame
41 // (older, stale frames are dropped rather than queued behind a newer one,
42 // which also keeps real-time video latency low).
43 size_t m_maxQueuedFrames = 1;
44
45 Vector<uint8_t> m_queue;
46 size_t m_queueHead = 0;
47 size_t m_queuedFrames = 0;
48
49 void clearQueue() {
50 m_queue.clear();
51 m_queueHead = 0;
52 m_queuedFrames = 0;
53 }
54
55 uint32_t readQueueU32(size_t pos) {
56 return ((uint32_t)m_queue[pos] << 24) | ((uint32_t)m_queue[pos + 1] << 16) |
57 ((uint32_t)m_queue[pos + 2] << 8) | (uint32_t)m_queue[pos + 3];
58 }
59
60 void appendFragment(const uint8_t *payload, size_t len, bool last) {
61 size_t pos = m_queue.size();
62 m_queue.resize(pos + 5 + len);
63 uint8_t *p = m_queue.data() + pos;
64 p[0] = (uint8_t)((len >> 24) & 0xFF);
65 p[1] = (uint8_t)((len >> 16) & 0xFF);
66 p[2] = (uint8_t)((len >> 8) & 0xFF);
67 p[3] = (uint8_t)(len & 0xFF);
68 p[4] = last ? 1 : 0;
69 memcpy(p + 5, payload, len);
70
71 if (last) {
72 ++m_queuedFrames;
73 if (m_queuedFrames > m_maxQueuedFrames) {
74 dropOldestFrame();
75 }
76 }
77 }
78
79 void dropOldestFrame() {
80 size_t pos = m_queueHead;
81 while (pos < m_queue.size()) {
82 uint32_t len = readQueueU32(pos);
83 bool last = m_queue[pos + 4] != 0;
84 pos += 5 + len;
85 if (last) break;
86 }
87 m_queueHead = pos;
88 if (m_queuedFrames > 0) --m_queuedFrames;
89 LOGW("RTSPFragmentQueue: consumer too slow, dropped oldest queued video frame");
90 }
91
92 int queuePacketSize() {
93 if (m_queueHead >= m_queue.size()) {
94 if (m_queue.size() > 0) clearQueue();
95 return 0;
96 }
97 return (int)readQueueU32(m_queueHead);
98 }
99
102 int queueReadBytes(void *dest, int maxBytes) {
103 if (m_queueHead >= m_queue.size()) return 0;
104
105 uint32_t len = readQueueU32(m_queueHead);
106 bool last = m_queue[m_queueHead + 4] != 0;
107 size_t payloadStart = m_queueHead + 5;
108 if (payloadStart + len > m_queue.size()) {
109 LOGE("RTSPFragmentQueue: corrupt fragment queue; resetting");
110 clearQueue();
111 return 0;
112 }
113
114 m_queueHead = payloadStart + len;
115 if (last && m_queuedFrames > 0) --m_queuedFrames;
116
117 if ((int)len > maxBytes) {
118 LOGE(
119 "RTSPFragmentQueue: fragment (%u bytes) exceeds destination "
120 "buffer (%d bytes); dropping",
121 (unsigned)len, maxBytes);
122 return 0;
123 }
124 memcpy(dest, m_queue.data() + payloadStart, len);
125 return (int)len;
126 }
127};
