304 void writeFMT(BaseBuffer<uint8_t> &buffer, const WAVAudioInfo &info) {
305 bool is_ms_adpcm = info.ext_adpcm_header && info.format == AudioFormat::ADPCM;
306 bool is_ima_adpcm = info.ext_adpcm_header && info.format == AudioFormat::DVI_ADPCM;
307 uint16_t fmt_len = 16;
308 if (is_ima_adpcm) fmt_len = 20;
309 else if (is_ms_adpcm) fmt_len = 50;
310
311 uint16_t spb = samplesPerBlock(info);
312 uint32_t byte_rate = info.byte_rate;
313 if ((is_ms_adpcm || is_ima_adpcm) && spb > 0 && info.block_align > 0) {
314 // average bytes/sec = (sample_rate * block_align) / samples_per_block
315 byte_rate = ((uint64_t)info.sample_rate * info.block_align) / spb;
316 }
317
318 buffer.writeArray((uint8_t *)"fmt ", 4);
319 write32(buffer, fmt_len);
320 write16(buffer, (uint16_t)info.format);
321 write16(buffer, info.channels);
322 write32(buffer, info.sample_rate);
323 write32(buffer, byte_rate);
324 write16(buffer, info.block_align); // frame size
325 write16(buffer, info.bits_per_sample);
326
327 if (is_ima_adpcm) {
328 write16(buffer, 2); // cbSize: size of extra format bytes
329 write16(buffer, spb); // wSamplesPerBlock
330 } else if (is_ms_adpcm) {
331 // standard MS ADPCM coefficient table (7 predictor pairs)
332 static const int16_t ms_adpcm_coef[7][2] = {
333 {256, 0}, {512, -256}, {0, 0}, {192, 64},
334 {240, 0}, {460, -208}, {392, -232}};
335 write16(buffer, 32); // cbSize: size of extra format bytes
336 write16(buffer, spb); // wSamplesPerBlock
337 write16(buffer, 7); // wNumCoef
338 for (auto &c : ms_adpcm_coef) {
339 write16(buffer, (uint16_t)c[0]);
340 write16(buffer, (uint16_t)c[1]);
341 }
342 }
343 }

347 void writeFactChunk(BaseBuffer<uint8_t> &buffer, const WAVAudioInfo &info) {
348 buffer.writeArray((uint8_t *)"fact", 4);
349 write32(buffer, 4); // chunk size
350 uint32_t sample_length = 0;
351 uint16_t spb = samplesPerBlock(info);
352 if (!info.is_streamed && spb > 0 && info.block_align > 0) {
353 sample_length = (info.data_length / info.block_align) * spb;
354 }
355 write32(buffer, sample_length);
356 }

358 void writeDataHeader(BaseBuffer<uint8_t> &buffer, const WAVAudioInfo &info) {
359 buffer.writeArray((uint8_t *)"data", 4);
360 uint32_t data_length = info.data_length;
361 uint32_t header_bytes = 36 + extraHeaderBytes(info);
362 if (headerInfo.is_streamed && data_length == 0) {
363 data_length = ~0; // use max value for streamed data if not set
364 }
365 if (!headerInfo.is_streamed && info.file_size >= header_bytes && (data_length == 0 || data_length == ~0)) {
366 data_length = info.file_size - header_bytes; // data length = file size - header size
367 }
368 LOGI("writeDataHeader: data_length=%u", data_length);
369 write32(buffer, data_length);
370 int offset = info.offset;
371 if (offset > 0) {
372 uint8_t empty[offset];
373 memset(empty, 0, offset);
374 buffer.writeArray(empty, offset); // resolve issue with wrong aligment
375 }
376 }

384 static uint16_t samplesPerBlock(const WAVAudioInfo &info) {
385 if (info.channels <= 0 || info.block_align <= 0) return 0;
386 switch (info.format) {
387 case AudioFormat::ADPCM: // MS ADPCM
388 return ((info.block_align / info.channels) - 7) * 2 + 2;
389 case AudioFormat::DVI_ADPCM: // IMA/DVI ADPCM
390 return ((info.block_align / info.channels) - 4) * 2 + 1;
391 default:
392 return 0;
393 }
394 }

569 virtual size_t write(const uint8_t *data, size_t len) override {
570 TRACED();
571 size_t result = 0;
572 if (active) {
573 if (isFirst) {
574 int data_start = decodeHeader((uint8_t *)data, len);
575 // we do not have the complete header yet: need more data
576 if (data_start == 0) return len;
577 // process the outstanding data - only if the format is supported;
578 // otherwise report 0 bytes written, consistent with subsequent
579 // write() calls once the format has been found to be invalid
580 if (isValid) {
581 result = data_start +
582 write_out((uint8_t *)data + data_start, len - data_start);
583 }
584
585 } else if (isValid) {
586 result = write_out((uint8_t *)data, len);
587 }
588 }
589 return result;
590 }

676 size_t write_out_8to16(const uint8_t *in_ptr, size_t in_size) {
677 size_t samples_remaining = in_size;
678 size_t offset = 0;
679 int16_t out_buf[batch_size];
680 while (samples_remaining > 0) {
681 size_t current_batch =
682 samples_remaining > batch_size ? batch_size : samples_remaining;
683 for (size_t i = 0; i < current_batch; ++i) {
684 out_buf[i] = ((int16_t)in_ptr[offset + i] - 128) << 8;
685 }
686 writeDataT<int16_t>(&out(), out_buf, current_batch);
687 offset += current_batch;
688 samples_remaining -= current_batch;
689 }
690 return in_size;

783 static int16_t alaw2linear(uint8_t a_val) {
784 a_val ^= 0x55;
785 int t = (a_val & 0x0f) << 4;
786 int seg = ((unsigned)a_val & 0x70) >> 4;
787 switch (seg) {
788 case 0:
789 t += 8;
790 break;
791 case 1:
792 t += 0x108;
793 break;
794 default:
795 t += 0x108;
796 t <<= (seg - 1);
797 break;
798 }
799 return (int16_t)((a_val & 0x80) ? t : -t);
800 }

803 static int16_t ulaw2linear(uint8_t u_val) {
804 u_val = ~u_val;
805 int t = ((u_val & 0x0f) << 3) + 0x84;
806 t <<= ((unsigned)u_val & 0x70) >> 4;
807 return (int16_t)((u_val & 0x80) ? (0x84 - t) : (t - 0x84));
808 }

811 int decodeHeader(uint8_t *in_ptr, size_t in_size) {
812 // we expect at least the full header
813 header.write(in_ptr, in_size);
814 if (!header.isDataComplete()) {
815 if (header.isOverflow()) {
816 // the header (up to and including the 'data' chunk tag) exceeds
817 // MAX_WAV_HEADER_LEN_LIMIT without ever completing - give up
818 // instead of waiting forever for data that will never arrive
819 LOGE(
820 "WAV header exceeds the maximum supported size of %d bytes "
821 "without a 'data' chunk - aborting",
822 (int)MAX_WAV_HEADER_LEN_LIMIT);
823 isFirst = false;
824 isValid = false;
825 return 0;
826 }
827 LOGW("WAV header misses 'data' section in len: %d",
828 (int)header.available());
829 header.dumpHeader();
830 return 0;
831 }
832 // parse header
833 if (!header.parse()) {
834 LOGE("WAV header parsing failed");
835 return 0;
836 }
837
838 isFirst = false;
839
840 LOGI("WAV sample_rate: %d", (int)header.audioInfo().sample_rate);

845 // was registered via setDecoder()/addDecoder()); otherwise fall back to

853 int block_size = header.audioInfo().block_align;
854 p_decoder->setBlockSize(block_size);
855 setupEncodedAudio();
856 }

858 // update sampling rate if the target supports it
859 AudioInfo bi = audioInfo();
860 notifyAudioChange(bi);
861 } else {

863 logSupportedFormats();

1053 wav_info.data_length = ~0;
1054 size_limit = 0;
1055 } else {
1056 wav_info.is_streamed = false;
1057 size_limit = wav_info.data_length;

1068
1070 virtual size_t write(const uint8_t *data, size_t len) override {
1071 if (!is_open) {
1072 LOGE("The WAVEncoder is not open - please call begin()");
1073 return 0;
1074 }
1075
1076 if (p_print == nullptr) {
1077 LOGE("No output stream was provided");
1078 return 0;
1079 }
1080
1081 if (!header_written) {
1082 LOGI("Writing Header");
1083 if (!header.writeHeader(p_print, wav_info)) {
1084 LOGE("Failed to write WAV header");
1085 is_open = false;
1086 return 0;
1087 }
1088 header_written = true;
1089 }
1090
1091 size_t result = 0;
1092 Print *p_out = p_encoder == nullptr ? p_print : &enc_out;
1093
1094 if (wav_info.is_streamed) {
1095 result = p_out->write((uint8_t *)data, len);
1096 } else if (size_limit > 0) {
1097 if (p_encoder == nullptr) {
1098 // uncompressed: input bytes map 1:1 to output bytes, so we can
1099 // truncate exactly at the declared data_length boundary
1100 size_t write_size = min((size_t)len, (size_t)size_limit);
1101 result = p_out->write((uint8_t *)data, write_size);
1102 size_limit -= result;
1103 } else {
1104 // compressed: the number of encoded bytes an external encoder
1105 // produces for a given amount of PCM input is not known in advance
1106 // (and may only be emitted once a full internal block is ready), so
1107 // AudioEncoder::write()'s return value (raw PCM input bytes
1108 // accepted) cannot be used to track progress towards data_length.
1109 // Track the *actual* encoded bytes written via counting_print, and
1110 // feed the encoder in frame-aligned chunks so we can stop as soon
1111 // as data_length is reached - result then correctly reflects only
1112 // the bytes of `data` that were actually consumed, so the caller
1113 // knows what still needs to be resent (e.g. to a new file).
1114 int frame_size = wav_info.bits_per_sample / 8 * wav_info.channels;
1115 if (frame_size <= 0) frame_size = 1;
1116 size_t chunk = (256 / frame_size) * frame_size;
1117 if (chunk == 0) chunk = frame_size;
1118
1119 while (result < len && size_limit > 0) {
1120 size_t n = min(chunk, len - result);
1121 counting_print.count = 0;
1122 size_t accepted = p_out->write((uint8_t *)data + result, n);
1123 size_limit -= counting_print.count;
1124 result += accepted;
1125 if (accepted < n) break; // encoder didn't accept the full chunk
1126 }
1127 }
1128
1129 if (size_limit <= 0) {
1130 LOGI("The defined size was written - so we close the WAVEncoder now");
1131 is_open = false;
1132 }
1133 }
1134 return result;
1135 }
