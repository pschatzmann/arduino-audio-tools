63 public:
64 GoertzelDetector() = default;
65
70 bool begin(const GoertzelConfig& config) {
71 this->config = config;
72 this->sample_count = 0;
73
74 if (config.target_frequency == 0.0f || config.sample_rate == 0) {
75 return false;
76 }
77
78 // Calculate the coefficient k and omega: the trigonometric functions are
79 // only evaluated here and not for each block
80 double omega =
81 (2.0 * M_PI * config.target_frequency) / config.sample_rate;
82 cos_omega = cos(omega);
83 sin_omega = sin(omega);
84 coeff = 2.0f * cos_omega;
85
86 // Reset state variables
87 reset();
88 return true;
89 }
90
96 bool processSample(float sample) {
97 // Goertzel algorithm core computation
98 float s0 = sample + coeff * s1 - s2;
99 s2 = s1;
100 s1 = s0;
101
102 sample_count++;
103
104 // Check if we've processed a complete block
105 if (sample_count >= config.block_size) {
106 // Calculate magnitude squared
107 float real = s1 - s2 * cos_omega;
108 float imag = s2 * sin_omega;
109 magnitude_squared = (real * real) + (imag * imag);
110 magnitude = sqrtf(magnitude_squared);
111
112 // Reset for next block
113 reset();
114 return true;
115 }
116
117 return false;
118 }

517 template <typename T = int16_t>
518 void processSamplesOfType(const uint8_t* data, size_t data_len,
519 int channels) {
520 const T* samples = reinterpret_cast<const T*>(data);
521 size_t num_samples = data_len / sizeof(T);
522
523 for (size_t i = 0; i < num_samples; i++) {
524 // select only samples for the configured channel
525 if (sample_no % channels == default_config.channel) {
526 float normalized = clip(NumberConverter::toFloatT<T>(samples[i]) *
527 default_config.volume);
528 LOGD("sample: %f", normalized);
529 // process all frequencies
530 for (auto& detector : detectors) {
531 if (detector.processSample(normalized)) {
532 checkDetection(detector);
533 }
534 }
535 }
536 sample_no++;
537 }
538 }

549 float clip(float value) {
550 // Clip the value to the range [-1.0, 1.0]
551 if (value > 1.0f) return 1.0f;
552 if (value < -1.0f) return -1.0f;

554 }

573 void processSamples(const uint8_t* data, size_t data_len) {
574 // return if there is nothing to detect
575 if (detectors.empty()) return;
576 int channels = default_config.channels;
577
578 switch (default_config.bits_per_sample) {
579 case 8:
580 processSamplesOfType<uint8_t>(data, data_len, channels);
581 break;
582 case 16:
583 processSamplesOfType<int16_t>(data, data_len, channels);
584 break;
585 case 24:
586 processSamplesOfType<int24_t>(data, data_len, channels);
587 break;
588 case 32:
589 processSamplesOfType<int32_t>(data, data_len, channels);
590 break;
591 default:
592 LOGE("Unsupported bits_per_sample: %d", default_config.bits_per_sample);
593 break;
594 }
