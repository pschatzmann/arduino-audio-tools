118 bool openStream(uint8_t daddr, const UsbEndpointDescriptorView& ep,
119 UsbHostStreamRole role) override {
120 tusb_desc_endpoint_t desc;
121 memset(&desc, 0, sizeof(desc));
122 desc.bLength = sizeof(tusb_desc_endpoint_t);
123 desc.bDescriptorType = TUSB_DESC_ENDPOINT;
124 desc.bEndpointAddress = ep.bEndpointAddress;
125 desc.bmAttributes.xfer = TUSB_XFER_ISOCHRONOUS;
126 desc.bmAttributes.sync = ep.sync;
127 desc.bmAttributes.usage = ep.usage;
128 desc.wMaxPacketSize = tu_htole16(ep.wMaxPacketSize);
129 desc.bInterval = ep.bInterval;
130 if (!tuh_edpt_open(daddr, &desc)) {
131 LOGE("tuh_edpt_open 0x%x failed", ep.bEndpointAddress);
132 return false;
133 }
134 Stream* s = find(daddr, ep.bEndpointAddress);
135 if (s == nullptr) {
136 streams_.push_back(Stream());
137 s = &streams_.back();
138 }
139 s->daddr = daddr;
140 s->ep_addr = ep.bEndpointAddress;
141 s->role = role;
142 s->max_packet = ep.packetSize();
143 s->buffer.resize(s->max_packet);
144 s->open = true;
145 s->due = false;
146 s->last_frame = frameNumber() - 1;
147 submit(*s);
148 return true;
149 }

151 void closeStream(uint8_t daddr, uint8_t ep_addr) override {
152 Stream* s = find(daddr, ep_addr);
153 if (s == nullptr || !s->open) return;
154 s->open = false;
155 s->due = false;
156 tuh_edpt_abort_xfer(daddr, ep_addr);
157 tuh_edpt_close(daddr, ep_addr);
158 }

175 static uint16_t driverOpen(uint8_t rhport, uint8_t daddr,
176 tusb_desc_interface_t const* itf_desc,
177 uint16_t max_len) {
178 USBAudioHostBackendTinyUSB* self = s_instance_;
179 if (self == nullptr || self->events_ == nullptr) return 0;
180 if (rhport != self->rhport_) return 0;
181 uint16_t vid = 0, pid = 0;
182 tuh_vid_pid_get(daddr, &vid, &pid);
183 const uint8_t* desc = (const uint8_t*)itf_desc;
184 uint16_t len = self->events_->onAttach(daddr, vid, pid, desc, max_len);
185 if (len < sizeof(tusb_desc_interface_t)) return 0;
186 // remember the last interface which belongs to the audio function
187 self->daddr_ = daddr;
188 self->last_itf_ = itf_desc->bInterfaceNumber;
189 for (const uint8_t* p = desc; p < desc + len && p[0] > 0; p += p[0]) {
190 if (p[1] == TUSB_DESC_INTERFACE && p[2] > self->last_itf_)
191 self->last_itf_ = p[2];
192 }
193 return len;
194 }

196 static bool driverSetConfig(uint8_t daddr, uint8_t itf_num) {
197 USBAudioHostBackendTinyUSB* self = s_instance_;
198 if (self && self->events_ && daddr == self->daddr_) {
199 self->events_->onConfigured(daddr);
200 usbh_driver_set_config_complete(daddr, self->last_itf_);
201 } else {
202 usbh_driver_set_config_complete(daddr, itf_num);
203 }
204 return true;
205 }

207 static bool driverXfer(uint8_t daddr, uint8_t ep_addr, xfer_result_t result,
208 uint32_t xferred_bytes) {
209 USBAudioHostBackendTinyUSB* self = s_instance_;
210 if (self == nullptr) return false;
211 Stream* s = self->find(daddr, ep_addr);
212 if (s == nullptr || !s->open) return true;
213 if (result == XFER_RESULT_SUCCESS && s->role != UsbHostStreamRole::TxData &&
214 self->events_) {
215 uint16_t len = xferred_bytes > s->max_packet ? s->max_packet
216 : (uint16_t)xferred_bytes;
217 self->events_->onIsoIn(s->role, s->buffer.data(), len);
218 }
219 if (!s->open) return true; // closed by the callback
220 if (self->frameAdvanced(*s)) {
221 self->submit(*s);
222 } else {
223 s->due = true;
224 }
225 return true;
226 }
227
228 static void driverClose(uint8_t daddr) {
229 USBAudioHostBackendTinyUSB* self = s_instance_;
230 if (self == nullptr) return;
231 for (auto& s : self->streams_) {
232 if (s.daddr == daddr) {
233 s.open = false;
234 s.due = false;
235 }
236 }
237 if (daddr == self->daddr_) self->daddr_ = 0;
238 if (self->events_) self->events_->onDetach(daddr);
239 }

279 bool frameAdvanced(const Stream& s) { return frameNumber() != s.last_frame; }
280
281 void submit(Stream& s) {
282 s.due = false;
283 s.last_frame = frameNumber();
284 uint16_t len = s.max_packet;
285 if (s.role == UsbHostStreamRole::TxData) {
286 len = events_ ? events_->onIsoOut(s.buffer.data(), s.max_packet) : 0;
287 }
288 if (!usbh_edpt_xfer(s.daddr, s.ep_addr, s.buffer.data(), len)) {
289 // e.g. still busy: try again with the next frame
290 s.due = true;
291 }
292 }

294 static void controlComplete(tuh_xfer_t* xfer) {
295 USBAudioHostBackendTinyUSB* self = s_instance_;
296 if (self == nullptr || self->events_ == nullptr) return;
297 self->events_->onControlComplete(xfer->result == XFER_RESULT_SUCCESS,
298 (uint16_t)xfer->actual_len);
299 }
