454 void onIsoIn(UsbHostStreamRole role, const uint8_t* data,
455 uint16_t len) override {
456 if (role == UsbHostStreamRole::Feedback) {
457 onFeedback(data, len);
458 return;
459 }
460 if (!rx_.active) return;
461 rx_.xfer_count++;
462 if (len == 0) return;

465 if (written < len) rx_.dropped_bytes += (len - written);

467
468 uint16_t onIsoOut(uint8_t* data, uint16_t max) override {
469 if (!tx_.active) return 0;
470 tx_.xfer_count++;
471 uint16_t frame = tx_.frameBytes();
472 if (frame == 0) return 0;
473 uint32_t samples = 0;
474 uint32_t fb = tx_.feedback;
475 if (fb != 0) {
476 tx_.acc_fb += fb;
477 samples = tx_.acc_fb >> 16;
478 tx_.acc_fb &= 0xFFFF;
479 } else {
480 tx_.acc += tx_.sample_rate;
481 samples = tx_.acc / tx_.packets_per_sec;
482 tx_.acc -= samples * tx_.packets_per_sec;
483 }
484 uint16_t bytes = (uint16_t)(samples * frame);
485 uint16_t max_aligned = (uint16_t)(max - (max % frame));
486 if (bytes > max_aligned) bytes = max_aligned;
487
488 // one-shot prime: wait until half of the buffer is filled
489 if (!tx_.primed) {
490 if (bufferTx().levelPercent() < 50.0f) {
491 memset(data, 0, bytes);
492 return bytes;
493 }
494 tx_.primed = true;
495 }
496
497 int avail = bufferTx().available();
498 uint16_t to_read = bytes;
499 if (avail < to_read) to_read = (uint16_t)(avail - (avail % frame));
500 int read = to_read > 0 ? bufferTx().readArray(data, to_read) : 0;
501 if (read < bytes) {
502 memset(data + read, 0, bytes - read);
503 tx_.dropped_bytes += (bytes - read);
504 }
505 return bytes;
506 }

552 uint16_t parseConfiguration(const uint8_t* desc, uint16_t len) {
553 const uint8_t* p = desc;
554 const uint8_t* end = desc + len;
555 itf_num_ac_ = decodeInterface(p).bInterfaceNumber;
556 uac_version_ = 0;
557 bool in_ac = true;
558 AltSetting* cur = nullptr;
559 AltSetting pending;
560 bool has_pending = false;
561 std::vector<AltSetting> all;
562
563 auto flush = [&]() {
564 if (has_pending && pending.has_ep && pending.is_pcm) all.push_back(pending);
565 has_pending = false;
566 cur = nullptr;
567 };
568
569 p = descNext(p);
570 while (p + 2 <= end && p[0] >= 2 && p + p[0] <= end) {
571 uint8_t type = descType(p);
572 if (type == kUsbDescTypeIad) break; // next function
573 if (type == kUsbDescTypeInterface && p[0] >= 9) {
574 UsbInterfaceDescriptorView itf = decodeInterface(p);
575 if (itf.bInterfaceClass != kUsbClassAudio) break;
576 if (itf.bInterfaceSubClass == kAudioSubclassControl) {
577 if (itf.bInterfaceNumber != itf_num_ac_) break; // next function
578 } else if (itf.bInterfaceSubClass == kAudioSubclassStreaming) {
579 in_ac = false;
580 flush();
581 if (itf.bNumEndpoints > 0) {
582 pending = AltSetting();
583 pending.itf = itf.bInterfaceNumber;
584 pending.alt = itf.bAlternateSetting;
585 has_pending = true;
586 cur = &pending;
587 }
588 } else {
589 break; // e.g. MIDI streaming: not handled by us
590 }
591 } else if (type == kUsbDescTypeCsInterface) {
592 if (in_ac) {
593 parseAcDescriptor(p);
594 } else if (cur) {
595 parseAsDescriptor(*cur, p);
596 }
597 } else if (type == kUsbDescTypeEndpoint && cur && p[0] >= 7) {
598 UsbEndpointDescriptorView ep = decodeEndpoint(p);
599 if (ep.xferType == UsbXferType::Isochronous) {
600 bool is_feedback = ep.usage == 1 ||
601 (cur->has_ep && ep.direction() == UsbDir::In &&
602 cur->ep.direction() == UsbDir::Out);
603 if (is_feedback) {
604 cur->has_fb = true;
605 cur->fb_ep = ep;
606 } else if (!cur->has_ep) {
607 cur->has_ep = true;
608 cur->ep = ep;
609 }
610 }
611 }
612 p = descNext(p);
613 }
614 flush();
615
616 for (auto& a : all) {
617 if (!isValidBitsPerSample(a.subslot * 8) || a.channels == 0) {
618 LOGI("Ignoring alt %d of itf %d: %d channels, %d bytes/sample", a.alt,
619 a.itf, a.channels, a.subslot);
620 continue;
621 }
622 bool is_rx = a.ep.direction() == UsbDir::In;
623 if (is_rx && !config_.enable_ep_in) continue;
624 if (!is_rx && !config_.enable_ep_out) continue;
625 Stream& s = is_rx ? rx_ : tx_;
626 // only one streaming interface per direction
627 if (s.present() && s.alts[0].itf != a.itf) continue;
628 s.alts.push_back(a);
629 }
630 resolveTopology(rx_, true);
631 resolveTopology(tx_, false);
632 return (uint16_t)(p - desc);
633 }

636 void parseAcDescriptor(const uint8_t* p) {
637 uint8_t subtype = descSubtype(p);
638 uint8_t n = p[0];
639 if (subtype == kAudioCsAcHeader && n >= 5) {
640 uac_version_ = descU16(p + 3) >= 0x0200 ? 2 : 1;
641 return;
642 }
643 if (n < 4) return;
644 bool v2 = uac_version_ == 2;
645 Entity e;
646 e.id = p[3];
647 e.subtype = subtype;
648 auto addSources = [&](uint8_t count_pos) {
649 if (n <= count_pos) return;
650 uint8_t cnt = p[count_pos];
651 for (uint8_t j = 0; j < cnt && e.source_count < 4 &&
652 (uint16_t)(count_pos + 1 + j) < n;
653 j++) {
654 e.sources[e.source_count++] = p[count_pos + 1 + j];
655 }
656 };
657 switch (subtype) {
658 case 0x02: // input terminal
659 if (n >= 8) {
660 e.terminal_type = descU16(p + 4);
661 if (v2) e.clock_source = p[7];
662 }
663 break;
664 case 0x03: // output terminal
665 if (n >= 8) {
666 e.terminal_type = descU16(p + 4);
667 e.sources[0] = p[7];
668 e.source_count = 1;
669 if (v2 && n >= 9) e.clock_source = p[8];
670 }
671 break;
672 case 0x04: // mixer unit
673 case 0x05: // selector unit
674 addSources(4);
675 break;
676 case kAudioCsAcFeatureUnit:
677 if (n >= 6) {
678 e.sources[0] = p[4];
679 e.source_count = 1;
680 parseFeatureUnitControls(e, p, v2);
681 }
682 break;
683 case 0x07: // UAC1 processing unit / UAC2 effect unit
684 if (v2) {
685 if (n >= 7) {
686 e.sources[0] = p[6];
687 e.source_count = 1;
688 }
689 } else {
690 addSources(6);
691 }
692 break;
693 case 0x08: // UAC1 extension unit / UAC2 processing unit
694 case 0x09: // UAC2 extension unit
695 addSources(6);
696 break;
697 case kAudioCsAcClockSource: // UAC2 clock source
698 break;
699 case 0x0B: // UAC2 clock selector
700 addSources(4);
701 break;
702 case 0x0C: // UAC2 clock multiplier
703 if (n >= 5) {
704 e.sources[0] = p[4];
705 e.source_count = 1;
706 }
707 break;
708 default:
709 return;
710 }
711 entities_.push_back(e);
712 }

715 void parseFeatureUnitControls(Entity& e, const uint8_t* p, bool v2) {
716 uint8_t n = p[0];
717 if (v2) {
718 // 4 bytes per channel starting at 5, 2 bits per control
719 for (uint8_t ch = 0; ch <= kMaxChannels && 5 + ch * 4 + 4 <= n - 1;
720 ch++) {
721 uint32_t ctrl = descU32(p + 5 + ch * 4);
722 if ((ctrl & 0x03) == 0x03) e.mute_mask |= (1u << ch);
723 if (((ctrl >> 2) & 0x03) == 0x03) e.volume_mask |= (1u << ch);
724 }
725 } else {
726 uint8_t size = p[5];
727 if (size == 0 || n < 7) return;
728 for (uint8_t ch = 0; ch <= kMaxChannels && 6 + (ch + 1) * size <= n - 1;
729 ch++) {
730 uint8_t ctrl = p[6 + ch * size];
731 if (ctrl & 0x01) e.mute_mask |= (1u << ch);
732 if (ctrl & 0x02) e.volume_mask |= (1u << ch);
733 }
734 }

738 void parseAsDescriptor(AltSetting& a, const uint8_t* p) {
739 uint8_t subtype = descSubtype(p);
740 uint8_t n = p[0];
741 if (uac_version_ == 2) {
742 if (subtype == kAudioCsAsGeneral && n >= 11) {
743 a.terminal_link = p[3];
744 // bFormatType == 1 and bmFormats bit 0 (PCM)
745 a.is_pcm = p[5] == kAudioFormatTypeI && (descU32(p + 6) & 0x01);
746 a.channels = p[10];
747 } else if (subtype == kAudioCsAsFormatType && n >= 6) {
748 if (p[3] != kAudioFormatTypeI) a.is_pcm = false;
749 a.subslot = p[4];
750 a.bits = p[5];
751 }
752 } else {
753 if (subtype == kAudioCsAsGeneral && n >= 7) {
754 a.terminal_link = p[3];
755 a.is_pcm = descU16(p + 5) == 0x0001; // wFormatTag: PCM
756 } else if (subtype == kAudioCsAsFormatType && n >= 8) {
757 if (p[3] != kAudioFormatTypeI) {
758 a.is_pcm = false;
759 return;
760 }
761 a.channels = p[4];
762 a.subslot = p[5];
763 a.bits = p[6];
764 uint8_t freq_type = p[7];
765 if (freq_type == 0 && n >= 14) {
766 a.continuous = true;
767 a.rates[0] = descU24(p + 8);
768 a.rates[1] = descU24(p + 11);
769 a.rate_count = 2;
770 } else {
771 for (uint8_t j = 0; j < freq_type && j < kMaxRates &&
772 8 + (j + 1) * 3 <= n;
773 j++) {
774 a.rates[a.rate_count++] = descU24(p + 8 + j * 3);
775 }
776 }
777 }
778 }

787 void resolveTopology(Stream& s, bool is_rx) {
788 if (!s.present()) return;
789 uint8_t link = s.alts[0].terminal_link;
790 Entity* term = findEntity(link);
791 if (term == nullptr) return;
792
793 if (uac_version_ == 2) s.clock_id = resolveClock(term->clock_source);
794
795 Entity* fu = nullptr;
796 if (is_rx) {
797 // microphone: walk upstream from the USB streaming output terminal
798 Entity* e = term;
799 for (int depth = 0; depth < 8 && e && e->source_count > 0; depth++) {
800 e = findEntity(e->sources[0]);
801 if (e && e->subtype == kAudioCsAcFeatureUnit) {
802 fu = e;
803 break;
804 }
805 }
806 } else {
807 // speaker: walk downstream from the USB streaming input terminal
808 uint8_t id = term->id;
809 for (int depth = 0; depth < 8 && fu == nullptr; depth++) {
810 Entity* next = nullptr;
811 for (auto& e : entities_) {
812 if (e.subtype == kAudioCsAcClockSource || e.subtype == 0x0B ||
813 e.subtype == 0x0C)
814 continue;
815 for (uint8_t j = 0; j < e.source_count; j++) {
816 if (e.sources[j] == id) next = &e;
817 }
818 if (next) break;
819 }
820 if (next == nullptr) break;
821 if (next->subtype == kAudioCsAcFeatureUnit) fu = next;
822 id = next->id;
823 }
824 }
825 if (fu) {
826 s.fu_id = fu->id;
827 s.fu_mute_mask = fu->mute_mask;
828 s.fu_volume_mask = fu->volume_mask;
829 }
830 LOGI("%s: itf %d, %d alt settings, feature unit %d, clock %d",
831 is_rx ? "mic" : "speaker", s.alts[0].itf, (int)s.alts.size(),
832 s.fu_id, s.clock_id);
833 }

836 uint8_t resolveClock(uint8_t id) {
837 for (int depth = 0; depth < 4 && id != 0; depth++) {
838 Entity* e = findEntity(id);
839 if (e == nullptr) return id;
840 if (e->subtype == kAudioCsAcClockSource) return id;
841 if (e->source_count == 0) return 0;
842 id = e->sources[0];
843 }
844 return id;
845 }

847 // ── Format negotiation ───────────────────────────────────────────────────
848
850 bool supportsRate(const AltSetting& a,
851 const std::vector<RateRange>& clock, uint32_t rate) {
852 if (!fitsBandwidth(a, rate)) return false;
853 if (uac_version_ == 2) {
854 if (clock.empty()) return true; // unknown: assume that it works
855 for (auto& r : clock) {
856 if (rate < r.min || rate > r.max) continue;
857 if (r.res == 0 || (rate - r.min) % r.res == 0) return true;
858 }
859 return false;
860 }
861 if (a.rate_count == 0) return true;
862 if (a.continuous) return rate >= a.rates[0] && rate <= a.rates[1];
863 for (uint8_t j = 0; j < a.rate_count; j++) {
864 if (a.rates[j] == rate) return true;
865 }
866 return false;
867 }
868
869 bool fitsBandwidth(const AltSetting& a, uint32_t rate) {
870 uint32_t pps = packetsPerSecond(a);
871 uint32_t samples = (rate + pps - 1) / pps;
872 return samples * a.channels * a.subslot <= a.ep.packetSize();
873 }
874
875 uint16_t packetsPerSecond(const AltSetting& a) {
876 uint8_t interval = a.ep.bInterval;
877 if (uac_version_ == 1 || interval == 0) interval = 1;
878 if (interval > 4) interval = 4;
879 uint16_t base = backend().speed() == UsbSpeed::High ? 8000 : 1000;
880 return (uint16_t)(base >> (interval - 1));
881 }
882
884 uint32_t closestRate(const AltSetting& a,
885 const std::vector<RateRange>& clock,
886 uint32_t wanted) {
887 if (supportsRate(a, clock, wanted)) return wanted;
888 static const uint32_t common[] = {48000, 44100, 32000, 24000, 22050,
889 16000, 11025, 8000, 96000, 88200};
890 uint32_t best = 0;
891 uint32_t best_diff = UINT32_MAX;
892 auto consider = [&](uint32_t r) {
893 if (r == 0 || !supportsRate(a, clock, r)) return;
894 uint32_t diff = r > wanted ? r - wanted : wanted - r;
895 if (diff < best_diff) {
896 best = r;
897 best_diff = diff;
898 }
899 };
900 if (uac_version_ == 2) {
901 for (auto& r : clock) {
902 consider(r.min);
903 consider(r.max);
904 }
905 } else {
906 for (uint8_t j = 0; j < a.rate_count; j++) consider(a.rates[j]);
907 }
908 for (auto r : common) consider(r);
909 return best;
910 }
911
913 bool selectAlt(Stream& s, const std::vector<RateRange>& clock,
914 AudioInfo& negotiated) {
915 s.selected = -1;
916 if (!s.present()) return false;
917 int best = -1;
918 int best_score = -1;
919 uint32_t best_rate = 0;
920 for (int j = 0; j < (int)s.alts.size(); j++) {
921 const AltSetting& a = s.alts[j];
922 uint32_t rate = closestRate(a, clock, config_.sample_rate);
923 if (rate == 0) continue;
924 int score = 0;
925 if (rate == config_.sample_rate) score += 4;
926 if (a.channels == config_.channels) score += 2;
927 if (a.subslot * 8 == config_.bits_per_sample) score += 1;
928 // prefer the smaller packet size on a tie
929 if (score > best_score ||
930 (score == best_score &&
931 a.ep.packetSize() < s.alts[best].ep.packetSize())) {
932 best = j;
933 best_score = score;
934 best_rate = rate;
935 }
936 }
937 if (best < 0) return false;
938 if (best_score != 7 && !config_.host_apply_device_format) {
939 LOGE("The attached device does not support %d Hz, %d channels, %d bits",
940 (int)config_.sample_rate, config_.channels, config_.bits_per_sample);
941 return false;
942 }
943 s.selected = best;
944 s.sample_rate = best_rate;
945 negotiated.sample_rate = best_rate;
946 negotiated.channels = s.alts[best].channels;
947 negotiated.bits_per_sample = s.alts[best].subslot * 8;
948 LOGI("Selected itf %d alt %d: %d Hz, %d channels, %d bits",
949 s.alts[best].itf, s.alts[best].alt, (int)best_rate,
950 negotiated.channels, negotiated.bits_per_sample);
951 return true;
952 }

1067
1069 void poll() {
1070 if (!is_started_ || !attached_ || !configured_) return;
1071 if (ctrl_busy_) {
1072 if (backend().millis() - ctrl_start_ms_ > config_.host_ctrl_timeout_ms) {
1073 LOGW("USB audio control request timed out");
1074 onControlComplete(false, 0);
1075 }
1076 return;
1077 }
1078 if (step_ != Step::Idle) {
1079 run();
1080 return;
1081 }
1082 if (!mounted_) return;
1083 if (reconfigure_pending_) {
1084 reconfigure_pending_ = false;
1085 reconfiguring_ = true;
1086 next(Step::StopRx);
1087 run();
1088 return;
1089 }
1090 startRuntimeRequest();
1091 }
1092
1094 void run() {
1095 if (in_run_) {
1096 run_again_ = true;
1097 return;
1098 }
1099 in_run_ = true;
1100 do {
1101 run_again_ = false;
1102 while (step_ != Step::Idle && !ctrl_busy_ && attached_) {
1103 if (!executeStep()) break; // retry in the next poll()
1104 }
1105 } while (run_again_);
1106 in_run_ = false;
1107 }
1108
1110 void next(Step s) {
1111 step_ = s;
1112 step_done_ = false;
1113 }
1114
1117 bool executeStep() {
1118 // Steps which issue a control request are entered twice: first to send
1119 // the request (step_done_ == false), then with the result.
1120 switch (step_) {
1121 case Step::StopRx:
1122 return stopStream(rx_, Step::StopTx);
1123 case Step::StopTx:
1124 return stopStream(tx_, Step::ClockRangeHeaderRx);
1125
1126 case Step::ClockRangeHeaderRx:
1127 return clockRangeHeader(rx_, clock_ranges_rx_, Step::ClockRangeRx,
1128 Step::ClockRangeHeaderTx);
1129 case Step::ClockRangeRx:
1130 return clockRange(rx_, clock_ranges_rx_, Step::ClockRangeHeaderTx);
1131 case Step::ClockRangeHeaderTx:
1132 if (tx_.present() && rx_.present() && tx_.clock_id == rx_.clock_id &&
1133 uac_version_ == 2) {
1134 clock_ranges_tx_ = clock_ranges_rx_;
1135 next(Step::Select);
1136 return true;
1137 }
1138 return clockRangeHeader(tx_, clock_ranges_tx_, Step::ClockRangeTx,
1139 Step::Select);
1140 case Step::ClockRangeTx:
1141 return clockRange(tx_, clock_ranges_tx_, Step::Select);
1142
1143 case Step::Select: {
1144 bool rx_ok = selectAlt(rx_, clock_ranges_rx_, rx_info_);
1145 bool tx_ok = selectAlt(tx_, clock_ranges_tx_, tx_info_);
1146 if (!rx_ok && !tx_ok) {
1147 LOGE("No usable audio stream on the attached device");
1148 next(Step::Idle);
1149 return true;
1150 }
1151 if (rx_.present() && !rx_ok) rx_.alts.clear();
1152 if (tx_.present() && !tx_ok) tx_.alts.clear();
1153 // a common clock can only run at one rate
1154 if (uac_version_ == 2 && rx_ok && tx_ok &&
1155 rx_.clock_id == tx_.clock_id && rx_.sample_rate != tx_.sample_rate) {
1156 LOGW("Shared clock: using %d Hz for both streams",
1157 (int)rx_.sample_rate);
1158 tx_.sample_rate = rx_.sample_rate;
1159 tx_info_.sample_rate = rx_.sample_rate;
1160 }
1161 updateBufferSize();
1162 next(Step::SetClockRx);
1163 return true;
1164 }
1165
1166 case Step::SetClockRx:
1167 return setClock(rx_, Step::SetInterfaceRx);
1168 case Step::SetInterfaceRx:
1169 return setInterface(rx_, Step::OpenRx, Step::SetClockTx);
1170 case Step::OpenRx:
1171 openStream(rx_, true);
1172 next(Step::SetEpRateRx);
1173 return true;
1174 case Step::SetEpRateRx:
1175 return setEpRate(rx_, Step::VolumeRangeRx);
1176 case Step::VolumeRangeRx:
1177 return volumeRange(rx_, false, Step::VolumeMaxRx, Step::SetClockTx);
1178 case Step::VolumeMaxRx:
1179 return volumeRange(rx_, true, Step::SetClockTx, Step::SetClockTx);
1180
1181 case Step::SetClockTx:
1182 if (tx_.selected >= 0 && rx_.selected >= 0 &&
1183 tx_.clock_id == rx_.clock_id) {
1184 next(Step::SetInterfaceTx); // shared clock was already set
1185 return true;
1186 }
1187 return setClock(tx_, Step::SetInterfaceTx);
1188 case Step::SetInterfaceTx:
1189 return setInterface(tx_, Step::OpenTx, Step::Done);
1190 case Step::OpenTx:
1191 openStream(tx_, false);
1192 next(Step::SetEpRateTx);
1193 return true;
1194 case Step::SetEpRateTx:
1195 return setEpRate(tx_, Step::VolumeRangeTx);
1196 case Step::VolumeRangeTx:
1197 return volumeRange(tx_, false, Step::VolumeMaxTx, Step::Done);
1198 case Step::VolumeMaxTx:
1199 return volumeRange(tx_, true, Step::Done, Step::Done);
1200
1201 case Step::Done:
1202 finishMount();
1203 next(Step::Idle);
1204 return true;
1205
1206 default:
1207 next(Step::Idle);
1208 return true;
1209 }
1210 }

1396 bool sendFeature(bool is_volume, uint8_t ch) {
1397 // the first call starts with rx; finishRuntimeRequest continues with tx
1398 pending_is_volume_ = is_volume;
1399 pending_channel_ = ch;
1400 pending_stream_ = &rx_;
1401 return sendFeatureFor(rx_) || sendFeatureNextStream();
1402 }
1403
1404 bool sendFeatureNextStream() {
1405 if (pending_stream_ == &rx_) {
1406 pending_stream_ = &tx_;
1407 if (sendFeatureFor(tx_)) return true;
1408 }
1409 pending_stream_ = nullptr;
1410 return false;
1411 }
1412
1413 bool sendFeatureFor(Stream& s) {
1414 if (!s.active || s.fu_id == 0) return false;
1415 uint8_t ch = pending_channel_;
1416 if (ch > kMaxChannels) return false;
1417 uint16_t mask = pending_is_volume_ ? s.fu_volume_mask : s.fu_mute_mask;
1418 if ((mask & (1u << ch)) == 0) {
1419 // no master control: apply the master to the individual channels
1420 if (ch == 0) {
1421 for (uint8_t c = 1; c < volume_.size() && c <= kMaxChannels; c++) {
1422 if (mask & (1u << c)) {
1423 if (pending_is_volume_)
1424 volume_pending_[c] = 1;
1425 else
1426 mute_pending_[c] = 1;
1427 }
1428 }
1429 }
1430 return false;
1431 }
1432 if (pending_is_volume_) {

1452 } else if (len >= 4) {
1453 value = descU32(data); // 16.16
1454 } else {
1455 return;
1456 }

1458 // devices send the 10.14 format in 4 bytes

1460 if (nom == 0) return;

1467 }
