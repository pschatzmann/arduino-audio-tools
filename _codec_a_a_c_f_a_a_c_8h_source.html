115 void end() override {
116 if (enc == nullptr) return;
117 TRACEI();
118 // encode the final partial frame
119 if (pcm_pos > 0) {
120 encode(pcm_buffer.data(), pcm_pos);
121 pcm_pos = 0;
122 }
123 // drain the encoder
124 uint32_t written = 0;
125 do {
126 if (!encode(nullptr, 0, &written)) break;
127 } while (written > 0);
128 faac_encoder_close(&enc);

133 if (enc == nullptr || data == nullptr) return 0;

136 for (size_t j = 0; j < sample_count; j++) {
