100// ── Descriptor byte-walking helpers ───────────────────────────────────────
101// Pure byte arithmetic — works identically against any USB stack's raw
102// descriptor buffer. Replace TinyUSB's tu_desc_next/_type/_subtype and
103// tu_unaligned_read16/32.
104
105inline uint8_t const* descNext(uint8_t const* p) { return p + p[0]; }
106inline uint8_t descType(uint8_t const* p) { return p[1]; }
107inline uint8_t descSubtype(uint8_t const* p) { return p[2]; }
108
109inline uint16_t descU16(uint8_t const* p) {
110 return (uint16_t)p[0] | ((uint16_t)p[1] << 8);
111}
112
113inline uint32_t descU32(uint8_t const* p) {
114 return (uint32_t)p[0] | ((uint32_t)p[1] << 8) | ((uint32_t)p[2] << 16) |
115 ((uint32_t)p[3] << 24);
116}

121inline UsbEndpointDescriptorView decodeEndpoint(uint8_t const* p) {
122 UsbEndpointDescriptorView v;
123 v.bEndpointAddress = p[2];
124 uint8_t attr = p[3];
125 v.xferType = (UsbXferType)(attr & 0x03);
126 v.sync = (attr >> 2) & 0x03;
127 v.usage = (attr >> 4) & 0x03;
128 v.wMaxPacketSize = descU16(p + 4);
129 v.bInterval = p[6];
130 return v;
131}
