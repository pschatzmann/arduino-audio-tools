117inline uint32_t adcContinuousAlignFrameSize(uint32_t frame_size_bytes) {
118 uint32_t alignment = SOC_ADC_DIGI_DATA_BYTES_PER_CONV;
119 if (alignment == 0) return frame_size_bytes;
120 uint32_t remainder = frame_size_bytes % alignment;
121 if (remainder == 0) return frame_size_bytes;
122 return frame_size_bytes + (alignment - remainder);
123}
124
125inline uint32_t adcContinuousMaxConvFrameBytes() {
126 return ADC_CONTINUOUS_RX_MAX_CONV_FRAME_BYTES;
127}
128
129inline uint32_t adcContinuousMaxResultsPerFrame() {
130 return adcContinuousMaxConvFrameBytes() / SOC_ADC_DIGI_RESULT_BYTES;

175 AnalogConfigESP32V1(RxTxMode rxtxMode = TX_MODE) {
176 sample_rate = 44100;
177 channels = 2;
178 bits_per_sample = 16;
179 rx_tx_mode = rxtxMode;
180 switch (rx_tx_mode) {
181 case RX_MODE: {
182 // make sure that the proposed sample rate is valid
183 if (sample_rate * channels > SOC_ADC_SAMPLE_FREQ_THRES_HIGH) {
184 sample_rate = SOC_ADC_SAMPLE_FREQ_THRES_HIGH / channels;
185 }
186 LOGI("I2S_MODE_ADC_BUILT_IN");
187 break;
188 }
189#ifdef HAS_ESP32_DAC
190 case TX_MODE:
191 use_apll = true;
192 LOGI("I2S_MODE_DAC_BUILT_IN");
193 break;
194#endif
195
196 default:
197 LOGE("RxTxMode not supported: %d", rx_tx_mode);
198 break;
199 }
200 }
