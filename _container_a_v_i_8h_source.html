21class ParseBuffer {
22public:
23 size_t writeArray(uint8_t *data, size_t len) {
24 int to_write = min(availableToWrite(), (size_t)len);
25 memmove(vector.data() + available_byte_count, data, to_write);
26 available_byte_count += to_write;
27 return to_write;
28 }
29 void consume(int size) {
30 // use pointer arithmetic (not the bounds-checked operator[]): when size
31 // == available_byte_count (buffer fully drained, common for chunks at
32 // least as big as the buffer capacity, e.g. large raw video frames),
33 // vector.data() + size legally points one-past-the-end and the memmove
34 // length is 0, which operator[] would reject even though it's a no-op
35 memmove(vector.data(), vector.data() + size, available_byte_count - size);
36 available_byte_count -= size;
37 }
38 bool resize(size_t size) {
39 vector.resize(size + 4);
40 return vector.data() != nullptr;
41 }
42
43 uint8_t *data() { return vector.data(); }
44
45 size_t availableToWrite() { return size() - available_byte_count; }
46
47 size_t available() { return available_byte_count; }
48
49 void clear() {
50 available_byte_count = 0;
51 memset(vector.data(), 0, vector.size());
52 }
53
54 bool isEmpty() { return available_byte_count == 0; }
55
56 size_t size() { return vector.size(); }
57
58 long indexOf(const char *str) {
59 uint8_t *ptr = (uint8_t *)memmem(vector.data(), available_byte_count, str,
60 strlen(str));
61 return ptr == nullptr ? -1l : ptr - vector.data();
62 }

435 bool parse() {
436 bool result = true;
437 switch (parse_state) {
438 case ParseHeader: {
439 result = parseHeader();
440 if (result)
441 parse_state = ParseHdrl;
442 } break;
443
444 case ParseHdrl: {
445 ParseObject hdrl = parseList("hdrl");
446 result = hdrl.isValid();
447 if (result) {
448 parse_state = ParseAvih;
449 }
450 } break;
451
452 case ParseAvih: {
453 ParseObject avih = parseChunk("avih");
454 result = avih.isValid();
455 if (result) {
456 main_header = *(avih.asAVIMainHeader(parse_buffer.data()));
457 stream_header.resize(main_header.dwStreams);
458 consume(avih.size());
459 parse_state = ParseStrl;
460 }
461 } break;
462
463 case ParseStrl: {
464 ParseObject strl = parseList("strl");
465 ParseObject strh = parseChunk("strh");
466 stream_header[++stream_header_idx] =
467 *(strh.asAVIStreamHeader(parse_buffer.data()));
468 consume(strh.size());
469 parse_state = ParseStrf;
470 } break;
471
472 case ParseStrf: {
473 ParseObject strf = parseChunk("strf");
474 if (isCurrentStreamAudio()) {
475 audio_info = *(strf.asAVIAudioFormat(parse_buffer.data()));
476 setupAudioInfo();

478 content_types.push_back(Audio);
479 consume(strf.size());

481 video_info = *(strf.asAVIVideoFormat(parse_buffer.data()));

490 parse_state = AfterStrf;
491 } break;
492
493 case AfterStrf: {
494 // ignore all data until we find a new List
495 int pos = parse_buffer.indexOf("LIST");
496 if (pos >= 0) {
497 consume(pos);
498 ParseObject tmp = tryParseList();
499 if (StrView(tmp.id()).equals("strl")) {
500 parse_state = ParseStrl;
501 } else if (StrView(tmp.id()).equals("movi")) {
502 parse_state = ParseMovi;
503 } else {
504 // e.g. ignore info
505 consume(tmp.size() + LIST_HEADER_SIZE);
506 }
507 } else {
508 // no valid data, so throw it away, we keep the last 4 digits in case
509 // if it contains the beginning of a LIST
510 cleanupStack();
511 consume(parse_buffer.available() - 4);
512 }
513 } break;
514
515 case ParseMovi: {
516 ParseObject movi = tryParseList();
517 if (StrView(movi.id()).equals("movi")) {
518 consume(LIST_HEADER_SIZE);
519 is_metadata_ready = true;
520 if (validation_cb)
521 is_parsing_active = (validation_cb(*this));
522 processStack(movi);
523 movi_end_pos = movi.end_pos;
524 parse_state = SubChunk;
525 // trigger new write
526 result = false;
527 }
528 } break;
529
530 case SubChunk: {
531 // rec is optinal
532 ParseObject hdrl = tryParseList();
533 if (StrView(hdrl.id()).equals("rec")) {
534 consume(CHUNK_HEADER_SIZE);
535 processStack(hdrl);
536 }
537
538 current_stream_data = parseAVIStreamData();
539 parse_state = SubChunkContinue;
540 open_subchunk_len = current_stream_data.open;
541 if (current_stream_data.isVideo()) {
542 LOGI("video:[%d]->[%d]", (int)current_stream_data.start_pos,
543 (int)current_stream_data.end_pos);
544 if (p_output_video != nullptr)
545 p_output_video->beginFrame(current_stream_data.declaredSize());
546 } else if (current_stream_data.isAudio()) {
547 LOGI("audio:[%d]->[%d]", (int)current_stream_data.start_pos,
548 (int)current_stream_data.end_pos);
549 } else {
550 LOGW("unknown subchunk at %d", (int)current_pos);
551 }
552
553 } break;
554
555 case SubChunkContinue: {
556 writeData();
557 if (open_subchunk_len == 0) {
558 if (current_stream_data.isVideo() && p_output_video != nullptr) {
559 uint32_t time_used_ms = p_output_video->endFrame();
560 p_synch->delayVideoFrame(main_header.dwMicroSecPerFrame, time_used_ms);
561 }
562 if (tryParseChunk("idx").isValid()) {
563 parse_state = ParseIgnore;
564 } else if (tryParseList("rec").isValid()) {
565 parse_state = ParseRec;
566 } else {
567 if (current_pos >= movi_end_pos) {
568 parse_state = ParseIgnore;
569 } else {
570 parse_state = SubChunk;
571 }
572 }
573 }
574 } break;
575
576 case ParseIgnore: {
577 LOGD("ParseIgnore");
578 parse_buffer.clear();
579 } break;
580
581 default:
582 result = false;
583 break;
584 }
585 return result;
586 }

642 void writeData() {
643 long to_write = min((long)parse_buffer.available(), open_subchunk_len);
644 // open_subchunk_len spans the word-aligned chunk (current_stream_data
645 // .size()), which for an odd declared size includes a trailing RIFF pad
646 // byte that must be consumed from the buffer but not delivered as
647 // payload to the audio/video sink - clip to the real, unpadded length.
648 long already_written = (long)current_stream_data.size() - open_subchunk_len;
649 long payload_left = (long)current_stream_data.declaredSize() - already_written;
650 if (payload_left < 0) payload_left = 0;
651 long payload_to_write = min(to_write, payload_left);
652
653 if (current_stream_data.isAudio()) {
654 LOGD("audio %d", (int)to_write);
655 if (!is_mute && payload_to_write > 0){
656 p_synch->writeAudio(p_output_audio, parse_buffer.data(), payload_to_write);
657 }
658 open_subchunk_len -= to_write;
659 cleanupStack();
660 consume(to_write);
661 } else if (current_stream_data.isVideo()) {
662 LOGD("video %d", (int)to_write);
663 if (p_output_video != nullptr && payload_to_write > 0)
664 p_output_video->write(parse_buffer.data(), payload_to_write);
665 open_subchunk_len -= to_write;
666 cleanupStack();
667 consume(to_write);
668 }
