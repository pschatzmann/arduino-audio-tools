69template <typename T = float>
70class FIR : public Filter<T> {
71 public:
72 template <size_t B>
73 FIR(const T (&b)[B], const T factor = 1.0)
74 : lenB(B), factor(factor), needs_divide(factor != T(1)) {
75 setValues(b);
76 }
77
78 template <size_t B>
79 void setValues(const T (&b)[B]) {
80 x.resize(lenB);
81 coeff_b.resize(2 * lenB - 1);
82 for (uint16_t i = 0; i < 2 * lenB - 1; i++) {
83 coeff_b[i] = b[(2 * lenB - 1 - i) % lenB];
84 }
85 }
86
87 void reset() override {
88 i_b = 0;
89 for (uint16_t i = 0; i < lenB; i++) x[i] = 0;
90 }
91
92 T process(T value) override {
93 x[i_b] = value;
94 T b_terms = 0;
95 T* b_shift = &coeff_b[lenB - i_b - 1];
96 for (uint16_t i = 0; i < lenB; i++) {
97 b_terms += b_shift[i] * x[i];
98 }
99 i_b++;
100 if (i_b == lenB) i_b = 0;
101
102#ifdef USE_TYPETRAITS
103 if (!(std::is_same<T, float>::value || std::is_same<T, double>::value)) {
104 b_terms = b_terms / factor;
105 }
106#else
107 // needs_divide is computed once in the constructor (not per sample) --
108 // for a bounded fixed-point T like q1_14_t, comparing against a float
109 // literal here on every process() call would otherwise force a
110 // per-sample float construction/comparison.
111 if (needs_divide) {
112 b_terms = b_terms / factor;
113 }
114#endif
115 return b_terms;
116 }
117
118 private:
119 const uint16_t lenB;
120 uint16_t i_b = 0;
121 Vector<T> x;
122 Vector<T> coeff_b;
123 T factor;
124 bool needs_divide;
125};

137template <typename T = float>
138class IIR : public Filter<T> {
139 public:
140 template <size_t B, size_t A>
141 IIR(const T (&b)[B], const T (&_a)[A], T factor = 1.0)
142 : factor(factor), needs_divide(factor != T(1)), lenB(B), lenA(A - 1) {
143 x.resize(lenB);
144 y.resize(lenA);
145 coeff_b.resize(2 * lenB - 1);
146 coeff_a.resize(2 * lenA - 1);
147 T a0 = _a[0];
148 const T* a = &_a[1];
149 for (uint16_t i = 0; i < 2 * lenB - 1; i++) {
150 coeff_b[i] = b[(2 * lenB - 1 - i) % lenB] / a0;
151 }
152 for (uint16_t i = 0; i < 2 * lenA - 1; i++) {
153 coeff_a[i] = a[(2 * lenA - 2 - i) % lenA] / a0;
154 }
155 }
156
157 void reset() override {
158 i_b = 0;
159 i_a = 0;
160 for (uint16_t i = 0; i < lenB; i++) x[i] = 0;
161 for (uint16_t i = 0; i < lenA; i++) y[i] = 0;
162 }
163
164 T process(T value) override {
165 x[i_b] = value;
166 T b_terms = 0;
167 T* b_shift = &coeff_b[lenB - i_b - 1];
168
169 T a_terms = 0;
170 T* a_shift = &coeff_a[lenA - i_a - 1];
171
172 for (uint16_t i = 0; i < lenB; i++) {
173 b_terms += x[i] * b_shift[i];
174 }
175 for (uint16_t i = 0; i < lenA; i++) {
176 a_terms += y[i] * a_shift[i];
177 }
178
179 T filtered = b_terms - a_terms;
180 y[i_a] = filtered;
181 i_b++;
182 if (i_b == lenB) i_b = 0;
183 i_a++;
184 if (i_a == lenA) i_a = 0;
185
186#ifdef USE_TYPETRAITS
187 if (!(std::is_same<T, float>::value || std::is_same<T, double>::value)) {
188 filtered = filtered / factor;
189 }
190#else
191 // needs_divide is computed once in the constructor (not per sample) --
192 // see FIR::process() for why.
193 if (needs_divide) {
194 filtered = filtered / factor;
195 }
196#endif
197 return filtered;
198 }
199
200 private:
201 T factor;
202 bool needs_divide;
203 const uint16_t lenB, lenA;
204 uint16_t i_b = 0, i_a = 0;
205 Vector<T> x;
206 Vector<T> y;
207 Vector<T> coeff_b;
208 Vector<T> coeff_a;
209};

356template <typename T>
357void checkCoeffRange(const BiQuadCoeffs &c) {
358 auto check = [](const char *name, float value) {
359 T t = value;
360 float roundtrip = (float)t;
361 // allow normal quantization noise (e.g. ~1/16384 for q1_14_t); anything
362 // larger indicates the value was clamped/saturated, not just rounded.
363 if (fabs(roundtrip - value) > fabs(value) * 0.01f + 1e-4f) {
364 LOGE(
365 "Filter coefficient %s=%f does not fit in the range of T (stored "
366 "as %f) - the filter will be inaccurate",
367 name, value, roundtrip);
368 }
369 };
370 check("b_0", c.b_0);
371 check("b_1", c.b_1);
372 check("b_2", c.b_2);
373 check("a_1", c.a_1);
374 check("a_2", c.a_2);
375}

378inline BiQuadCoeffs calculateLowPassCoeffs(float frequency, float sampleRate,
379 float q) {
380 float w0 = frequency * (2.0f * PI / sampleRate);
381 float sinW0 = sin(w0);
382 float alpha = sinW0 / (q * 2.0f);
383 float cosW0 = cos(w0);
384 float scale = 1.0f / (1.0f + alpha);
385 BiQuadCoeffs c;
386 c.b_0 = ((1.0f - cosW0) / 2.0f) * scale;
387 c.b_1 = (1.0f - cosW0) * scale;
388 c.b_2 = c.b_0;
389 c.a_1 = (-2.0f * cosW0) * scale;
390 c.a_2 = (1.0f - alpha) * scale;
391 return c;
392}

723inline BiQuadCoeffs calculateLowShelfCoeffs(float frequency, float sampleRate,
724 float gain, float slope) {
725 float a = pow(10.0f, gain / 40.0f);
726 float w0 = frequency * (2.0f * PI / sampleRate);
727 float sinW0 = sin(w0);
728 // float alpha = (sinW0 * sqrt((a+1/a)*(1/slope-1)+2) ) / 2.0;
729 float cosW0 = cos(w0);
730 // generate three helper-values (intermediate results):
731 float sinsq =
732 sinW0 * sqrt((pow(a, 2.0f) + 1.0f) * (1.0f / slope - 1.0f) + 2.0f * a);
733 float aMinus = (a - 1.0f) * cosW0;
734 float aPlus = (a + 1.0f) * cosW0;
735 float scale = 1.0f / ((a + 1.0f) + aMinus + sinsq);
736 BiQuadCoeffs c;
737 c.b_0 = a * ((a + 1.0f) - aMinus + sinsq) * scale;
738 c.b_1 = 2.0f * a * ((a - 1.0f) - aPlus) * scale;
739 c.b_2 = a * ((a + 1.0f) - aMinus - sinsq) * scale;
740 c.a_1 = -2.0f * ((a - 1.0f) + aPlus) * scale;
741 c.a_2 = ((a + 1.0f) + aMinus - sinsq) * scale;
742 return c;
743}

831inline BiQuadCoeffs calculateHighShelfCoeffs(float frequency,
832 float sampleRate, float gain,
833 float slope) {
834 float a = pow(10.0f, gain / 40.0f);
835 float w0 = frequency * (2.0f * PI / sampleRate);
836 float sinW0 = sin(w0);
837 // float alpha = (sinW0 * sqrt((a+1/a)*(1/slope-1)+2) ) / 2.0;
838 float cosW0 = cos(w0);
839 // generate three helper-values (intermediate results):
840 float sinsq =
841 sinW0 * sqrt((pow(a, 2.0f) + 1.0f) * (1.0f / slope - 1.0f) + 2.0f * a);

938inline BiQuadCoeffs calculatePeakCoeffs(float frequency, float sampleRate,
939 float q, float gain) {
940 float a = powf(10.0f, gain / 40.0f);
941 float w0 = frequency * (2.0f * PI / sampleRate);
942 float cosW0 = cosf(w0);
943 float alpha = sinf(w0) / (2.0f * q);
944 float scale = 1.0f / (1.0f + alpha / a);
945 BiQuadCoeffs c;
946 c.b_0 = (1.0f + alpha * a) * scale;
947 c.b_1 = -2.0f * cosW0 * scale;
948 c.b_2 = (1.0f - alpha * a) * scale;
949 c.a_1 = -2.0f * cosW0 * scale;
950 c.a_2 = (1.0f - alpha / a) * scale;
951 return c;
952}
