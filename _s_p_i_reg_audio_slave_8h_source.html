183 static void encodeInfo(const AudioInfo& in, uint8_t* out) {
184 out[0] = in.sample_rate & 0xFF;
185 out[1] = (in.sample_rate >> 8) & 0xFF;
186 out[2] = (in.sample_rate >> 16) & 0xFF;
187 out[3] = (in.sample_rate >> 24) & 0xFF;
188 out[4] = in.channels & 0xFF;
189 out[5] = (in.channels >> 8) & 0xFF;
190 out[6] = in.bits_per_sample;
191 out[7] = 0;
192 }

194 static void decodeInfo(const uint8_t* in, AudioInfo& out) {
195 out.sample_rate = (uint32_t)in[0] | ((uint32_t)in[1] << 8) |
196 ((uint32_t)in[2] << 16) | ((uint32_t)in[3] << 24);
197 out.channels = (uint16_t)in[4] | ((uint16_t)in[5] << 8);
198 out.bits_per_sample = in[6];
199 }

206 void onFrame(size_t len) {
207 int c = spi.read();
208 if (c < 0) return;
209 len--;
210 uint8_t cmd = (uint8_t)c;
211 uint8_t addr = cmd & SPIREG_ADDR_MASK;
212 if (cmd & SPIREG_READ_MASK) {
213 size_t want = 0;
214 if (len >= 1) {
215 want = (uint8_t)spi.read();
216 len--;
217 }
218 while (len--) spi.read();
219 stageReply(addr, want);
220 } else {
221 storeWrite(addr, len);
222 }
223 }
224
225 void storeWrite(uint8_t addr, size_t len) {
226 switch (addr) {
227 case SPIREG_DATA: {
228 uint8_t chunk[32];
229 while (len) {
230 size_t n = len < sizeof(chunk) ? len : sizeof(chunk);
231 n = spi.read(chunk, n);
232 if (n == 0) break;
233 len -= n;
234 int free = p_buffer->availableForWrite();
235 int w = free > 0 ? p_buffer->writeArray(chunk, min((int)n, free)) : 0;
236 overflow_bytes = overflow_bytes + (n - w);
237 }
238 break;
239 }
240 case SPIREG_AUDIO_INFO:
241 if (len == SPIREG_AUDIO_INFO_SIZE) {
242 tinyspi::detail::Critical c;
243 spi.read(info_raw, SPIREG_AUDIO_INFO_SIZE);
244 pending_info = true;
245 len = 0;
246 }
247 break;
248 case SPIREG_MIME:
249 if (len > 0 && len <= SPIREG_MIME_SIZE) {
250 tinyspi::detail::Critical c;
251 memset(mime_raw, 0, sizeof(mime_raw));
252 spi.read((uint8_t*)mime_raw, len);
253 mime_raw[SPIREG_MIME_SIZE - 1] = 0;
254 pending_mime = true;
255 len = 0;
256 }
257 break;
258 default:
259 break;
260 }
261 while (len--) spi.read(); // discard what was not consumed
262 }
263
264 void stageReply(uint8_t addr, size_t want) {
265 uint8_t reg[SPIREG_MIME_SIZE];
266 size_t size = 0;
267 memset(reg, 0, sizeof(reg));
268 switch (addr) {
269 case SPIREG_AUDIO_INFO:
270 {
271 tinyspi::detail::Critical c;
272 memcpy(reg, info_raw, SPIREG_AUDIO_INFO_SIZE);
273 }
274 size = SPIREG_AUDIO_INFO_SIZE;
275 break;
276 case SPIREG_MIME:
277 {
278 tinyspi::detail::Critical c;
279 memcpy(reg, mime_raw, SPIREG_MIME_SIZE);
280 }
281 size = SPIREG_MIME_SIZE;
282 break;
283 case SPIREG_AVAILABLE: {
284 uint32_t space = p_buffer->availableForWrite();
285 reg[0] = space & 0xFF;
286 reg[1] = (space >> 8) & 0xFF;
287 reg[2] = (space >> 16) & 0xFF;
288 reg[3] = (space >> 24) & 0xFF;
289 size = SPIREG_AVAILABLE_SIZE;
290 break;
291 }
292 default:
293 break;
294 }
295 spi.flushTx();
296 if (want == 0) want = size;
297 size_t room = (size_t)spi.availableForWrite();
298 if (want > room) want = room;
299 for (size_t j = 0; j < want; j++) {
300 spi.write(j < size ? reg[j] : 0);
301 }
302 }
