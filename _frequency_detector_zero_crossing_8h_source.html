173 void detectChannel(int channel, T* samples, size_t len) {
174 ChannelState& state = states[channel];
175 for (int i = channel; i < (int)(len - info.channels); i += info.channels) {
176 if (state.active) state.count++;
177 // Detect upward zero crossing (negative to positive)
178 if (samples[i] <= 0 && samples[i + info.channels] > 0) {
179 if (state.count > 0) {
180 state.freq = (1.0f * info.sample_rate) / state.count;
181 if (freq_callback) freq_callback(channel, state.freq);
