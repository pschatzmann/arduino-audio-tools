138 // soft_float_t has both a converting constructor from float and a
139 // conversion operator to float, so `pseudoFloatValue * 0.7f` (and every
140 // other mixed op) would otherwise be ambiguous -- the compiler can't
141 // choose between converting the float to soft_float_t (to use the member
142 // operator above) or converting the soft_float_t to float (to use the
143 // built-in operator), and both need exactly one user-defined conversion.
144 // These exact-match overloads (no conversion needed for either operand)
145 // resolve that ambiguity, since the algorithm this class backs mixes
146 // float literals and soft_float_t variables constantly.

184 static int32_t clampRound(double v, double lo, double hi) {
185 if (v >= hi) return (int32_t)hi;

191 void fromFloat(float f) {
192 if (f == 0.0f) {
193 m_ = 0;
194 e_ = 0;
195 return;
196 }
197 bool neg = f < 0;
198 double mag = neg ? -(double)f : (double)f;
199 int32_t e = 0;
200 while (mag >= 32768.0) {
201 mag *= 0.5;
202 e++;
203 }
204 while (mag < 16384.0) {
205 mag *= 2.0;
206 e--;
207 }
208 int16_t mag16 = (int16_t)mag;
209 m_ = neg ? (int16_t)(-(int32_t)mag16) : mag16;
210 e_ = (int16_t)e;
211 }

213 float toFloat() const {
214 return (float)((double)m_ * pow(2.0, (double)e_));
215 }

219 static soft_float_t normalize(int32_t val, int32_t e) {
220 if (val == 0) return soft_float_t((int16_t)0, (int16_t)0);
221 bool neg = val < 0;
222 int64_t mag = neg ? -(int64_t)val : (int64_t)val;
223 while (mag > 32767) {
224 mag >>= 1;
225 e++;
226 }
227 while (mag < 16384) {
228 mag <<= 1;
229 e--;
230 }
231 int16_t mi = (int16_t)(neg ? -mag : mag);
232 return soft_float_t(mi, (int16_t)e);
233 }

237 static soft_float_t addAligned(int16_t ma, int16_t ea, int16_t mb,
238 int16_t eb) {
239 if (ma == 0) return soft_float_t(mb, eb);
240 if (mb == 0) return soft_float_t(ma, ea);
241 int32_t a = ma, b = mb;
242 int32_t e;
243 if (ea >= eb) {
244 int shift = ea - eb;
245 b = (shift >= 31) ? 0 : (b >> shift);
246 e = ea;
247 } else {
248 int shift = eb - ea;
249 a = (shift >= 31) ? 0 : (a >> shift);
250 e = eb;
251 }
252 return normalize(a + b, e);
253 }
