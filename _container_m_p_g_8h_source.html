37 public:
38 void resize(size_t size) { vec.resize(size + 4); }
39 size_t writeArray(const uint8_t *data, size_t len) {
40 size_t to_write = min(availableToWrite(), len);
41 memmove(vec.data() + count, data, to_write);
42 count += to_write;
43 return to_write;
44 }
45 void consume(size_t len) {
46 if (len > count) len = count;
47 memmove(vec.data(), vec.data() + len, count - len);
48 count -= len;
49 }
50 uint8_t *data() { return vec.data(); }
51 size_t available() { return count; }
52 size_t availableToWrite() { return vec.size() - count; }
53 void clear() { count = 0; }
54
56 long indexOfStartCode(size_t from = 0) {
57 if (count < 3 || from + 3 > count) return -1;
58 for (size_t i = from; i + 2 < count; i++) {
59 if (vec[i] == 0 && vec[i + 1] == 0 && vec[i + 2] == 1) return (long)i;
60 }
61 return -1;
62 }

242 // Accumulates one complete access unit (a whole picture, or a whole
243 // audio frame) across every PES fragment and every parse_buffer-sized
244 // write() chunk that contributes to it, so the decoder downstream gets
245 // one single write() call per unit instead of an arbitrary number of
246 // partial ones - see deliver()/flushVideoUnit()/flushAudioUnit(). A
247 // container-demuxed stream's chunk boundaries don't line up with
248 // picture/frame boundaries at all (the parse buffer above is a small,
249 // fixed size; a real access unit is routinely several times larger),
250 // and feeding a video decoder like TinyMPGDecoder a boundary-split
251 // buffer under its default whole-buffer contract can silently
252 // reconstruct a picture from stale/wrong macroblocks rather than
253 // failing cleanly - see that project's own TinyMPGDecoder::write() doc
254 // comment, which calls this exact demuxer out by name.
255 //
256 // Separate buffers for audio/video rather than one shared one: a real
257 // encoder (ffmpeg's MPEG-PS muxer among them, confirmed by testing
258 // against files it produced) routinely interleaves audio PES packets
259 // in the middle of a video picture's own PES fragments, before that
260 // picture's bytes are complete - unlike MuxerMPG (this project's own
261 // encoder), which happens to write one access unit's fragments back to
262 // back. A single shared buffer can't survive that: whichever unit was
263 // mid-accumulation would have to be flushed the moment the other type's
264 // bytes arrived, truncating it and permanently losing/misaligning
265 // whatever hadn't arrived yet - confirmed by testing (a truncated video
266 // unit's trailing bytes reappear, unprefixed by their own start code,
267 // corrupting the *next* unit too). Two independent buffers mean
268 // real interleaving no longer needs any special-casing at all - each
269 // side just keeps accumulating exactly where it left off.
270 //
271 // *_unit_len is tracked separately from each Vector's own size(), which
272 // is deliberately over-allocated for amortized growth - Vector::resize()
273 // always allocates exactly the requested size with no spare capacity
274 // (AudioBasic/Collections/Vector.h), so resizing it to the exact new
275 // length on every single delivered fragment - one picture can need a
276 // dozen of these - would reallocate and copy everything accumulated so
277 // far *every single call*, the same O(N^2) cost this project's
278 // ChunkedSampleTableStore (SampleTableStore.h) exists to avoid for
279 // MP4's sample tables.

304 bool parse() {
305 if (unit_open) return continuePayload();
306
307 if (parse_buffer.available() < 4) return false;
308 uint8_t *buf = parse_buffer.data();
309 if (!(buf[0] == 0 && buf[1] == 0 && buf[2] == 1)) {
310 long idx = parse_buffer.indexOfStartCode();
311 if (idx < 0) {
312 size_t avail = parse_buffer.available();
313 if (avail > 2) parse_buffer.consume(avail - 2);
314 return false;
315 }
316 parse_buffer.consume((size_t)idx);
317 if (parse_buffer.available() < 4) return false;
318 buf = parse_buffer.data();
319 }
320
321 uint8_t id = buf[3];
322 if (id == MPG_PACK_START_CODE) return parsePack();
323 if (id == MPG_SYSTEM_HEADER_START_CODE) return parseSystemHeader();
324 if (id == MPG_PROGRAM_END_CODE) {
325 parse_buffer.consume(4);
326 if (video_frame_open) flushVideoUnit();
327 if (audio_frame_open) flushAudioUnit();
328 video_frame_open = false;
329 audio_frame_open = false;
330 LOGI("MPEG_program_end_code");
331 return true;
332 }
333 if (id == MPG_PROGRAM_STREAM_MAP || id == MPG_PRIVATE_STREAM_1 ||
334 id == MPG_PADDING_STREAM || id == MPG_PRIVATE_STREAM_2)
335 return parseSkip();
336 if (id >= 0xC0 && id <= 0xDF) return parsePes(id, false);
337 if (id >= 0xE0 && id <= 0xEF) return parsePes(id, true);
338
339 // reserved/unrecognized id: drop this byte and resync on the next
340 // start code
341 parse_buffer.consume(1);
342 return true;
343 }

378 bool parsePes(uint8_t id, bool isVideo) {
379 // need the fixed 6-byte PES header plus 1 byte to peek the optional
380 // header's shape
381 if (parse_buffer.available() < 7) return false;
382 uint8_t *buf = parse_buffer.data();
383 uint16_t pes_len = ((uint16_t)buf[4] << 8) | buf[5];
384 uint8_t peek = buf[6];
385
386 size_t optional_len;
387 bool has_pts = false;
388 if (peek == 0x0F) {
389 optional_len = 1;
390 } else if ((peek >> 4) == 0x02) {
391 optional_len = 5;
392 has_pts = true;
393 } else if ((peek >> 4) == 0x03) {
394 optional_len = 10;
395 has_pts = true;
396 } else {
397 LOGW("DemuxerMPG: unsupported PES optional header 0x%02X - treating as no-timestamp",
398 (int)peek);
399 optional_len = 1;
400 }
401 if (parse_buffer.available() < 6 + optional_len) return false;
402
403 parse_buffer.consume(6 + optional_len);
404 (void)id;
405
406 bool is_unbounded = false;
407 size_t payload_len = 0;
408 if (pes_len == 0) {
409 // PES_packet_length == 0 ("unbounded") is only meaningful for video
410 is_unbounded = isVideo;
411 } else if (pes_len < optional_len) {
412 payload_len = 0; // malformed - be defensive rather than underflow
413 } else {
414 payload_len = (size_t)pes_len - optional_len;
415 }
416
417 if (!isVideo && has_pts) {
418 // a PTS marks the first fragment of a new audio access unit - close
419 // out the previous one (see audio_unit_buffer's comment)
420 if (audio_frame_open) flushAudioUnit();
421 audio_frame_open = true;
422 }
423
424 if (isVideo && has_pts) {
425 // a PTS marks the first fragment of a new access unit - close out
426 // the previous one
427 if (video_frame_open) flushVideoUnit();
428 video_frame_open = true;
429 // No wall-clock scheduling or skip decision here anymore - every
430 // picture is always decoded and forwarded immediately, in full.
431 // Pacing is the caller's responsibility now - see VideoAudioSyncTask
432 // (Video/VideoAudioSyncTask.h), which schedules frames from its own
433 // background task instead of blocking this dispatch loop (and, with
434 // it, audio delivery) the way an inline wait here used to.
435 }
436
437 unit_open = true;
438 is_skip_unit = false;
439 is_video_unit = isVideo;
440 unbounded = is_unbounded;
441 remaining = payload_len;
442 return continuePayload();
443 }

481 bool continuePayload() {
482 if (unbounded) {
483 long idx = indexOfContainerStartCode();
484 size_t avail = parse_buffer.available();
485 if (idx >= 0) {
486 deliver((size_t)idx);
487 parse_buffer.consume((size_t)idx);
488 unit_open = false;
489 return true;
490 }
491 if (avail > 2) {
492 deliver(avail - 2);
493 parse_buffer.consume(avail - 2);
494 }
495 return false;
496 }
497 size_t avail = parse_buffer.available();
498 size_t to_write = min(avail, remaining);
499 deliver(to_write);
500 parse_buffer.consume(to_write);
501 remaining -= to_write;
502 if (remaining == 0) {
503 unit_open = false;
504 return true;
505 }
506 return false;
507 }

570 void deliverVideo(const uint8_t *data, size_t len) {
571 size_t old_len = video_unit_len;
572 appendToVideo(data, len);
573 // 3-byte lookback so a start code split across the previously
574 // buffered tail and this call's newly appended bytes is still found -
575 // video_unit_buffer now holds both contiguously, no separate boundary
576 // handling needed.
577 size_t search_from = old_len > 3 ? old_len - 3 : 0;
578 uint8_t *buf = video_unit_buffer.data();
579 while (true) {
580 long idx = -1;
581 for (size_t i = search_from; i + 3 < video_unit_len; i++) {
582 if (buf[i] == 0 && buf[i + 1] == 0 && buf[i + 2] == 1 && buf[i + 3] == 0) {
583 idx = (long)i;
584 break;
585 }
586 }
587 if (idx < 0) break;
588 if (!video_unit_seen_picture) {
589 video_unit_seen_picture = true;
590 search_from = (size_t)idx + 4;
591 continue;
592 }
593 size_t remaining = video_unit_len - (size_t)idx;
594 video_unit_len = (size_t)idx;
595 flushVideoUnit(); // writes [0, idx), resets video_unit_len to 0
596 memmove(buf, buf + idx, remaining);
597 video_unit_len = remaining;
598 video_unit_seen_picture = true; // the retained remainder already starts with one
599 search_from = 4; // skip the start code now sitting at offset 0
600 }
601 }

634 void probeVideoHeader(const uint8_t *data, size_t len) {
635 size_t old_size = (size_t)video_probe.size();
636 size_t take = old_size < VIDEO_PROBE_MAX ? VIDEO_PROBE_MAX - old_size : 0;
637 if (take > len) take = len;
638 for (size_t i = 0; i < take; i++) {
639 uint8_t b = data[i];
640 video_probe.push_back(b);
641 }
642
643 size_t n = (size_t)video_probe.size();
644 for (size_t i = 0; i + 7 < n; i++) {
645 if (video_probe[i] == 0 && video_probe[i + 1] == 0 &&
646 video_probe[i + 2] == 1 && video_probe[i + 3] == 0xB3) {
647 uint8_t b0 = video_probe[i + 4], b1 = video_probe[i + 5],
648 b2 = video_probe[i + 6], b3 = video_probe[i + 7];
649 video_width = (uint16_t)(((uint16_t)b0 << 4) | (b1 >> 4));
650 video_height = (uint16_t)(((uint16_t)(b1 & 0x0F) << 8) | b2);
651 static const float rates[16] = {0, 23.976f, 24.0f, 25.0f,
652 29.97f, 30.0f, 50.0f, 59.94f,
653 60.0f, 0, 0, 0,
654 0, 0, 0, 0};
655 video_fps = rates[b3 & 0x0F];
656 video_header_parsed = true;
657 break;
658 }
659 }
660 if (video_header_parsed || n >= VIDEO_PROBE_MAX) {
661 video_probe.clear();
662 video_probe.shrink_to_fit();
663 }
664 }

668 void probeAudioHeader(const uint8_t *data, size_t len) {
669 size_t old_size = (size_t)audio_probe.size();
670 size_t take = old_size < 4 ? 4 - old_size : 0;
671 if (take > len) take = len;
672 for (size_t i = 0; i < take; i++) {
673 uint8_t b = data[i];
674 audio_probe.push_back(b);
675 }
676 if ((size_t)audio_probe.size() < 4) return;
677
678 uint8_t b0 = audio_probe[0], b1 = audio_probe[1], b2 = audio_probe[2],
679 b3 = audio_probe[3];
680 if (b0 == 0xFF && (b1 & 0xE0) == 0xE0) {
681 uint8_t version = (b1 >> 3) & 0x03; // 11=MPEG1, 10=MPEG2, 00=MPEG2.5
682 uint8_t layer_bits = (b1 >> 1) & 0x03; // 01=Layer III, 10=Layer II, 11=Layer I
683 uint8_t sr_index = (b2 >> 2) & 0x03;
684 uint8_t mode = (b3 >> 6) & 0x03; // 11=mono
685 static const uint32_t rates_v1[4] = {44100, 48000, 32000, 0};
686 static const uint32_t rates_v2[4] = {22050, 24000, 16000, 0};
687 static const uint32_t rates_v25[4] = {11025, 12000, 8000, 0};
688 uint32_t rate = 0;
689 if (version == 0x03)
690 rate = rates_v1[sr_index];
691 else if (version == 0x02)
692 rate = rates_v2[sr_index];
693 else if (version == 0x00)
694 rate = rates_v25[sr_index];
695 if (rate > 0) {
696 audio_sample_rate = rate;
697 audio_channels = (mode == 0x03) ? 1 : 2;
698 // layer_bits -> standard layer number (I/II/III), 0 if reserved
699 static const uint8_t layers[4] = {0, 3, 2, 1};
700 audio_layer = layers[layer_bits];
701 audio_header_parsed = true;
702 }
703 } else {
704 // not a frame sync where we expected one - stop probing so we don't
705 // keep re-checking every write(); getAudioInfo() falls back to
706 // 44100/stereo
707 audio_header_parsed = true;
708 }
709 audio_probe.clear();
710 audio_probe.shrink_to_fit();
711 }

934 uint64_t clampScr(uint64_t scr) {
935 if (scr < last_scr) scr = last_scr;
936 last_scr = scr;
937 return scr;

939
940 void writePackHeader(uint64_t scr_in) {
941 uint64_t scr = clampScr(scr_in);
942 static const uint8_t start[4] = {0x00, 0x00, 0x01, MPG_PACK_START_CODE};
943 p_out->write(start, 4);
944 uint8_t b[8];
945 b[0] = (uint8_t)(0x20 | (((scr >> 30) & 0x07) << 1) | 0x01);
946 b[1] = (uint8_t)((scr >> 22) & 0xFF);
947 b[2] = (uint8_t)((((scr >> 15) & 0x7F) << 1) | 0x01);
948 b[3] = (uint8_t)((scr >> 7) & 0xFF);
949 b[4] = (uint8_t)(((scr & 0x7F) << 1) | 0x01);
950 b[5] = (uint8_t)(0x80 | ((MUX_RATE >> 15) & 0x7F));
951 b[6] = (uint8_t)((MUX_RATE >> 7) & 0xFF);
952 b[7] = (uint8_t)(((MUX_RATE & 0x7F) << 1) | 0x01);
953 p_out->write(b, 8);
954 }
955
956 void writeStreamEntry(uint8_t id, bool isVideo) {
957 // P-STD buffer bound: informational only - 46 (x1024 = ~46KB) for
958 // video, 4 (x128 = 512 bytes, ~1 audio frame) for audio
959 uint16_t bound = isVideo ? 46 : 4;
960 uint8_t b[3];
961 b[0] = id;
962 b[1] = (uint8_t)(0xC0 | (isVideo ? 0x20 : 0x00) | ((bound >> 8) & 0x1F));
963 b[2] = (uint8_t)(bound & 0xFF);
964 p_out->write(b, 3);
965 }
966
967 void writeSystemHeader() {
968 int num_streams = 1 + (has_audio ? 1 : 0);
969 uint16_t header_length = (uint16_t)(6 + 3 * num_streams);
970 static const uint8_t start[4] = {0x00, 0x00, 0x01,
971 MPG_SYSTEM_HEADER_START_CODE};
972 p_out->write(start, 4);
973 uint8_t len_b[2] = {(uint8_t)(header_length >> 8),
974 (uint8_t)header_length};
975 p_out->write(len_b, 2);
976
977 uint16_t audio_bound = has_audio ? 1 : 0;
978 uint16_t video_bound = 1;
979 uint8_t b[6];
980 // marker(1) + mux_rate(22) + marker(1)
981 b[0] = (uint8_t)(0x80 | ((MUX_RATE >> 15) & 0x7F));
982 b[1] = (uint8_t)((MUX_RATE >> 7) & 0xFF);
983 b[2] = (uint8_t)(((MUX_RATE & 0x7F) << 1) | 0x01);
984 // audio_bound(6) + fixed_flag(1)=0 + CSPS_flag(1)=0
985 b[3] = (uint8_t)((audio_bound & 0x3F) << 2);
986 // audio_locked(1)=0 + video_locked(1)=0 + marker(1)=1 + video_bound(5)
987 b[4] = (uint8_t)(0x20 | (video_bound & 0x1F));
988 b[5] = 0xFF; // reserved_byte
989 p_out->write(b, 6);
990
991 writeStreamEntry(MPG_VIDEO_STREAM_ID, true);
992 if (has_audio) writeStreamEntry(MPG_AUDIO_STREAM_ID, false);
993 }
994
995 void writeTimestampField(uint8_t id4, uint64_t ts) {
996 uint8_t b[5];
997 b[0] = (uint8_t)((id4 << 4) | (((ts >> 30) & 0x07) << 1) | 0x01);
998 uint16_t mid = (uint16_t)((((ts >> 15) & 0x7FFF) << 1) | 0x01);
999 uint16_t low = (uint16_t)(((ts & 0x7FFF) << 1) | 0x01);
1000 b[1] = (uint8_t)(mid >> 8);
1001 b[2] = (uint8_t)mid;
1002 b[3] = (uint8_t)(low >> 8);
1003 b[4] = (uint8_t)low;
1004 p_out->write(b, 5);
1005 }

1008 size_t writeElementaryChunk(uint8_t stream_id, const uint8_t *data,
1009 size_t len, bool withPts, uint64_t pts) {
1010 writePackHeader(pts);
1011 uint16_t optional_len = withPts ? 5 : 1;
1012 uint16_t pes_len = (uint16_t)(len + optional_len);
1013 uint8_t hdr[6] = {0,
1014 0,
1015 1,
1016 stream_id,
1017 (uint8_t)(pes_len >> 8),
1018 (uint8_t)pes_len};
1019 p_out->write(hdr, 6);
1020 if (withPts) {
1021 writeTimestampField(0x02, pts);
1022 } else {
1023 // avoid Print::write(uint8_t): some Print/Stream implementations
1024 // (e.g. BaseStream-derived ones) buffer single-byte writes
1025 // separately from the write(data, len) path below, which would
1026 // reorder this marker byte after the payload - write it as a
1027 // one-element array instead so it goes through the same path.
1028 static const uint8_t no_ts = 0x0F;
1029 p_out->write(&no_ts, 1);
1030 }
1031 return p_out->write(data, len);
1032 }
1033
1037 size_t writeAccessUnit(uint8_t stream_id, const uint8_t *data, size_t len,
1038 uint64_t pts) {
1039 size_t offset = 0;
1040 size_t written = 0;
1041 bool first = true;
1042 do {
1043 size_t chunk = len - offset;
1044 if (chunk > MAX_PES_PAYLOAD) chunk = MAX_PES_PAYLOAD;
1045 written +=
1046 writeElementaryChunk(stream_id, data + offset, chunk, first, pts);
1047 offset += chunk;
1048 first = false;
1049 } while (offset < len);
1050 return written;
1051 }
1052};
