37class MPGParseBuffer {
38 public:
39 void resize(size_t size) { vec.resize(size + 4); }
40 size_t writeArray(const uint8_t *data, size_t len) {
41 size_t to_write = min(availableToWrite(), len);
42 memmove(vec.data() + count, data, to_write);
43 count += to_write;
44 return to_write;
45 }
46 void consume(size_t len) {
47 if (len > count) len = count;
48 memmove(vec.data(), vec.data() + len, count - len);
49 count -= len;
50 }
51 uint8_t *data() { return vec.data(); }
52 size_t available() { return count; }
53 size_t availableToWrite() { return vec.size() - count; }
54 void clear() { count = 0; }
55
57 long indexOfStartCode(size_t from = 0) {
58 if (count < 3 || from + 3 > count) return -1;
59 for (size_t i = from; i + 2 < count; i++) {
60 if (vec[i] == 0 && vec[i + 1] == 0 && vec[i + 2] == 1) return (long)i;
61 }
62 return -1;
63 }

65 protected:
66 Vector<uint8_t> vec{0};
67 size_t count = 0;

250 // Accumulates one complete access unit (a whole picture, or a whole
251 // audio frame) across every PES fragment and every parse_buffer-sized
252 // write() chunk that contributes to it, so the decoder downstream gets
253 // one single write() call per unit instead of an arbitrary number of
254 // partial ones - see deliver()/flushVideoUnit()/flushAudioUnit(). A
255 // container-demuxed stream's chunk boundaries don't line up with
256 // picture/frame boundaries at all (the parse buffer above is a small,
257 // fixed size; a real access unit is routinely several times larger),
258 // and feeding a video decoder like TinyMPGDecoder a boundary-split
259 // buffer under its default whole-buffer contract can silently
260 // reconstruct a picture from stale/wrong macroblocks rather than
261 // failing cleanly - see that project's own TinyMPGDecoder::write() doc
262 // comment, which calls this exact demuxer out by name.
263 //
264 // One buffer shared between audio and video rather than one each: a
265 // muxer (see MuxerMPG) writes one access unit's PES fragments back to
266 // back, so in practice audio and video are never both mid-unit at the
267 // same time - one buffer, sized to whichever unit turns out biggest
268 // (routinely a video picture, far larger than one audio frame), covers
269 // both. unit_holds_video says which type currently owns its contents
270 // (only meaningful while unit_len > 0); if some other encoder's output
271 // ever *did* interleave the two mid-unit, deliver() flushes whichever
272 // unit is already in the buffer first rather than letting the new
273 // bytes corrupt it - a premature flush of that one unit, not silent
274 // corruption.
275 //
276 // unit_len is tracked separately from the Vector's own size(), which is
277 // deliberately over-allocated for amortized growth - Vector::resize()
278 // always allocates exactly the requested size with no spare capacity
279 // (AudioBasic/Collections/Vector.h), so resizing it to the exact new
280 // length on every single delivered fragment - one picture can need a
281 // dozen of these - would reallocate and copy everything accumulated so
282 // far *every single call*, the same O(N^2) cost this project's
283 // ChunkedSampleTableStore (SampleTableStore.h) exists to avoid for
284 // MP4's sample tables.

312 bool parse() {
313 if (unit_open) return continuePayload();
314
315 if (parse_buffer.available() < 4) return false;
316 uint8_t *buf = parse_buffer.data();
317 if (!(buf[0] == 0 && buf[1] == 0 && buf[2] == 1)) {
318 long idx = parse_buffer.indexOfStartCode();
319 if (idx < 0) {
320 size_t avail = parse_buffer.available();
321 if (avail > 2) parse_buffer.consume(avail - 2);
322 return false;
323 }
324 parse_buffer.consume((size_t)idx);
325 if (parse_buffer.available() < 4) return false;
326 buf = parse_buffer.data();
327 }
328
329 uint8_t id = buf[3];
330 if (id == MPG_PACK_START_CODE) return parsePack();
331 if (id == MPG_SYSTEM_HEADER_START_CODE) return parseSystemHeader();
332 if (id == MPG_PROGRAM_END_CODE) {
333 parse_buffer.consume(4);
334 if (video_frame_open) flushVideoUnit();
335 if (audio_frame_open) flushAudioUnit();
336 video_frame_open = false;
337 audio_frame_open = false;
338 LOGI("MPEG_program_end_code");
339 return true;
340 }
341 if (id == MPG_PROGRAM_STREAM_MAP || id == MPG_PRIVATE_STREAM_1 ||
342 id == MPG_PADDING_STREAM || id == MPG_PRIVATE_STREAM_2)
343 return parseSkip();
344 if (id >= 0xC0 && id <= 0xDF) return parsePes(id, false);
345 if (id >= 0xE0 && id <= 0xEF) return parsePes(id, true);
346
347 // reserved/unrecognized id: drop this byte and resync on the next
348 // start code
349 parse_buffer.consume(1);
350 return true;
351 }

386 bool parsePes(uint8_t id, bool isVideo) {
387 // need the fixed 6-byte PES header plus 1 byte to peek the optional
388 // header's shape
389 if (parse_buffer.available() < 7) return false;
390 uint8_t *buf = parse_buffer.data();
391 uint16_t pes_len = ((uint16_t)buf[4] << 8) | buf[5];
392 uint8_t peek = buf[6];
393
394 size_t optional_len;
395 bool has_pts = false;
396 if (peek == 0x0F) {
397 optional_len = 1;
398 } else if ((peek >> 4) == 0x02) {
399 optional_len = 5;
400 has_pts = true;
401 } else if ((peek >> 4) == 0x03) {
402 optional_len = 10;
403 has_pts = true;
404 } else {
405 LOGW("DemuxerMPG: unsupported PES optional header 0x%02X - treating as no-timestamp",
406 (int)peek);
407 optional_len = 1;
408 }
409 if (parse_buffer.available() < 6 + optional_len) return false;
410
411 uint64_t pts = 0;
412 if (has_pts) pts = readTimestamp(buf + 6);
413 parse_buffer.consume(6 + optional_len);
414 (void)id;
415
416 bool is_unbounded = false;
417 size_t payload_len = 0;
418 if (pes_len == 0) {
419 // PES_packet_length == 0 ("unbounded") is only meaningful for video
420 is_unbounded = isVideo;
421 } else if (pes_len < optional_len) {
422 payload_len = 0; // malformed - be defensive rather than underflow
423 } else {
424 payload_len = (size_t)pes_len - optional_len;
425 }
426
427 if (!isVideo && has_pts) {
428 // a PTS marks the first fragment of a new audio access unit - close
429 // out the previous one (see audio_unit_buffer's comment)
430 if (audio_frame_open) flushAudioUnit();
431 audio_frame_open = true;
432 }
433
434 bool skip_late_picture = false;
435 if (isVideo && has_pts) {
436 // a PTS marks the first fragment of a new access unit - close out
437 // the previous one
438 if (video_frame_open) flushVideoUnit();
439 video_frame_open = true;
440
441 // Pace this new picture against wall-clock time using its PTS (a
442 // MPG_CLOCK_HZ=90kHz tick count) - mirrors DemuxerMP4::dispatchVideo()'s
443 // schedule-anchored wait.
444 uint32_t scheduledMs = (uint32_t)(pts * 1000 / MPG_CLOCK_HZ);
445 uint32_t nowMs = millis();
446 if (!playback_start_set) {
447 playback_start_ms = nowMs;
448 playback_start_set = true;
449 }
450 uint32_t scheduledWallMs = playback_start_ms + scheduledMs;
451 if (nowMs > scheduledWallMs) {
452 // Only skip if this picture is never used as a reference (a
453 // B-picture, coding_type==3) - see peekPictureCodingType(). An
454 // I/P picture (or a coding type we can't determine yet) is
455 // decoded immediately instead, no wait either since we're
456 // already behind.
457 int codingType = peekPictureCodingType(parse_buffer.data(),
458 parse_buffer.available());
459 if (codingType == 3) {
460 LOGW("DemuxerMPG: video picture %u ms late - skipping (B-picture)",
461 (unsigned)(nowMs - scheduledWallMs));
462 skip_late_picture = true;
463 } else {
464 LOGW("DemuxerMPG: video picture %u ms late - decoding anyway (%s)",
465 (unsigned)(nowMs - scheduledWallMs),
466 codingType < 0 ? "coding type unknown" : "referenced by later pictures");
467 }
468 } else if (nowMs < scheduledWallMs) {
469 uint32_t waitMs = scheduledWallMs - nowMs;
470 p_synch->delayVideoFrame((int32_t)(std::min(waitMs, 200u) * 1000), 0);
471 }
472 }
473
474 unit_open = true;
475 is_skip_unit = skip_late_picture;
476 is_video_unit = isVideo;
477 unbounded = is_unbounded;
478 remaining = payload_len;
479 return continuePayload();
480 }

482 bool continuePayload() {
483 if (unbounded) {
484 long idx = parse_buffer.indexOfStartCode();
485 size_t avail = parse_buffer.available();
486 if (idx >= 0) {
487 deliver((size_t)idx);
488 parse_buffer.consume((size_t)idx);
489 unit_open = false;
490 return true;
491 }
492 if (avail > 2) {
493 deliver(avail - 2);
494 parse_buffer.consume(avail - 2);
495 }
496 return false;
497 }
498 size_t avail = parse_buffer.available();
499 size_t to_write = min(avail, remaining);
500 deliver(to_write);
501 parse_buffer.consume(to_write);
502 remaining -= to_write;
503 if (remaining == 0) {
504 unit_open = false;
505 return true;
506 }
507 return false;
508 }

514 void deliver(size_t len) {
515 if (len == 0 || is_skip_unit) return;
516 uint8_t *data = parse_buffer.data();
517 // Defensive: the buffer already holds an in-progress unit of the
518 // *other* type - real interleaving, not expected from MuxerMPG's own
519 // output but not this demuxer's to assume of every encoder. Flush it
520 // now rather than let these bytes corrupt it.
521 if (unit_len > 0 && unit_holds_video != is_video_unit) {
522 if (unit_holds_video) flushVideoUnit(); else flushAudioUnit();
523 }
524 if (is_video_unit) {
525 if (!video_header_parsed) probeVideoHeader(data, len);
526 } else {
527 if (!audio_header_parsed) probeAudioHeader(data, len);
528 }
529 unit_holds_video = is_video_unit;
530 appendTo(data, len);
531 }
532
535 void appendTo(const uint8_t *data, size_t len) {
536 size_t needed = unit_len + len;
537 if (needed > (size_t)unit_buffer.size()) {
538 size_t newCap = unit_buffer.size() == 0 ? 8192 : (size_t)unit_buffer.size() * 2;
539 if (newCap < needed) newCap = needed;
540 unit_buffer.resize(newCap);
541 }
542 memcpy(unit_buffer.data() + unit_len, data, len);
543 unit_len += len;
544 }

579 static uint64_t readTimestamp(const uint8_t *p) {
580 uint64_t ts = 0;
581 ts |= ((uint64_t)(p[0] >> 1) & 0x07) << 30;
582 ts |= ((uint64_t)((((uint16_t)p[1] << 8) | p[2]) >> 1) & 0x7FFF) << 15;
583 ts |= (uint64_t)((((uint16_t)p[3] << 8) | p[4]) >> 1) & 0x7FFF;
584 return ts;
585 }

597 static int peekPictureCodingType(const uint8_t *data, size_t len) {
598 for (size_t i = 0; i + 6 <= len; i++) {
599 if (data[i] == 0 && data[i + 1] == 0 && data[i + 2] == 1 &&
600 data[i + 3] == 0x00) {
601 return (data[i + 5] >> 3) & 0x07;

611 void probeVideoHeader(const uint8_t *data, size_t len) {
612 size_t old_size = (size_t)video_probe.size();
613 size_t take = old_size < VIDEO_PROBE_MAX ? VIDEO_PROBE_MAX - old_size : 0;
614 if (take > len) take = len;
615 for (size_t i = 0; i < take; i++) {
616 uint8_t b = data[i];
617 video_probe.push_back(b);
618 }
619
620 size_t n = (size_t)video_probe.size();
621 for (size_t i = 0; i + 7 < n; i++) {
622 if (video_probe[i] == 0 && video_probe[i + 1] == 0 &&
623 video_probe[i + 2] == 1 && video_probe[i + 3] == 0xB3) {
624 uint8_t b0 = video_probe[i + 4], b1 = video_probe[i + 5],
625 b2 = video_probe[i + 6], b3 = video_probe[i + 7];
626 video_width = (uint16_t)(((uint16_t)b0 << 4) | (b1 >> 4));
627 video_height = (uint16_t)(((uint16_t)(b1 & 0x0F) << 8) | b2);
628 static const float rates[16] = {0, 23.976f, 24.0f, 25.0f,
629 29.97f, 30.0f, 50.0f, 59.94f,
630 60.0f, 0, 0, 0,
631 0, 0, 0, 0};
632 video_fps = rates[b3 & 0x0F];
633 video_header_parsed = true;
634 break;
635 }
636 }
637 if (video_header_parsed || n >= VIDEO_PROBE_MAX) {
638 video_probe.clear();
639 video_probe.shrink_to_fit();
640 }
641 }

645 void probeAudioHeader(const uint8_t *data, size_t len) {
646 size_t old_size = (size_t)audio_probe.size();
647 size_t take = old_size < 4 ? 4 - old_size : 0;
648 if (take > len) take = len;
649 for (size_t i = 0; i < take; i++) {
650 uint8_t b = data[i];
651 audio_probe.push_back(b);
652 }
653 if ((size_t)audio_probe.size() < 4) return;
654
655 uint8_t b0 = audio_probe[0], b1 = audio_probe[1], b2 = audio_probe[2],
656 b3 = audio_probe[3];
657 if (b0 == 0xFF && (b1 & 0xE0) == 0xE0) {
658 uint8_t version = (b1 >> 3) & 0x03; // 11=MPEG1, 10=MPEG2, 00=MPEG2.5
659 uint8_t layer_bits = (b1 >> 1) & 0x03; // 01=Layer III, 10=Layer II, 11=Layer I
660 uint8_t sr_index = (b2 >> 2) & 0x03;
661 uint8_t mode = (b3 >> 6) & 0x03; // 11=mono
662 static const uint32_t rates_v1[4] = {44100, 48000, 32000, 0};
663 static const uint32_t rates_v2[4] = {22050, 24000, 16000, 0};
664 static const uint32_t rates_v25[4] = {11025, 12000, 8000, 0};
665 uint32_t rate = 0;
666 if (version == 0x03)
667 rate = rates_v1[sr_index];
668 else if (version == 0x02)
669 rate = rates_v2[sr_index];
670 else if (version == 0x00)
671 rate = rates_v25[sr_index];
672 if (rate > 0) {
673 audio_sample_rate = rate;
674 audio_channels = (mode == 0x03) ? 1 : 2;
675 // layer_bits -> standard layer number (I/II/III), 0 if reserved
676 static const uint8_t layers[4] = {0, 3, 2, 1};
677 audio_layer = layers[layer_bits];
678 audio_header_parsed = true;
679 }
680 } else {
681 // not a frame sync where we expected one - stop probing so we don't
682 // keep re-checking every write(); getAudioInfo() falls back to
683 // 44100/stereo
684 audio_header_parsed = true;
685 }
686 audio_probe.clear();
687 audio_probe.shrink_to_fit();
688 }

917 void writePackHeader(uint64_t scr_in) {
918 uint64_t scr = clampScr(scr_in);
919 static const uint8_t start[4] = {0x00, 0x00, 0x01, MPG_PACK_START_CODE};
920 p_out->write(start, 4);
921 uint8_t b[8];
922 b[0] = (uint8_t)(0x20 | (((scr >> 30) & 0x07) << 1) | 0x01);
923 b[1] = (uint8_t)((scr >> 22) & 0xFF);
924 b[2] = (uint8_t)((((scr >> 15) & 0x7F) << 1) | 0x01);
925 b[3] = (uint8_t)((scr >> 7) & 0xFF);
926 b[4] = (uint8_t)(((scr & 0x7F) << 1) | 0x01);
927 b[5] = (uint8_t)(0x80 | ((MUX_RATE >> 15) & 0x7F));
928 b[6] = (uint8_t)((MUX_RATE >> 7) & 0xFF);
929 b[7] = (uint8_t)(((MUX_RATE & 0x7F) << 1) | 0x01);
930 p_out->write(b, 8);
931 }

933 void writeStreamEntry(uint8_t id, bool isVideo) {
934 // P-STD buffer bound: informational only - 46 (x1024 = ~46KB) for
935 // video, 4 (x128 = 512 bytes, ~1 audio frame) for audio
936 uint16_t bound = isVideo ? 46 : 4;
937 uint8_t b[3];
938 b[0] = id;
939 b[1] = (uint8_t)(0xC0 | (isVideo ? 0x20 : 0x00) | ((bound >> 8) & 0x1F));
940 b[2] = (uint8_t)(bound & 0xFF);
941 p_out->write(b, 3);
942 }

944 void writeSystemHeader() {
945 int num_streams = 1 + (has_audio ? 1 : 0);
946 uint16_t header_length = (uint16_t)(6 + 3 * num_streams);
947 static const uint8_t start[4] = {0x00, 0x00, 0x01,
948 MPG_SYSTEM_HEADER_START_CODE};
949 p_out->write(start, 4);
950 uint8_t len_b[2] = {(uint8_t)(header_length >> 8),
951 (uint8_t)header_length};
952 p_out->write(len_b, 2);
953
954 uint16_t audio_bound = has_audio ? 1 : 0;
955 uint16_t video_bound = 1;
956 uint8_t b[6];
957 // marker(1) + mux_rate(22) + marker(1)
958 b[0] = (uint8_t)(0x80 | ((MUX_RATE >> 15) & 0x7F));
959 b[1] = (uint8_t)((MUX_RATE >> 7) & 0xFF);
960 b[2] = (uint8_t)(((MUX_RATE & 0x7F) << 1) | 0x01);
961 // audio_bound(6) + fixed_flag(1)=0 + CSPS_flag(1)=0
962 b[3] = (uint8_t)((audio_bound & 0x3F) << 2);
963 // audio_locked(1)=0 + video_locked(1)=0 + marker(1)=1 + video_bound(5)
964 b[4] = (uint8_t)(0x20 | (video_bound & 0x1F));
965 b[5] = 0xFF; // reserved_byte
966 p_out->write(b, 6);
967
968 writeStreamEntry(MPG_VIDEO_STREAM_ID, true);
969 if (has_audio) writeStreamEntry(MPG_AUDIO_STREAM_ID, false);
970 }

972 void writeTimestampField(uint8_t id4, uint64_t ts) {
973 uint8_t b[5];
974 b[0] = (uint8_t)((id4 << 4) | (((ts >> 30) & 0x07) << 1) | 0x01);
975 uint16_t mid = (uint16_t)((((ts >> 15) & 0x7FFF) << 1) | 0x01);
976 uint16_t low = (uint16_t)(((ts & 0x7FFF) << 1) | 0x01);
977 b[1] = (uint8_t)(mid >> 8);
978 b[2] = (uint8_t)mid;
979 b[3] = (uint8_t)(low >> 8);
980 b[4] = (uint8_t)low;
981 p_out->write(b, 5);
982 }

985 size_t writeElementaryChunk(uint8_t stream_id, const uint8_t *data,
986 size_t len, bool withPts, uint64_t pts) {
987 writePackHeader(pts);
988 uint16_t optional_len = withPts ? 5 : 1;
989 uint16_t pes_len = (uint16_t)(len + optional_len);
990 uint8_t hdr[6] = {0,
991 0,
992 1,
993 stream_id,
994 (uint8_t)(pes_len >> 8),
995 (uint8_t)pes_len};
996 p_out->write(hdr, 6);
997 if (withPts) {
998 writeTimestampField(0x02, pts);
999 } else {
1000 // avoid Print::write(uint8_t): some Print/Stream implementations
1001 // (e.g. BaseStream-derived ones) buffer single-byte writes
1002 // separately from the write(data, len) path below, which would
1003 // reorder this marker byte after the payload - write it as a
1004 // one-element array instead so it goes through the same path.
1005 static const uint8_t no_ts = 0x0F;
1006 p_out->write(&no_ts, 1);
1007 }
1008 return p_out->write(data, len);
1009 }
1010
1014 size_t writeAccessUnit(uint8_t stream_id, const uint8_t *data, size_t len,
1015 uint64_t pts) {
1016 size_t offset = 0;
1017 size_t written = 0;
1018 bool first = true;
1019 do {
1020 size_t chunk = len - offset;
1021 if (chunk > MAX_PES_PAYLOAD) chunk = MAX_PES_PAYLOAD;
1022 written +=
1023 writeElementaryChunk(stream_id, data + offset, chunk, first, pts);
1024 offset += chunk;
1025 first = false;
1026 } while (offset < len);
1027 return written;
1028 }
