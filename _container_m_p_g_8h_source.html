36class MPGParseBuffer {
37 public:
38 void resize(size_t size) { vec.resize(size + 4); }
39 size_t writeArray(const uint8_t *data, size_t len) {
40 size_t to_write = min(availableToWrite(), len);
41 memmove(vec.data() + count, data, to_write);
42 count += to_write;
43 return to_write;
44 }
45 void consume(size_t len) {
46 if (len > count) len = count;
47 memmove(vec.data(), vec.data() + len, count - len);
48 count -= len;
49 }
50 uint8_t *data() { return vec.data(); }
51 size_t available() { return count; }
52 size_t availableToWrite() { return vec.size() - count; }
53 void clear() { count = 0; }
54
56 long indexOfStartCode(size_t from = 0) {
57 if (count < 3 || from + 3 > count) return -1;
58 for (size_t i = from; i + 2 < count; i++) {
59 if (vec[i] == 0 && vec[i + 1] == 0 && vec[i + 2] == 1) return (long)i;
60 }
61 return -1;
62 }

64 protected:
65 Vector<uint8_t> vec{0};
66 size_t count = 0;

234 bool parse() {
235 if (unit_open) return continuePayload();
236
237 if (parse_buffer.available() < 4) return false;
238 uint8_t *buf = parse_buffer.data();
239 if (!(buf[0] == 0 && buf[1] == 0 && buf[2] == 1)) {
240 long idx = parse_buffer.indexOfStartCode();
241 if (idx < 0) {
242 size_t avail = parse_buffer.available();
243 if (avail > 2) parse_buffer.consume(avail - 2);
244 return false;
245 }
246 parse_buffer.consume((size_t)idx);
247 if (parse_buffer.available() < 4) return false;
248 buf = parse_buffer.data();
249 }
250
251 uint8_t id = buf[3];
252 if (id == MPG_PACK_START_CODE) return parsePack();
253 if (id == MPG_SYSTEM_HEADER_START_CODE) return parseSystemHeader();
254 if (id == MPG_PROGRAM_END_CODE) {
255 parse_buffer.consume(4);
256 if (video_frame_open && p_output_video != nullptr)
257 p_output_video->flush();
258 video_frame_open = false;
259 LOGI("MPEG_program_end_code");
260 return true;
261 }
262 if (id == MPG_PROGRAM_STREAM_MAP || id == MPG_PRIVATE_STREAM_1 ||
263 id == MPG_PADDING_STREAM || id == MPG_PRIVATE_STREAM_2)
264 return parseSkip();
265 if (id >= 0xC0 && id <= 0xDF) return parsePes(id, false);
266 if (id >= 0xE0 && id <= 0xEF) return parsePes(id, true);
267
268 // reserved/unrecognized id: drop this byte and resync on the next
269 // start code
270 parse_buffer.consume(1);
271 return true;
272 }

307 bool parsePes(uint8_t id, bool isVideo) {
308 // need the fixed 6-byte PES header plus 1 byte to peek the optional
309 // header's shape
310 if (parse_buffer.available() < 7) return false;
311 uint8_t *buf = parse_buffer.data();
312 uint16_t pes_len = ((uint16_t)buf[4] << 8) | buf[5];
313 uint8_t peek = buf[6];
314
315 size_t optional_len;
316 bool has_pts = false;
317 if (peek == 0x0F) {
318 optional_len = 1;
319 } else if ((peek >> 4) == 0x02) {
320 optional_len = 5;
321 has_pts = true;
322 } else if ((peek >> 4) == 0x03) {
323 optional_len = 10;
324 has_pts = true;
325 } else {
326 LOGW("DemuxerMPG: unsupported PES optional header 0x%02X - treating as no-timestamp",
327 (int)peek);
328 optional_len = 1;
329 }
330 if (parse_buffer.available() < 6 + optional_len) return false;
331
332 uint64_t pts = 0;
333 if (has_pts) pts = readTimestamp(buf + 6);
334 parse_buffer.consume(6 + optional_len);
335 (void)id;
336 (void)pts; // parsed for spec-completeness; not currently exposed
337
338 bool is_unbounded = false;
339 size_t payload_len = 0;
340 if (pes_len == 0) {
341 // PES_packet_length == 0 ("unbounded") is only meaningful for video
342 is_unbounded = isVideo;
343 } else if (pes_len < optional_len) {
344 payload_len = 0; // malformed - be defensive rather than underflow
345 } else {
346 payload_len = (size_t)pes_len - optional_len;
347 }
348
349 if (isVideo && has_pts) {
350 // a PTS marks the first fragment of a new access unit - close out
351 // the previous one
352 if (video_frame_open && p_output_video != nullptr)
353 p_output_video->flush();
354 video_frame_open = true;
355 }
356
357 unit_open = true;
358 is_skip_unit = false;
359 is_video_unit = isVideo;
360 unbounded = is_unbounded;
361 remaining = payload_len;
362 return continuePayload();
363 }

365 bool continuePayload() {
366 if (unbounded) {
367 long idx = parse_buffer.indexOfStartCode();
368 size_t avail = parse_buffer.available();
369 if (idx >= 0) {
370 deliver((size_t)idx);
371 parse_buffer.consume((size_t)idx);
372 unit_open = false;
373 return true;
374 }
375 if (avail > 2) {
376 deliver(avail - 2);
377 parse_buffer.consume(avail - 2);
378 }
379 return false;
380 }
381 size_t avail = parse_buffer.available();
382 size_t to_write = min(avail, remaining);
383 deliver(to_write);
384 parse_buffer.consume(to_write);
385 remaining -= to_write;
386 if (remaining == 0) {
387 unit_open = false;
388 return true;
389 }
390 return false;
391 }

398 if (p_output_video != nullptr) p_output_video->write(data, len);
399 } else {

401 if (p_output_audio != nullptr) p_output_audio->write(data, len);
402 }

408 static uint64_t readTimestamp(const uint8_t *p) {
409 uint64_t ts = 0;
410 ts |= ((uint64_t)(p[0] >> 1) & 0x07) << 30;
411 ts |= ((uint64_t)((((uint16_t)p[1] << 8) | p[2]) >> 1) & 0x7FFF) << 15;
412 ts |= (uint64_t)((((uint16_t)p[3] << 8) | p[4]) >> 1) & 0x7FFF;
413 return ts;
414 }

420 void probeVideoHeader(const uint8_t *data, size_t len) {
421 size_t old_size = (size_t)video_probe.size();
422 size_t take = old_size < VIDEO_PROBE_MAX ? VIDEO_PROBE_MAX - old_size : 0;
423 if (take > len) take = len;
424 for (size_t i = 0; i < take; i++) {
425 uint8_t b = data[i];
426 video_probe.push_back(b);
427 }
428
429 size_t n = (size_t)video_probe.size();
430 for (size_t i = 0; i + 7 < n; i++) {
431 if (video_probe[i] == 0 && video_probe[i + 1] == 0 &&
432 video_probe[i + 2] == 1 && video_probe[i + 3] == 0xB3) {
433 uint8_t b0 = video_probe[i + 4], b1 = video_probe[i + 5],
434 b2 = video_probe[i + 6], b3 = video_probe[i + 7];
435 video_width = (uint16_t)(((uint16_t)b0 << 4) | (b1 >> 4));
436 video_height = (uint16_t)(((uint16_t)(b1 & 0x0F) << 8) | b2);
437 static const float rates[16] = {0, 23.976f, 24.0f, 25.0f,
438 29.97f, 30.0f, 50.0f, 59.94f,
439 60.0f, 0, 0, 0,
440 0, 0, 0, 0};
441 video_fps = rates[b3 & 0x0F];
442 video_header_parsed = true;
443 break;
444 }
445 }
446 if (video_header_parsed || n >= VIDEO_PROBE_MAX) {
447 video_probe.clear();
448 video_probe.shrink_to_fit();
449 }
450 }

454 void probeAudioHeader(const uint8_t *data, size_t len) {
455 size_t old_size = (size_t)audio_probe.size();
456 size_t take = old_size < 4 ? 4 - old_size : 0;
457 if (take > len) take = len;
458 for (size_t i = 0; i < take; i++) {
459 uint8_t b = data[i];
460 audio_probe.push_back(b);
461 }
462 if ((size_t)audio_probe.size() < 4) return;
463
464 uint8_t b0 = audio_probe[0], b1 = audio_probe[1], b2 = audio_probe[2],
465 b3 = audio_probe[3];
466 if (b0 == 0xFF && (b1 & 0xE0) == 0xE0) {
467 uint8_t version = (b1 >> 3) & 0x03; // 11=MPEG1, 10=MPEG2, 00=MPEG2.5
468 uint8_t sr_index = (b2 >> 2) & 0x03;
469 uint8_t mode = (b3 >> 6) & 0x03; // 11=mono
470 static const uint32_t rates_v1[4] = {44100, 48000, 32000, 0};
471 static const uint32_t rates_v2[4] = {22050, 24000, 16000, 0};
472 static const uint32_t rates_v25[4] = {11025, 12000, 8000, 0};
473 uint32_t rate = 0;
474 if (version == 0x03)
475 rate = rates_v1[sr_index];
476 else if (version == 0x02)
477 rate = rates_v2[sr_index];
478 else if (version == 0x00)
479 rate = rates_v25[sr_index];
480 if (rate > 0) {
481 audio_sample_rate = rate;
482 audio_channels = (mode == 0x03) ? 1 : 2;
483 audio_header_parsed = true;
484 }
485 } else {
486 // not a frame sync where we expected one - stop probing so we don't
487 // keep re-checking every write(); getAudioInfo() falls back to
488 // 44100/stereo
489 audio_header_parsed = true;
490 }
491 audio_probe.clear();
492 audio_probe.shrink_to_fit();
493 }

722 void writePackHeader(uint64_t scr_in) {

728 b[1] = (uint8_t)((scr >> 22) & 0xFF);
729 b[2] = (uint8_t)((((scr >> 15) & 0x7F) << 1) | 0x01);
730 b[3] = (uint8_t)((scr >> 7) & 0xFF);
731 b[4] = (uint8_t)(((scr & 0x7F) << 1) | 0x01);

738 void writeStreamEntry(uint8_t id, bool isVideo) {
739 // P-STD buffer bound: informational only - 46 (x1024 = ~46KB) for
740 // video, 4 (x128 = 512 bytes, ~1 audio frame) for audio
741 uint16_t bound = isVideo ? 46 : 4;
742 uint8_t b[3];
743 b[0] = id;
744 b[1] = (uint8_t)(0xC0 | (isVideo ? 0x20 : 0x00) | ((bound >> 8) & 0x1F));
745 b[2] = (uint8_t)(bound & 0xFF);
746 p_out->write(b, 3);

749 void writeSystemHeader() {
750 int num_streams = 1 + (has_audio ? 1 : 0);
751 uint16_t header_length = (uint16_t)(6 + 3 * num_streams);
752 static const uint8_t start[4] = {0x00, 0x00, 0x01,
753 MPG_SYSTEM_HEADER_START_CODE};
754 p_out->write(start, 4);
755 uint8_t len_b[2] = {(uint8_t)(header_length >> 8),
756 (uint8_t)header_length};
757 p_out->write(len_b, 2);
758
759 uint16_t audio_bound = has_audio ? 1 : 0;
760 uint16_t video_bound = 1;
761 uint8_t b[6];
762 // marker(1) + mux_rate(22) + marker(1)
763 b[0] = (uint8_t)(0x80 | ((MUX_RATE >> 15) & 0x7F));
764 b[1] = (uint8_t)((MUX_RATE >> 7) & 0xFF);
765 b[2] = (uint8_t)(((MUX_RATE & 0x7F) << 1) | 0x01);
766 // audio_bound(6) + fixed_flag(1)=0 + CSPS_flag(1)=0
767 b[3] = (uint8_t)((audio_bound & 0x3F) << 2);
768 // audio_locked(1)=0 + video_locked(1)=0 + marker(1)=1 + video_bound(5)
769 b[4] = (uint8_t)(0x20 | (video_bound & 0x1F));
770 b[5] = 0xFF; // reserved_byte
771 p_out->write(b, 6);
772
773 writeStreamEntry(MPG_VIDEO_STREAM_ID, true);
774 if (has_audio) writeStreamEntry(MPG_AUDIO_STREAM_ID, false);
775 }

777 void writeTimestampField(uint8_t id4, uint64_t ts) {
778 uint8_t b[5];

781 uint16_t low = (uint16_t)(((ts & 0x7FFF) << 1) | 0x01);
782 b[1] = (uint8_t)(mid >> 8);
783 b[2] = (uint8_t)mid;
784 b[3] = (uint8_t)(low >> 8);

786 p_out->write(b, 5);
787 }

790 size_t writeElementaryChunk(uint8_t stream_id, const uint8_t *data,
791 size_t len, bool withPts, uint64_t pts) {
792 writePackHeader(pts);
793 uint16_t optional_len = withPts ? 5 : 1;
794 uint16_t pes_len = (uint16_t)(len + optional_len);
795 uint8_t hdr[6] = {0,
796 0,
797 1,
798 stream_id,
799 (uint8_t)(pes_len >> 8),
800 (uint8_t)pes_len};
801 p_out->write(hdr, 6);
802 if (withPts) {
803 writeTimestampField(0x02, pts);
804 } else {
805 // avoid Print::write(uint8_t): some Print/Stream implementations
806 // (e.g. BaseStream-derived ones) buffer single-byte writes
807 // separately from the write(data, len) path below, which would
808 // reorder this marker byte after the payload - write it as a
809 // one-element array instead so it goes through the same path.
810 static const uint8_t no_ts = 0x0F;
811 p_out->write(&no_ts, 1);
812 }
813 return p_out->write(data, len);
814 }
815
819 size_t writeAccessUnit(uint8_t stream_id, const uint8_t *data, size_t len,
820 uint64_t pts) {
821 size_t offset = 0;
822 size_t written = 0;
823 bool first = true;
824 do {
825 size_t chunk = len - offset;
826 if (chunk > MAX_PES_PAYLOAD) chunk = MAX_PES_PAYLOAD;
827 written +=
828 writeElementaryChunk(stream_id, data + offset, chunk, first, pts);
829 offset += chunk;
830 first = false;
831 } while (offset < len);
832 return written;
833 }
834};
