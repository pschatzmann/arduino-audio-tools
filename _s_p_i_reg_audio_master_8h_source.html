122 size_t write(const uint8_t* data, size_t len) override {
123 if (!active || data == nullptr) return 0;
124 size_t written = 0;
125 while (written < len) {
126 size_t chunk = min(len - written, config.max_payload);
127 int space = availableForWrite();
128 if (space < (int)chunk) {
129 if (!config.is_blocking_write) {
130 chunk = space; // send what fits
131 } else {
132 delay(5);

134 }
135 }
136 if (chunk == 0) break;

138 written += chunk;
