330 bool tryStartNewBox(size_t bufferSize) {
331 if (parseOffset + 8 > bufferSize) return false;
332 char type[5];
333 box_seq = 0;
334
335 // get basic box information
336 parseOffset = checkParseOffset();
337 const uint8_t* p = buffer.data() + parseOffset;
338 uint32_t size32 = readU32(p);
339 strncpy(type, (char*)(p + 4), 4);
340 type[4] = '\0';
341
342 uint64_t boxSize = size32;
343 size_t headerSize = 8;
344 // A 32-bit size of 0 or 1 are reserved ISO/IEC 14496-12 special cases
345 // that a plain 32-bit compare against headerSize can't handle.
346 bool unbounded = false;
347 if (size32 == 1) {
348 // 64-bit "largesize" follows the type field
349 if (parseOffset + 16 > bufferSize) return false; // wait for largesize
350 boxSize = readU64(p + 8);
351 headerSize = 16;
352 } else if (size32 == 0) {
353 // box extends to the end of the file/stream (e.g. a streamed mdat
354 // written by an encoder that never seeks back to patch in the size)
355 unbounded = true;
356 }
357
358 if (!unbounded && boxSize < headerSize) return false;
359
360 int level = static_cast<int>(levelStack.size());
361
362 // mdat holds the audio payload and must be preceded by moov, otherwise
363 // sample/format info needed to decode it is missing.
364 if (level == 0 && strcmp(type, "mdat") == 0 && !moov_found) {
365 if (require_moov_before_mdat) {
366 LOGE(
367 "mdat box found before moov box: moov must precede mdat (e.g. "
368 "use 'ffmpeg -movflags +faststart') or this file will not play "
369 "correctly");
370 } else {
371 LOGW(
372 "mdat box found before moov box: moov should precede mdat (e.g. "
373 "use 'ffmpeg -movflags +faststart') - continuing anyway since "
374 "the requirement was opted out of");
375 }
376 }
377
378 bool is_container = isContainerBox(type);
379
380 if (is_container) {
381 if (unbounded) {
382 LOGE("Unsupported: container box '%s' with size 0 (extends to EOF)",
383 type);
384 return false;
385 }
386 handleContainerBox(type, boxSize, headerSize, level);
387 return true;
388 }
389
390 size_t payload_size =
391 unbounded ? SIZE_MAX : static_cast<size_t>(boxSize - headerSize);
392 if (!unbounded && parseOffset + boxSize <= bufferSize) {
393 // start with full buffer!
394 handleCompleteBox(type, p, headerSize, payload_size, level);
395 parseOffset += boxSize;
396 } else {
397 startIncrementalBox(type, p, headerSize, payload_size, level, bufferSize);
398 return false; // Wait for more data
399 }
400 return true;
401 }

514 bool continueIncrementalBox() {
515 size_t to_read = std::min((size_t)box_bytes_expected - box_bytes_received,
516 (size_t)buffer.available());
517 if (to_read == 0) return true;
518 strcpy(box.type, box_type);
519 ++box.id;
520 box.data = buffer.data();
521 box.size = box_bytes_expected;
522 box.data_size = box_bytes_expected;
523 box.available = to_read;
524 box.level = box_level;
525 box.file_offset = incremental_offset;
526 box.is_complete = (box_bytes_received + to_read == box_bytes_expected);
527 box.is_container = false;
528 box.is_incremental = true;
529 box.seq = ++box_seq;
530 processCallback(box);
531 box_bytes_received += to_read;
532 // fileOffset += to_read;
533 buffer.clearArray(to_read);
534 incremental_offset += to_read;
535
536 if (box_bytes_received >= box_bytes_expected) {
537 box_in_progress = false;
538 }
539 return false;

595 void processCallback(Box& box) {
596 bool is_called = false;
597 bool call_generic = true;

669 size_t checkParseOffset() {
670 size_t current = parseOffset;
671 const char* type = (char*)(buffer.data() + parseOffset + 4);
672 for (int j = 0; j < buffer.available() - parseOffset - 4; j += 4) {
673 if (isValidType(type, j)) {
