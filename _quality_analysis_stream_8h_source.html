409 void analyze(const uint8_t* data, size_t len) {
410 if (data == nullptr || len == 0) return;
411 // Feed the built-in tonality detector with the same raw audio (an
412 // externally-owned one set via setTonalitySource() is fed by the caller)
413 if (owns_tonality && tonality_enabled && p_tonality != nullptr) {
414 p_tonality->write(data, len);
415 }
416 switch (info.bits_per_sample) {
417 case 8:
418 analyzeT<int8_t>(data, len);
419 break;
420 case 16:
421 analyzeT<int16_t>(data, len);
422 break;
423 case 24:
424 analyzeT<int24_t>(data, len);
425 break;
426 case 32:
427 analyzeT<int32_t>(data, len);
428 break;
429 default:
430 LOGE("QualityAnalysisStream: unsupported bits_per_sample %d",
431 info.bits_per_sample);
432 break;
433 }
434 reportIfDue();
435 }

437 template <typename T>
438 void analyzeT(const uint8_t* buffer, size_t size) {
439 T* samples = (T*)buffer;
440 int count = size / sizeof(T);
441 int channels = info.channels > 0 ? info.channels : 1;
442 float abs_click = click_threshold * max_value;
443 float abs_silence = silence_threshold * max_value;
444 float abs_clip = max_value * (1.0f - clipping_margin);
445
446 for (int i = 0; i < count; i++) {
447 float val = static_cast<float>(static_cast<int>(samples[i]));
448 float abs_val = val < 0 ? -val : val;
449 int ch = i % channels;
450
451 stats_data.total_samples++;
452
453 // Click / pop detection
454 if (has_prev_sample) {
455 float delta = val - prev_sample[ch];
456 if (delta < 0) delta = -delta;
457 if (delta > abs_click) {
458 stats_data.click_count++;
459 if (callback) callback(QualityIssue::Click, stats_data.click_count);
460 }
461 }
462 prev_sample[ch] = val;
463
464 // Dropout detection
465 if (abs_val <= abs_silence) {
466 consecutive_silent++;
467 if (!in_dropout && consecutive_silent >= dropout_min_samples) {
468 in_dropout = true;
469 stats_data.dropout_count++;
470 if (callback)
471 callback(QualityIssue::Dropout, stats_data.dropout_count);
472 }
473 } else {
474 consecutive_silent = 0;
475 in_dropout = false;
476 }
477
478 // Clipping detection
479 if (abs_val >= abs_clip) {
480 consecutive_clipped++;
481 if (!in_clipping && consecutive_clipped >= clipping_min_samples) {
482 in_clipping = true;
483 stats_data.clipping_count++;
484 if (callback)
485 callback(QualityIssue::Clipping, stats_data.clipping_count);
486 }
487 } else {
488 consecutive_clipped = 0;
489 in_clipping = false;
490 }
491
492 // Signal vs. noise (RMS / noise-floor) detection
493 rms_sum_sq += (double)val * (double)val;
494 rms_block_count++;
495 if (rms_block_count >= rms_block_size) {
496 processRmsBlock();
497 }
498 }
499
500 // Mark that we have valid previous samples after the first buffer
501 if (count >= channels) has_prev_sample = true;
502 }
503
506 void processRmsBlock() {
507 float block_rms = sqrt((float)(rms_sum_sq / rms_block_count));
508 rms_sum_sq = 0;
509 rms_block_count = 0;
510
511 if (!noise_floor_initialized) {
512 stats_data.noise_floor = block_rms;
513 noise_floor_initialized = true;
514 } else if (block_rms < stats_data.noise_floor) {
515 stats_data.noise_floor +=
516 (block_rms - stats_data.noise_floor) * noise_floor_attack;
517 } else {
518 stats_data.noise_floor +=
519 (block_rms - stats_data.noise_floor) * noise_floor_release;
520 }
521
522 stats_data.rms_level = block_rms;
523 float floor = stats_data.noise_floor > 1.0f ? stats_data.noise_floor : 1.0f;
524 float level = block_rms > 1.0f ? block_rms : 1.0f;
525 stats_data.snr_db = 20.0f * log10(level / floor);
526
527 if (p_tonality != nullptr) {
528 stats_data.tonality_confidence = p_tonality->confidence(tonality_channel);
529 }
530
531 float threshold = stats_data.noise_floor * pow(10.0f, signal_margin_db / 20.0f);
532 bool level_ok = block_rms > threshold;
533 bool tonal_ok = p_tonality == nullptr ||
534 stats_data.tonality_confidence >= tonality_min_confidence;
535 if (level_ok && tonal_ok) {
536 consecutive_signal_blocks++;
537 consecutive_noise_blocks = 0;
538 if (!stats_data.signal_present &&
539 consecutive_signal_blocks >= signal_present_min_blocks) {
540 stats_data.signal_present = true;
541 stats_data.signal_detected_count++;
542 if (callback)
543 callback(QualityIssue::SignalDetected,
544 stats_data.signal_detected_count);
545 }
546 } else {
547 consecutive_noise_blocks++;
548 consecutive_signal_blocks = 0;
549 if (stats_data.signal_present &&
550 consecutive_noise_blocks >= signal_present_min_blocks) {
551 stats_data.signal_present = false;
552 stats_data.signal_lost_count++;
553 if (callback)
554 callback(QualityIssue::SignalLost, stats_data.signal_lost_count);
555 }
556 }
557 }
