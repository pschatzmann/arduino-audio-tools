289 // decode+render isn't keeping up right now. A non-keyframe is
290 // dropped outright instead of enqueued (see droppedFrameCount()) to
291 // catch back up without blocking this call, which would otherwise
292 // stall the caller's own thread (typically a demuxer's dispatch
293 // loop, delaying audio too). Two triggers: the queue is actually
294 // full, or the render task's most recently rendered frame was
295 // already more than setCatchUpThresholdFrames() frame periods late.
296 // Keyframes are never dropped here (losing one would desync every
297 // dependent P-frame) - if the queue can't fit one, this still blocks
298 // below.
299 bool queue_full = (size_t)queue.availableForWrite() < needed;
300 bool falling_behind =

303 (int32_t)(catch_up_threshold_frames * frame_period_ms);
304 if (!is_key && (queue_full || falling_behind)) {
305 dropped_frame_count++;
306 frame_count++;
307 // Rate-limited to one line per unbroken run of drops - this can
308 // trigger on every P-frame in a row for a long stretch, and
309 // logging each one individually is expensive enough (Serial I/O on
310 // this thread) to measurably worsen the backlog it's trying to
311 // fix.

317 queue_full ? "render queue full" : "catching up",
318 queue.availableForWrite(), (unsigned)queue.size(),
319 (unsigned)needed, (int)render_lateness_ms.load());
320 logged_drop_burst = true;
321 }
322 return len;
323 }
324 logged_drop_burst = false;
325
326 bool logged_wait = false;
327 while ((size_t)queue.availableForWrite() < needed) {
328 if (!logged_wait) {

337 }
338 queue.writeArray((const uint8_t*)&header, sizeof(header));
339 queue.writeArray(data, len);
340 if (is_key) queued_i_frame_count++;
341 LOGD(

658 bool drainQueueKeepingLastKeyframe(uint32_t& out_target_ms) {
659 bool found_key = false;
660 int count = 0;
661 while (true) {
662 if (!have_header) {
663 if ((size_t)queue.available() < sizeof(FrameHeader)) break;
664 queue.readArray((uint8_t*)&current_header, sizeof(FrameHeader));
665 have_header = true;
666 }
667 if ((size_t)queue.available() < current_header.size) break;
668
669 if (current_header.is_key) {
670 if (found_key) dropped_i_frame_count++; // superseded by this one
671 frame_buf.resize(current_header.size);
672 queue.readArray(frame_buf.data(), current_header.size);
673 out_target_ms = current_header.target_ms;
674 found_key = true;
675 queued_i_frame_count--;
676 } else {
677 uint8_t scratch[256];
678 size_t remaining = current_header.size;
679 while (remaining > 0) {
680 size_t chunk = remaining < sizeof(scratch) ? remaining : sizeof(scratch);
681 size_t got = (size_t)queue.readArray(scratch, chunk);
682 if (got == 0) break; // shouldn't happen; avoids ever spinning forever
683 remaining -= got;
684 }
685 dropped_frame_count++;
686 }
687 have_header = false;
688 if (++count % kDrainYieldEvery == 0) delay(1);
689 }
690 return found_key;
691 }

698 void taskLoop() {
699 if (!have_header) {
700 if ((size_t)queue.available() < sizeof(FrameHeader)) {
701 delay(1);
702 return;
703 }
704 queue.readArray((uint8_t*)&current_header, sizeof(FrameHeader));
705 have_header = true;
706 }
707 // The header may already be visible before its payload fully is -
708 // write() writes them as two separate writeArray() calls - so just
709 // wait for the rest; have_header stays true across calls in the
710 // meantime.
711 if ((size_t)queue.available() < current_header.size) {
712 delay(1);
713 return;
714 }
715 frame_buf.resize(current_header.size);
716 queue.readArray(frame_buf.data(), current_header.size);
717 uint32_t target_ms = current_header.target_ms;
718 bool is_key = current_header.is_key != 0;
719 have_header = false;
720 // Leaving the queue for rendering (not a discard - see
721 // drainQueueKeepingLastKeyframe() for that side of this same counter).
722 if (is_key) queued_i_frame_count--;
723
724 // Still waiting for a keyframe after a previous resync - skip
725 // straight past any P-frame without considering its schedule,
726 // reaching the next keyframe as fast as the queue allows.
727 if (awaiting_keyframe && !is_key) {
728 dropped_frame_count++;
729 frame_count++;
730 return;
731 }
732
733 if (!start_set) {
734 start_ms = clockMs();
735 start_set = true;
736 LOGI("PacedVideoOutput: playback anchored at %u ms (%s clock)",
737 (unsigned)start_ms, p_clock != nullptr ? "external audio" : "wall");
738 }
739 // scheduling_delay_ms (see setSchedulingDelayMs()) only ever pushes
740 // this later, correcting for the audio output's own buffering
741 // latency between "accepted by write()" and "actually audible".
742 uint32_t scheduled_ms = start_ms + target_ms + scheduling_delay_ms;
743 while ((int32_t)(scheduled_ms - clockMs()) > 0) {
744 delay(1);
745 }
746 // Captured right as the wait loop exits, before write()/flush() - a
747 // measure of how far behind schedule this frame already was when we
748 // started rendering it, separate from processMs (how long rendering
749 // itself then took).
750 int32_t lateness_ms = (int32_t)(clockMs() - scheduled_ms);
751 // Dropping non-keyframes alone can only prevent the backlog from
752 // growing, not shrink it - the producer can't deliver frames faster
753 // than real time. Past resync_threshold_ms, give up on the skipped
754 // content and jump the schedule anchor forward, making *this* frame
755 // "on time" - a visible jump instead of an ever-growing lag.
756 //
757 // Three independent triggers, since they can diverge: lateness_ms
758 // only reflects the frame we're about to render - dropped P-frames
759 // can keep that low even while newer bytes keep piling up in the
760 // queue, since overall render throughput can still trail the arrival
761 // rate. queue fill catches that directly, but only once
762 // substantially full; queued keyframe count (setMaxQueuedIFrames())
763 // catches it earlier still.
764 bool lateness_resync =
765 resync_threshold_ms > 0 && lateness_ms > (int32_t)resync_threshold_ms;
766 bool queue_resync =
767 resync_queue_fill_fraction > 0 && queue.size() > 0 &&
768 (float)queue.available() >=
769 resync_queue_fill_fraction * (float)queue.size();
770 bool iframe_resync = max_queued_i_frames > 0 &&
771 queued_i_frame_count.load() >= max_queued_i_frames;
772 if (lateness_resync || queue_resync || iframe_resync) {
773 // Keep the freshest keyframe found in the backlog instead of
774 // discarding it too (see drainQueueKeepingLastKeyframe()).
775 uint32_t new_target_ms = target_ms;
776 bool found_fresher_key = drainQueueKeepingLastKeyframe(new_target_ms);
777 LOGW(
778 "PacedVideoOutput: resyncing (%s) - %s instead of trying to "
779 "catch up",
780 lateness_resync
781 ? "lateness"
782 : (queue_resync ? "queue fill" : "I-frame backlog"),
783 found_fresher_key ? "jumping to the newest available keyframe"
784 : "jumping the schedule forward");
785 if (found_fresher_key) {
786 // Supersede whatever was already dequeued.
787 if (is_key) {
788 dropped_i_frame_count++;
789 } else {
790 dropped_frame_count++;
791 }
792 target_ms = new_target_ms;
793 is_key = true;
794 awaiting_keyframe = false;
795 } else {
796 // A resync can leave the decoder's reference state unreliable,
797 // so require a real (self-contained) keyframe before resuming -
798 // discard any P-frame in between (see the check below) - unless
799 // the frame we already have is one.
800 awaiting_keyframe = true;
801 }
802 // Recomputed for whichever frame we're actually about to render.
803 start_ms = clockMs() - target_ms - scheduling_delay_ms;
804 scheduled_ms = start_ms + target_ms + scheduling_delay_ms; // == clockMs() now
805 lateness_ms = 0;
806 }
807 if (awaiting_keyframe) {
808 if (!is_key) {
809 dropped_frame_count++;
810 frame_count++;
811 render_lateness_ms = 0;
812 return;
813 }
814 awaiting_keyframe = false;
815 }
816 // Published for write()'s proactive drop check (see
817 // setCatchUpThresholdFrames()) before doing the possibly-slow render
818 // call, so a caller blocked in write() sees it as soon as possible
819 // rather than only after this frame finishes too.
820 render_lateness_ms = lateness_ms;
821 uint32_t processStart = millis();
822 p_target->write(frame_buf.data(), frame_buf.size());
823 p_target->flush();
824 uint32_t processMs = millis() - processStart;
825 // See VideoOutput::hadOutput() - true for every synchronous decoder
826 // (the common case), but a decoder like MPGDecoder can legitimately
827 // do real decode work here without a picture actually reaching the
828 // screen yet (held back for B-picture display-order reordering), or
829 // emit an earlier held picture instead. Only count/time this call as
830 // a rendered frame when a picture was actually produced - otherwise
831 // outputFPS()/frameCountI()/frameCountP()/avgFrameMs() would
832 // overcount calls that did no real rendering work.
833 if (p_target->hadOutput()) {
834 if (!output_start_set) {
835 output_start_ms = millis();
836 output_start_set = true;
837 }
838 if (is_key) {
839 i_frame_count++;
840 i_frame_total_ms += processMs;
841 } else {
842 p_frame_count++;
843 p_frame_total_ms += processMs;
844 }
845 LOGD(
846 "PacedVideoOutput: rendered %s frame - scheduled=%u actual=%u "
847 "(%d ms late), process=%u ms",
848 is_key ? "I" : "P", (unsigned)scheduled_ms, (unsigned)clockMs(),
849 (int)lateness_ms, (unsigned)processMs);
850 } else {
851 LOGD(
852 "PacedVideoOutput: decoded %s frame, no picture emitted yet "
