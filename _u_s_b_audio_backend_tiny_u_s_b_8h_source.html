169 static usbd_class_driver_t driver;
170 driver.name = "AUDIO";
171 driver.init = []() { USBAudioDeviceBase::activeInstance().audiod_init(); };
172 driver.deinit = []() {
173 return USBAudioDeviceBase::activeInstance().audiod_deinit();
174 };
175 driver.reset = [](uint8_t rhport) {
176 USBAudioDeviceBase::activeInstance().audiod_reset(rhport);
177 };
178 driver.open = [](uint8_t rhport, tusb_desc_interface_t const* itf_desc,
179 uint16_t max_len) {
180 UsbInterfaceDescriptorView view =
181 USBAudioBackendTinyUSB::fromTusbInterface(itf_desc);
182 return USBAudioDeviceBase::activeInstance().audiod_open(
183 rhport, view, (uint8_t const*)itf_desc, max_len);
184 };
185 driver.control_xfer_cb = [](uint8_t rhport, uint8_t stage,
186 tusb_control_request_t const* request) {
187 UsbSetupPacket req = USBAudioBackendTinyUSB::fromTusbRequest(request);
188 return USBAudioDeviceBase::activeInstance().audiod_control_xfer_cb(
189 rhport, stage, req);
190 };
191 driver.xfer_cb = [](uint8_t rhport, uint8_t ep_addr, xfer_result_t result,
192 uint32_t xferred_bytes) {
193 UsbXferResult r = (result == XFER_RESULT_SUCCESS) ? UsbXferResult::Success
194 : (result == XFER_RESULT_STALLED) ? UsbXferResult::Stalled
195 : UsbXferResult::Failed;
196 return USBAudioDeviceBase::activeInstance().audiod_xfer_cb(
197 rhport, ep_addr, r, xferred_bytes);
198 };
199 driver.sof = [](uint8_t rhport, uint32_t frame_count) {
200 USBAudioDeviceBase::activeInstance().audiod_sof_isr(rhport, frame_count);
201 };
202
203 *count = 1;
204 return &driver;
205}

209// Custom driver registration — routes to whichever USBAudioDeviceBase
210// subclass was constructed (base or derived; set via s_active_ in the
211// constructor).
212//
213// Deliberately NOT marked `inline`: TinyUSB's own internal default
214// implementation of this function is a weak symbol, and this override must
215// win at link time by being a *strong* symbol. Marking a function `inline`
216// in C++ causes the compiler to emit it with weak linkage too (for ODR
217// merging across translation units) -- two weak symbols racing lets the
218// linker pick either one, and it was silently picking TinyUSB's built-in
219// default (which registers zero class drivers), so this driver never got
220// attached: every control request targeting our interfaces stalled and no
221// audio ever streamed. This header is only ever included from one
222// translation unit in practice (the platform-specific USB device header
223// pulled in by the sketch), so plain external linkage is correct and safe.
224extern "C" usbd_class_driver_t const* usbd_app_driver_get_cb(uint8_t* count) {
225 return audio_tools::usbAudioGetClassDriver(count);
226}
